     1→'use client'
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react'
     4→import { motion } from 'framer-motion'
     5→import {
     6→  Video,
     7→  VideoOff,
     8→  Mic,
     9→  MicOff,
    10→  Phone,
    11→  PhoneOff,
    12→  Clock,
    13→  MessageSquare,
    14→  Star,
    15→  Monitor,
    16→  FileText,
    17→  Plus,
    18→  Eye,
    19→  Circle,
    20→  CheckCircle2,
    21→} from 'lucide-react'
    22→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
    23→import { Button } from '@/components/ui/button'
    24→import { Badge } from '@/components/ui/badge'
    25→import { Input } from '@/components/ui/input'
    26→import { Label } from '@/components/ui/label'
    27→import { Textarea } from '@/components/ui/textarea'
    28→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    29→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog'
    30→import { Separator } from '@/components/ui/separator'
    31→import { Skeleton } from '@/components/ui/skeleton'
    32→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    33→import { useAppStore } from '@/lib/store'
    34→import type { Modality } from '@/lib/store'
    35→import { ModalityBadge } from '@/components/clinical/modality-badge'
    36→import { toast } from '@/hooks/use-toast'
    37→
    38→// ─── Types ────────────────────────────────────────────────────────────
    39→
    40→type SessionStatus =
    41→  | 'SCHEDULED'
    42→  | 'WAITING'
    43→  | 'IN_PROGRESS'
    44→  | 'COMPLETED'
    45→  | 'CANCELLED'
    46→  | 'FAILED'
    47→
    48→interface Practitioner {
    49→  id: string
    50→  name: string
    51→  specialty: string
    52→  modality: Modality
    53→}
    54→
    55→interface Session {
    56→  id: string
    57→  patientId: string
    58→  patientName: string
    59→  practitionerId: string
    60→  practitionerName: string
    61→  modality: Modality
    62→  status: SessionStatus
    63→  scheduledAt: string
    64→  startedAt: string | null
    65→  endedAt: string | null
    66→  durationMinutes: number | null
    67→  rating: number | null
    68→  feedback: string | null
    69→  isRecording: boolean
    70→  notes: string
    71→  prescriptionId: string | null
    72→  followUpScheduled: boolean
    73→  createdAt: string
    74→}
    75→
    76→interface Patient {
    77→  id: string
    78→  firstName: string
    79→  lastName: string
    80→}
    81→
    82→// ─── Style Maps ───────────────────────────────────────────────────────
    83→
    84→const statusStyles: Record<SessionStatus, string> = {
    85→  SCHEDULED: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300',
    86→  WAITING: 'bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200',
    87→  IN_PROGRESS: 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200',
    88→  COMPLETED: 'bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200',
    89→  CANCELLED: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
    90→  FAILED: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
    91→}
    92→
    93→const statusDotColor: Record<SessionStatus, string> = {
    94→  SCHEDULED: 'bg-slate-400',
    95→  WAITING: 'bg-amber-500 animate-pulse',
    96→  IN_PROGRESS: 'bg-green-500 animate-pulse',
    97→  COMPLETED: 'bg-teal-500',
    98→  CANCELLED: 'bg-red-500',
    99→  FAILED: 'bg-red-500',
   100→}
   101→
   102→const fadeSlide = {
   103→  initial: { opacity: 0, y: 12 },
   104→  animate: { opacity: 1, y: 0 },
   105→  transition: { duration: 0.3 },
   106→}
   107→
   108→const MODALITIES: Modality[] = ['ALLOPATHY', 'AYURVEDA', 'HOMEOPATHY']
   109→
   110→// ─── Component ────────────────────────────────────────────────────────
   111→
   112→export function TelemedicineSection() {
   113→  const { activeModality, setActiveModality } = useAppStore()
   114→
   115→  const [patients, setPatients] = useState<Patient[]>([])
   116→  const [practitioners, setPractitioners] = useState<Practitioner[]>([])
   117→  const [sessions, setSessions] = useState<Session[]>([])
   118→  const [loading, setLoading] = useState(true)
   119→
   120→  // Modality tab
   121→  const [modalityTab, setModalityTab] = useState<Modality>(activeModality)
   122→
   123→  // Filters
   124→  const [filterStatus, setFilterStatus] = useState<string>('ALL')
   125→
   126→  // Schedule dialog
   127→  const [scheduleOpen, setScheduleOpen] = useState(false)
   128→  const [formPatientId, setFormPatientId] = useState('')
   129→  const [formPractitionerId, setFormPractitionerId] = useState('')
   130→  const [formScheduledAt, setFormScheduledAt] = useState('')
   131→  const [formNotes, setFormNotes] = useState('')
   132→  const [submitting, setSubmitting] = useState(false)
   133→
   134→  // Session detail / consultation dialog
   135→  const [detailSession, setDetailSession] = useState<Session | null>(null)
   136→
   137→  // Consultation state (simulated)
   138→  const [inCall, setInCall] = useState(false)
   139→  const [callElapsed, setCallElapsed] = useState(0)
   140→  const [micOn, setMicOn] = useState(true)
   141→  const [camOn, setCamOn] = useState(true)
   142→  const [chatNotes, setChatNotes] = useState('')
   143→  const callTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   144→
   145→  // Post-consultation
   146→  const [showFeedback, setShowFeedback] = useState(false)
   147→  const [rating, setRating] = useState(0)
   148→  const [feedbackText, setFeedbackText] = useState('')
   149→
   150→  // ─── Data Loading ─────────────────────────────────────────────────
   151→
   152→  useEffect(() => {
   153→    fetch('/api/patients')
   154→      .then((r) => r.json())
   155→      .then((d) => {
   156→        const list = d.data ?? d.patients ?? []
   157→        setPatients(list)
   158→        if (list.length > 0) setFormPatientId(list[0].id)
   159→      })
   160→      .catch(() => {})
   161→  }, [])
   162→
   163→  useEffect(() => {
   164→    fetch('/api/telemedicine?practitioners=true')
   165→      .then((r) => r.json())
   166→      .then((d) => {
   167→        setPractitioners(d.practitioners ?? [])
   168→      })
   169→      .catch(() => {})
   170→  }, [])
   171→
   172→  const loadSessions = useCallback(async () => {
   173→    setLoading(true)
   174→    try {
   175→      const params = new URLSearchParams()
   176→      params.set('modality', modalityTab)
   177→      if (filterStatus !== 'ALL') params.set('status', filterStatus)
   178→      const res = await fetch(`/api/telemedicine?${params.toString()}`)
   179→      const data = await res.json()
   180→      setSessions(data.data ?? data.sessions ?? [])
   181→    } catch {
   182→      toast({ title: 'Error', description: 'Failed to load sessions', variant: 'destructive' })
   183→    } finally {
   184→      setLoading(false)
   185→    }
   186→  }, [modalityTab, filterStatus])
   187→
   188→  useEffect(() => { loadSessions() }, [loadSessions])
   189→
   190→  // ─── Call Timer ───────────────────────────────────────────────────
   191→
   192→  useEffect(() => {
   193→    if (inCall) {
   194→      callTimerRef.current = setInterval(() => {
   195→        setCallElapsed((prev) => prev + 1)
   196→      }, 1000)
   197→    } else {
   198→      if (callTimerRef.current) clearInterval(callTimerRef.current)
   199→      callTimerRef.current = null
   200→    }
   201→    return () => {
   202→      if (callTimerRef.current) clearInterval(callTimerRef.current)
   203→    }
   204→  }, [inCall])
   205→
   206→  const formatTime = (seconds: number) => {
   207→    const m = Math.floor(seconds / 60).toString().padStart(2, '0')
   208→    const s = (seconds % 60).toString().padStart(2, '0')
   209→    return `${m}:${s}`
   210→  }
   211→
   212→  // ─── Stats ────────────────────────────────────────────────────────
   213→
   214→  const stats = {
   215→    total: sessions.length,
   216→    today: sessions.filter((s) => {
   217→      const d = new Date(s.scheduledAt)
   218→      const now = new Date()
   219→      return d.toDateString() === now.toDateString()
   220→    }).length,
   221→    inProgress: sessions.filter((s) => s.status === 'IN_PROGRESS').length,
   222→    avgRating: sessions.filter((s) => s.rating !== null).length > 0
   223→      ? (sessions.filter((s) => s.rating !== null).reduce((a, s) => a + (s.rating ?? 0), 0) /
   224→          sessions.filter((s) => s.rating !== null).length).toFixed(1)
   225→      : '—',
   226→  }
   227→
   228→  // ─── Actions ──────────────────────────────────────────────────────
   229→
   230→  const handleSchedule = async () => {
   231→    if (!formPatientId || !formPractitionerId || !formScheduledAt) {
   232→      toast({ title: 'Validation', description: 'Fill all required fields', variant: 'destructive' })
   233→      return
   234→    }
   235→    setSubmitting(true)
   236→    try {
   237→      const res = await fetch('/api/telemedicine', {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          patientId: formPatientId,
   242→          practitionerId: formPractitionerId,
   243→          modality: modalityTab,
   244→          scheduledAt: formScheduledAt,
   245→          notes: formNotes,
   246→        }),
   247→      })
   248→      if (res.ok) {
   249→        toast({ title: 'Session Scheduled', description: 'Telemedicine session created' })
   250→        setScheduleOpen(false)
   251→        setFormNotes('')
   252→        setFormScheduledAt('')
   253→        loadSessions()
   254→      } else {
   255→        toast({ title: 'Error', description: 'Failed to schedule session', variant: 'destructive' })
   256→      }
   257→    } catch {
   258→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   259→    } finally {
   260→      setSubmitting(false)
   261→    }
   262→  }
   263→
   264→  const handleJoinCall = (session: Session) => {
   265→    setDetailSession(session)
   266→    setInCall(true)
   267→    setCallElapsed(0)
   268→    setMicOn(true)
   269→    setCamOn(true)
   270→    setChatNotes('')
   271→    setShowFeedback(false)
   272→    toast({ title: 'Joined Call', description: `Connected with ${session.patientName}` })
   273→  }
   274→
   275→  const handleEndCall = async () => {
   276→    setInCall(false)
   277→    // Update session to completed
   278→    if (detailSession) {
   279→      try {
   280→        await fetch('/api/telemedicine', {
   281→          method: 'PUT',
   282→          headers: { 'Content-Type': 'application/json' },
   283→          body: JSON.stringify({ id: detailSession.id, status: 'COMPLETED', durationMinutes: Math.round(callElapsed / 60) }),
   284→        })
   285→      } catch { /* silent */ }
   286→    }
   287→    setShowFeedback(true)
   288→    toast({ title: 'Call Ended', description: `Duration: ${formatTime(callElapsed)}` })
   289→  }
   290→
   291→  const handleSubmitFeedback = async () => {
   292→    if (detailSession) {
   293→      try {
   294→        await fetch('/api/telemedicine', {
   295→          method: 'PUT',
   296→          headers: { 'Content-Type': 'application/json' },
   297→          body: JSON.stringify({ id: detailSession.id, rating, feedback: feedbackText }),
   298→        })
   299→        toast({ title: 'Feedback Submitted' })
   300→      } catch { /* silent */ }
   301→    }
   302→    setShowFeedback(false)
   303→    setDetailSession(null)
   304→    setRating(0)
   305→    setFeedbackText('')
   306→    loadSessions()
   307→  }
   308→
   309→  const handleCancelSession = async (id: string) => {
   310→    try {
   311→      const res = await fetch('/api/telemedicine', {
   312→        method: 'PUT',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({ id, status: 'CANCELLED' }),
   315→      })
   316→      if (res.ok) {
   317→        toast({ title: 'Session Cancelled' })
   318→        loadSessions()
   319→        setDetailSession(null)
   320→      }
   321→    } catch {
   322→      toast({ title: 'Error', description: 'Failed to cancel', variant: 'destructive' })
   323→    }
   324→  }
   325→
   326→  // ─── Render ───────────────────────────────────────────────────────
   327→
   328→  return (
   329→    <motion.div {...fadeSlide} className="space-y-4">
   330→      {/* Stats Cards */}
   331→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   332→        {[
   333→          { label: 'Total Sessions', value: stats.total, icon: Monitor, color: 'text-slate-600 dark:text-slate-400' },
   334→          { label: 'Today', value: stats.today, icon: Clock, color: 'text-amber-600 dark:text-amber-400' },
   335→          { label: 'In Progress', value: stats.inProgress, icon: Video, color: 'text-green-600 dark:text-green-400' },
   336→          { label: 'Avg Rating', value: stats.avgRating, icon: Star, color: 'text-yellow-600 dark:text-yellow-400' },
   337→        ].map((s) => (
   338→          <Card key={s.label}>
   339→            <CardContent className="p-4 flex items-center gap-3">
   340→              <s.icon className={`h-5 w-5 ${s.color}`} />
   341→              <div>
   342→                <p className="text-2xl font-bold">{s.value}</p>
   343→                <p className="text-xs text-muted-foreground">{s.label}</p>
   344→              </div>
   345→            </CardContent>
   346→          </Card>
   347→        ))}
   348→      </div>
   349→
   350→      {/* Modality Tabs + Toolbar */}
   351→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   352→        <Tabs
   353→          value={modalityTab}
   354→          onValueChange={(v) => { setModalityTab(v as Modality); setActiveModality(v as Modality) }}
   355→        >
   356→          <TabsList>
   357→            <TabsTrigger value="ALLOPATHY">Allopathy</TabsTrigger>
   358→            <TabsTrigger value="AYURVEDA">Ayurveda</TabsTrigger>
   359→            <TabsTrigger value="HOMEOPATHY">Homeopathy</TabsTrigger>
   360→          </TabsList>
   361→        </Tabs>
   362→
   363→        <Select value={filterStatus} onValueChange={setFilterStatus}>
   364→          <SelectTrigger className="w-32 h-8"><SelectValue placeholder="Status" /></SelectTrigger>
   365→          <SelectContent>
   366→            <SelectItem value="ALL">All Status</SelectItem>
   367→            <SelectItem value="SCHEDULED">Scheduled</SelectItem>
   368→            <SelectItem value="WAITING">Waiting</SelectItem>
   369→            <SelectItem value="IN_PROGRESS">In Progress</SelectItem>
   370→            <SelectItem value="COMPLETED">Completed</SelectItem>
   371→            <SelectItem value="CANCELLED">Cancelled</SelectItem>
   372→            <SelectItem value="FAILED">Failed</SelectItem>
   373→          </SelectContent>
   374→        </Select>
   375→
   376→        <div className="sm:ml-auto">
   377→          <Dialog open={scheduleOpen} onOpenChange={setScheduleOpen}>
   378→            <DialogTrigger asChild>
   379→              <Button className="gap-2">
   380→                <Plus className="h-4 w-4" />
   381→                Schedule Session
   382→              </Button>
   383→            </DialogTrigger>
   384→            <DialogContent className="max-w-lg">
   385→              <DialogHeader>
   386→                <DialogTitle className="flex items-center gap-2">
   387→                  <Video className="h-5 w-5" />
   388→                  Schedule Telemedicine Session
   389→                </DialogTitle>
   390→                <DialogDescription>
   391→                  For <ModalityBadge modality={modalityTab} /> modality
   392→                </DialogDescription>
   393→              </DialogHeader>
   394→              <div className="grid gap-4 py-4">
   395→                <div className="space-y-2">
   396→                  <Label>Patient</Label>
   397→                  <Select value={formPatientId} onValueChange={setFormPatientId}>
   398→                    <SelectTrigger><SelectValue placeholder="Select patient" /></SelectTrigger>
   399→                    <SelectContent>
   400→                      {patients.map((p) => (
   401→                        <SelectItem key={p.id} value={p.id}>
   402→                          {p.firstName} {p.lastName}
   403→                        </SelectItem>
   404→                      ))}
   405→                    </SelectContent>
   406→                  </Select>
   407→                </div>
   408→
   409→                <div className="space-y-2">
   410→                  <Label>Practitioner</Label>
   411→                  <Select value={formPractitionerId} onValueChange={setFormPractitionerId}>
   412→                    <SelectTrigger><SelectValue placeholder="Select practitioner" /></SelectTrigger>
   413→                    <SelectContent>
   414→                      {practitioners
   415→                        .filter((p) => p.modality === modalityTab)
   416→                        .map((p) => (
   417→                          <SelectItem key={p.id} value={p.id}>
   418→                            {p.name} — {p.specialty}
   419→                          </SelectItem>
   420→                        ))}
   421→                    </SelectContent>
   422→                  </Select>
   423→                </div>
   424→
   425→                <div className="space-y-2">
   426→                  <Label>Scheduled Time</Label>
   427→                  <Input
   428→                    type="datetime-local"
   429→                    value={formScheduledAt}
   430→                    onChange={(e) => setFormScheduledAt(e.target.value)}
   431→                  />
   432→                </div>
   433→
   434→                <div className="space-y-2">
   435→                  <Label>Notes</Label>
   436→                  <Textarea
   437→                    value={formNotes}
   438→                    onChange={(e) => setFormNotes(e.target.value)}
   439→                    placeholder="Reason for consultation..."
   440→                    rows={2}
   441→                  />
   442→                </div>
   443→              </div>
   444→              <DialogFooter>
   445→                <Button variant="outline" onClick={() => setScheduleOpen(false)}>Cancel</Button>
   446→                <Button onClick={handleSchedule} disabled={submitting}>
   447→                  {submitting ? 'Scheduling...' : 'Schedule'}
   448→                </Button>
   449→              </DialogFooter>
   450→            </DialogContent>
   451→          </Dialog>
   452→        </div>
   453→      </div>
   454→
   455→      {/* Session List */}
   456→      <Card>
   457→        <CardHeader>
   458→          <CardTitle className="text-base flex items-center gap-2">
   459→            <Monitor className="h-5 w-5" />
   460→            Sessions
   461→            <ModalityBadge modality={modalityTab} className="ml-1" />
   462→          </CardTitle>
   463→          <CardDescription>Telemedicine sessions and remote consultations</CardDescription>
   464→        </CardHeader>
   465→        <CardContent>
   466→          {loading ? (
   467→            <div className="space-y-2">
   468→              {[1, 2, 3, 4].map((i) => (
   469→                <Skeleton key={i} className="h-14 w-full" />
   470→              ))}
   471→            </div>
   472→          ) : (
   473→            <div className="space-y-2 max-h-96 overflow-y-auto">
   474→              {sessions.map((session) => (
   475→                <div
   476→                  key={session.id}
   477→                  className="flex items-center gap-3 p-3 rounded-lg border hover:bg-muted/50 transition-colors"
   478→                >
   479→                  {/* Status dot */}
   480→                  <div className="shrink-0">
   481→                    <Circle className={`h-3 w-3 fill-current ${statusDotColor[session.status]}`} />
   482→                  </div>
   483→
   484→                  {/* Info */}
   485→                  <div className="flex-1 min-w-0">
   486→                    <p className="font-medium text-sm">{session.patientName} ↔ {session.practitionerName}</p>
   487→                    <p className="text-xs text-muted-foreground">
   488→                      {new Date(session.scheduledAt).toLocaleString()}
   489→                      {session.durationMinutes && ` • ${session.durationMinutes} min`}
   490→                      {session.isRecording && ' • 🔴 Recording'}
   491→                    </p>
   492→                  </div>
   493→
   494→                  {/* Badges */}
   495→                  <Badge variant="outline" className={statusStyles[session.status]}>
   496→                    {session.status.replace('_', ' ')}
   497→                  </Badge>
   498→
   499→                  {/* Actions */}
   500→                  <div className="flex items-center gap-1">
   501→                    <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={() => { setDetailSession(session); setShowFeedback(false); setInCall(false) }}>
   502→                      <Eye className="h-3.5 w-3.5" />
   503→                    </Button>
   504→                    {(session.status === 'SCHEDULED' || session.status === 'WAITING') && (
   505→                      <Button
   506→                        size="sm"
   507→                        variant="outline"
   508→                        className="h-7 gap-1 text-xs text-green-700 dark:text-green-400"
   509→                        onClick={() => handleJoinCall(session)}
   510→                      >
   511→                        <Video className="h-3 w-3" /> Join
   512→                      </Button>
   513→                    )}
   514→                  </div>
   515→                </div>
   516→              ))}
   517→              {sessions.length === 0 && (
   518→                <div className="text-center text-muted-foreground py-8">
   519→                  No sessions found for this modality
   520→                </div>
   521→              )}
   522→            </div>
   523→          )}
   524→        </CardContent>
   525→      </Card>
   526→
   527→      {/* Detail / Consultation Dialog */}
   528→      <Dialog
   529→        open={!!detailSession || inCall || showFeedback}
   530→        onOpenChange={(open) => {
   531→          if (!open) {
   532→            if (inCall) {
   533→              setInCall(false)
   534→            } else {
   535→              setDetailSession(null)
   536→              setShowFeedback(false)
   537→              setInCall(false)
   538→            }
   539→          }
   540→        }}
   541→      >
   542→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   543→          {detailSession && (
   544→            <>
   545→              <DialogHeader>
   546→                <DialogTitle className="flex items-center gap-2">
   547→                  {inCall ? <Video className="h-5 w-5 text-green-600" /> : <Monitor className="h-5 w-5" />}
   548→                  {inCall ? 'Live Consultation' : 'Session Detail'}
   549→                </DialogTitle>
   550→                <DialogDescription>
   551→                  {detailSession.patientName} ↔ {detailSession.practitionerName}
   552→                  <ModalityBadge modality={detailSession.modality} className="ml-2" />
   553→                </DialogDescription>
   554→              </DialogHeader>
   555→
   556→              {inCall ? (
   557→                /* ─── Video Call UI ─── */
   558→                <div className="grid gap-4 py-4">
   559→                  {/* Video area */}
   560→                  <div className="relative bg-slate-900 rounded-xl aspect-video flex items-center justify-center overflow-hidden">
   561→                    {camOn ? (
   562→                      <div className="text-center text-slate-400">
   563→                        <Video className="h-16 w-16 mx-auto mb-2" />

   157→   157→  return 'critical'
   158→   158→}

   179→   179→export function DrugInteractionsSection() {
   180→   180→  // Medicine selection state
   181→   181→  const [searchQuery, setSearchQuery] = useState('')
   182→   182→  const [searchResults, setSearchResults] = useState<MedicineSearchResult[]>([])
   183→   183→  const [selectedMedicines, setSelectedMedicines] = useState<SelectedMedicine[]>([])
   184→   184→  const [searching, setSearching] = useState(false)
   185→   185→  const [showDropdown, setShowDropdown] = useState(false)
   186→   186→  const searchInputRef = useRef<HTMLInputElement>(null)
   187→   187→  const dropdownRef = useRef<HTMLDivElement>(null)
   188→   188→
   189→   189→  // Results state
   190→   190→  const [results, setResults] = useState<DrugInteractionResult | null>(null)
   191→   191→  const [loading, setLoading] = useState(false)
   192→   192→  const [showAllInteractions, setShowAllInteractions] = useState(false)
   193→   193→
   194→   194→  // ── Search medicines ───────────────────────
   195→   195→
   196→   196→  const searchMedicines = useCallback(async (query: string) => {
   197→   197→    if (!query.trim() || query.length < 2) {
   198→   198→      setSearchResults([])
   199→   199→      setShowDropdown(false)
   200→   200→      return
   201→   201→    }
   202→   202→
   203→   203→    setSearching(true)
   204→   204→    try {
   205→   205→      const res = await fetch(`/api/medicines?search=${encodeURIComponent(query)}&limit=10`)
   206→   206→      const data = await res.json()
   207→   207→      const list: MedicineSearchResult[] = data.data ?? data.medicines ?? []
   208→   208→      // Filter out already selected
   209→   209→      const selectedIds = selectedMedicines.map((m) => m.id)
   210→   210→      const filtered = list.filter((m) => !selectedIds.includes(m.id))
   211→   211→      setSearchResults(filtered)
   212→   212→      setShowDropdown(filtered.length > 0)
   213→   213→    } catch {
   214→   214→      setSearchResults([])
   215→   215→    } finally {
   216→   216→      setSearching(false)
   217→   217→    }
   218→   218→  }, [selectedMedicines])
   219→   219→
   220→   220→  // Debounced search
   221→   221→  useEffect(() => {
   222→   222→    const timer = setTimeout(() => {
   223→   223→      searchMedicines(searchQuery)
   224→   224→    }, 300)
   225→   225→    return () => clearTimeout(timer)
   226→   226→  }, [searchQuery, searchMedicines])
   227→   227→
   228→   228→  // Close dropdown on outside click
   229→   229→  useEffect(() => {
   230→   230→    const handler = (e: MouseEvent) => {
   231→   231→      if (
   232→   232→        dropdownRef.current &&
   233→   233→        !dropdownRef.current.contains(e.target as Node) &&
   234→   234→        searchInputRef.current &&
   235→   235→        !searchInputRef.current.contains(e.target as Node)
   236→   236→      ) {
   237→   237→        setShowDropdown(false)
   238→   238→      }
   239→   239→    }
   240→   240→    document.addEventListener('mousedown', handler)
   241→   241→    return () => document.removeEventListener('mousedown', handler)
   242→   242→  }, [])
   243→   243→
   244→   244→  // ── Medicine management ────────────────────
   245→   245→
   246→   246→  const addMedicine = (medicine: MedicineSearchResult) => {
   247→   247→    if (!selectedMedicines.find((m) => m.id === medicine.id)) {
   248→   248→      setSelectedMedicines((prev) => [...prev, { id: medicine.id, name: medicine.name }])
   249→   249→    }
   250→   250→    setSearchQuery('')
   251→   251→    setSearchResults([])
   252→   252→    setShowDropdown(false)
   253→   253→    searchInputRef.current?.focus()
   254→   254→  }
   255→   255→
   256→   256→  const removeMedicine = (id: string) => {
   257→   257→    setSelectedMedicines((prev) => prev.filter((m) => m.id !== id))
   258→   258→  }
   259→   259→
   260→   260→  // ── Check interactions ─────────────────────
   261→   261→
   262→   262→  const handleCheckInteractions = async () => {
   263→   263→    if (selectedMedicines.length < 2) {
   264→   264→      toast({
   265→   265→        title: 'Add more medicines',
   266→   266→        description: 'Please add at least 2 medicines to check interactions',
   267→   267→        variant: 'destructive',
   268→   268→      })
   269→   269→      return
   270→   270→    }
   271→   271→
   272→   272→    setLoading(true)
   273→   273→    setResults(null)
   274→   274→    setShowAllInteractions(false)
   275→   275→
   276→   276→    try {
   277→   277→      const res = await fetch('/api/drug-interactions', {
   278→   278→        method: 'POST',
   279→   279→        headers: { 'Content-Type': 'application/json' },
   280→   280→        body: JSON.stringify({
   281→   281→          medicineIds: selectedMedicines.map((m) => m.id),
   282→   282→          medicineNames: selectedMedicines.map((m) => m.name),
   283→   283→        }),
   284→   284→      })
   285→   285→      const data = await res.json()
   286→   286→      const result: DrugInteractionResult = data.data ?? data
   287→   287→      setResults(result)
   288→   288→
   289→   289→      const riskLabel = RISK_LEVEL_CONFIG[result.riskLevel]?.label ?? result.riskLevel
   290→   290→      toast({
   291→   291→        title: 'Interaction Check Complete',
   292→   292→        description: `Safety score: ${result.safetyScore}/100 — ${riskLabel}`,
   293→   293→      })
   294→   294→    } catch {
   295→   295→      toast({ title: 'Error', description: 'Failed to check drug interactions', variant: 'destructive' })
   296→   296→    } finally {
   297→   297→      setLoading(false)
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  // ── Helpers ────────────────────────────────
   302→   302→
   303→   303→  const safetyTier = results ? getSafetyTier(results.safetyScore) : null
   304→   304→  const tierColors = safetyTier ? SAFETY_SCORE_COLORS[safetyTier] : null
   305→   305→
   306→   306→  const visibleInteractions = results?.pairwiseInteractions
   307→   307→    ? showAllInteractions
   308→   308→      ? results.pairwiseInteractions
   309→   309→      : results.pairwiseInteractions.slice(0, 5)
   310→   310→    : []
   311→   311→
   312→   312→  // ── Render ─────────────────────────────────
   313→   313→
   314→   314→  return (
   315→   315→    <TooltipProvider>
   316→   316→      <motion.div {...fadeSlide} className="space-y-6">
   317→   317→        {/* ── Input Card ─────────────────────── */}
   318→   318→        <Card>
   319→   319→          <CardHeader>
   320→   320→            <CardTitle className="text-lg flex items-center gap-2">
   321→   321→              <Pill className="h-5 w-5" />
   322→   322→              Drug Interaction Checker
   323→   323→            </CardTitle>
   324→   324→            <CardDescription>
   325→   325→              Add medicines to check for potential drug-drug interactions and safety concerns
   326→   326→            </CardDescription>
   327→   327→          </CardHeader>
   328→   328→          <CardContent className="space-y-5">
   329→   329→            {/* Medicine Search Input */}
   330→   330→            <div className="space-y-2">
   331→   331→              <Label>Add Medicines</Label>
   332→   332→              <div className="relative">
   333→   333→                <div className="relative">
   334→   334→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   335→   335→                  <Input
   336→   336→                    ref={searchInputRef}
   337→   337→                    type="text"
   338→   338→                    value={searchQuery}
   339→   339→                    onChange={(e) => setSearchQuery(e.target.value)}
   340→   340→                    onFocus={() => {
   341→   341→                      if (searchResults.length > 0) setShowDropdown(true)
   342→   342→                    }}
   343→   343→                    placeholder="Search medicines by name..."
   344→   344→                    className="pl-9"
   345→   345→                  />
   346→   346→                  {searching && (
   347→   347→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 animate-spin text-muted-foreground" />
   348→   348→                  )}
   349→   349→                </div>
   350→   350→
   351→   351→                {/* Search Dropdown */}
   352→   352→                <AnimatePresence>
   353→   353→                  {showDropdown && searchResults.length > 0 && (
   354→   354→                    <motion.div
   355→   355→                      ref={dropdownRef}
   356→   356→                      initial={{ opacity: 0, y: -4 }}
   357→   357→                      animate={{ opacity: 1, y: 0 }}
   358→   358→                      exit={{ opacity: 0, y: -4 }}
   359→   359→                      className="absolute top-full left-0 right-0 mt-1 border rounded-lg bg-popover shadow-lg z-20 overflow-hidden"
   360→   360→                    >
   361→   361→                      <ScrollArea className="max-h-56">
   362→   362→                        {searchResults.map((medicine) => (
   363→   363→                          <button
   364→   364→                            key={medicine.id}
   365→   365→                            type="button"
   366→   366→                            onClick={() => addMedicine(medicine)}
   367→   367→                            className="w-full text-left px-3 py-2.5 text-sm hover:bg-accent transition-colors flex items-center gap-3"
   368→   368→                          >
   369→   369→                            <Plus className="h-4 w-4 text-muted-foreground shrink-0" />
   370→   370→                            <div className="flex-1 min-w-0">
   371→   371→                              <span className="font-medium">{medicine.name}</span>
   372→   372→                              {medicine.genericName && (
   373→   373→                                <span className="text-muted-foreground ml-1">({medicine.genericName})</span>
   374→   374→                              )}
   375→   375→                            </div>
   376→   376→                            <Badge variant="outline" className="text-xs shrink-0">
   377→   377→                              {medicine.modality}
   378→   378→                            </Badge>
   379→   379→                          </button>
   380→   380→                        ))}
   381→   381→                      </ScrollArea>
   382→   382→                    </motion.div>
   383→   383→                  )}
   384→   384→                </AnimatePresence>
   385→   385→              </div>
   386→   386→            </div>
   387→   387→
   388→   388→            {/* Selected Medicine Tags */}
   389→   389→            {selectedMedicines.length > 0 && (
   390→   390→              <div className="space-y-2">
   391→   391→                <Label className="text-muted-foreground">
   392→   392→                  Selected Medicines ({selectedMedicines.length})
   393→   393→                </Label>
   394→   394→                <div className="flex flex-wrap gap-2">
   395→   395→                  <AnimatePresence>
   396→   396→                    {selectedMedicines.map((med) => (
   397→   397→                      <motion.span
   398→   398→                        key={med.id}
   399→   399→                        initial={{ opacity: 0, scale: 0.8 }}
   400→   400→                        animate={{ opacity: 1, scale: 1 }}
   401→   401→                        exit={{ opacity: 0, scale: 0.8 }}
   402→   402→                        className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   403→   403→                      >
   404→   404→                        <Pill className="h-3.5 w-3.5" />
   405→   405→                        {med.name}
   406→   406→                        <button
   407→   407→                          type="button"
   408→   408→                          onClick={() => removeMedicine(med.id)}
   409→   409→                          className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   410→   410→                        >
   411→   411→                          <X className="h-3 w-3" />
   412→   412→                        </button>
   413→   413→                      </motion.span>
   414→   414→                    ))}
   415→   415→                  </AnimatePresence>
   416→   416→                </div>
   417→   417→              </div>
   418→   418→            )}
   419→   419→
   420→   420→            {/* Action Buttons */}
   421→   421→            <div className="flex items-center gap-3">
   422→   422→              <Button
   423→   423→                onClick={handleCheckInteractions}
   424→   424→                disabled={selectedMedicines.length < 2 || loading}
   425→   425→                className="gap-2 min-w-[200px]"
   426→   426→              >
   427→   427→                {loading ? (
   428→   428→                  <>
   429→   429→                    <Loader2 className="h-4 w-4 animate-spin" />
   430→   430→                    Checking...
   431→   431→                  </>
   432→   432→                ) : (
   433→   433→                  <>
   434→   434→                    <ShieldAlert className="h-4 w-4" />
   435→   435→                    Check Interactions
   436→   436→                  </>
   437→   437→                )}
   438→   438→              </Button>
   439→   439→              {selectedMedicines.length > 0 && (
   440→   440→                <Button
   441→   441→                  variant="ghost"
   442→   442→                  size="sm"
   443→   443→                  onClick={() => {
   444→   444→                    setSelectedMedicines([])
   445→   445→                    setSearchQuery('')
   446→   446→                    setResults(null)
   447→   447→                  }}
   448→   448→                  className="text-muted-foreground"
   449→   449→                >
   450→   450→                  Clear all
   451→   451→                </Button>
   452→   452→              )}
   453→   453→            </div>
   454→   454→
   455→   455→            {selectedMedicines.length > 0 && selectedMedicines.length < 2 && (
   456→   456→              <p className="text-xs text-muted-foreground flex items-center gap-1">
   457→   457→                <Info className="h-3 w-3" />
   458→   458→                Add at least 2 medicines to check for interactions
   459→   459→              </p>
   460→   460→            )}
   461→   461→          </CardContent>
   462→   462→        </Card>
   463→   463→
   464→   464→        {/* ── Loading Skeleton ────────────────── */}
   465→   465→        <AnimatePresence>
   466→   466→          {loading && (
   467→   467→            <motion.div
   468→   468→              initial={{ opacity: 0 }}
   469→   469→              animate={{ opacity: 1 }}
   470→   470→              exit={{ opacity: 0 }}
   471→   471→              className="space-y-4"
   472→   472→            >
   473→   473→              <Card>
   474→   474→                <CardContent className="p-6 space-y-4">
   475→   475→                  <Skeleton className="h-8 w-64" />
   476→   476→                  <Skeleton className="h-4 w-full" />
   477→   477→                  <Skeleton className="h-32 w-full" />
   478→   478→                  <Skeleton className="h-24 w-full" />
   479→   479→                </CardContent>

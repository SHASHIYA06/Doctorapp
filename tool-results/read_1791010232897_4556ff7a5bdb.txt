   149→   149→
   150→   150→export function SymptomCheckerSection() {
   151→   151→  const { activeModality } = useAppStore()
   152→   152→
   153→   153→  // Form state
   154→   154→  const [symptoms, setSymptoms] = useState<string[]>([])
   155→   155→  const [symptomInput, setSymptomInput] = useState('')
   156→   156→  const [age, setAge] = useState('')
   157→   157→  const [gender, setGender] = useState('')
   158→   158→  const [modality, setModality] = useState<Modality>(activeModality)
   159→   159→
   160→   160→  // Suggestion state
   161→   161→  const [showSuggestions, setShowSuggestions] = useState(false)
   162→   162→  const [filteredSuggestions, setFilteredSuggestions] = useState<string[]>([])
   163→   163→  const inputRef = useRef<HTMLInputElement>(null)
   164→   164→  const suggestionsRef = useRef<HTMLDivElement>(null)
   165→   165→
   166→   166→  // Results state
   167→   167→  const [results, setResults] = useState<SymptomCheckResult | null>(null)
   168→   168→  const [loading, setLoading] = useState(false)
   169→   169→  const [expandedIssue, setExpandedIssue] = useState<string | null>(null)
   170→   170→
   171→   171→  // ── Symptom management ─────────────────────
   172→   172→
   173→   173→  const addSymptom = (symptom: string) => {
   174→   174→    const trimmed = symptom.trim().toLowerCase()
   175→   175→    if (trimmed && !symptoms.includes(trimmed)) {
   176→   176→      setSymptoms((prev) => [...prev, trimmed])
   177→   177→    }
   178→   178→    setSymptomInput('')
   179→   179→    setShowSuggestions(false)
   180→   180→    inputRef.current?.focus()
   181→   181→  }
   182→   182→
   183→   183→  const removeSymptom = (symptom: string) => {
   184→   184→    setSymptoms((prev) => prev.filter((s) => s !== symptom))
   185→   185→  }
   186→   186→
   187→   187→  const handleSymptomKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   188→   188→    if (e.key === 'Enter' && symptomInput.trim()) {
   189→   189→      e.preventDefault()
   190→   190→      addSymptom(symptomInput)
   191→   191→    }
   192→   192→    if (e.key === 'Backspace' && !symptomInput && symptoms.length > 0) {
   193→   193→      setSymptoms((prev) => prev.slice(0, -1))
   194→   194→    }
   195→   195→  }
   196→   196→
   197→   197→  const handleSymInputChange = (value: string) => {
   198→   198→    setSymptomInput(value)
   199→   199→    if (value.trim()) {
   200→   200→      const lower = value.toLowerCase()
   201→   201→      const filtered = COMMON_SYMPTOMS.filter(
   202→   202→        (s) => s.includes(lower) && !symptoms.includes(s)
   203→   203→      )
   204→   204→      setFilteredSuggestions(filtered.slice(0, 8))
   205→   205→      setShowSuggestions(filtered.length > 0)
   206→   206→    } else {
   207→   207→      setShowSuggestions(false)
   208→   208→    }
   209→   209→  }
   210→   210→
   211→   211→  // Close suggestions on outside click
   212→   212→  useEffect(() => {
   213→   213→    const handler = (e: MouseEvent) => {
   214→   214→      if (
   215→   215→        suggestionsRef.current &&
   216→   216→        !suggestionsRef.current.contains(e.target as Node) &&
   217→   217→        inputRef.current &&
   218→   218→        !inputRef.current.contains(e.target as Node)
   219→   219→      ) {
   220→   220→        setShowSuggestions(false)
   221→   221→      }
   222→   222→    }
   223→   223→    document.addEventListener('mousedown', handler)
   224→   224→    return () => document.removeEventListener('mousedown', handler)
   225→   225→  }, [])
   226→   226→
   227→   227→  // ── Submit handler ─────────────────────────
   228→   228→
   229→   229→  const handleCheckSymptoms = async () => {
   230→   230→    if (symptoms.length === 0) {
   231→   231→      toast({ title: 'Add symptoms', description: 'Please add at least one symptom', variant: 'destructive' })
   232→   232→      return
   233→   233→    }
   234→   234→    if (!age) {
   235→   235→      toast({ title: 'Select age group', description: 'Please select your age range', variant: 'destructive' })
   236→   236→      return
   237→   237→    }
   238→   238→    if (!gender) {
   239→   239→      toast({ title: 'Select gender', description: 'Please select your gender', variant: 'destructive' })
   240→   240→      return
   241→   241→    }
   242→   242→
   243→   243→    setLoading(true)
   244→   244→    setResults(null)
   245→   245→    setExpandedIssue(null)
   246→   246→
   247→   247→    try {
   248→   248→      const res = await fetch('/api/symptom-checker', {
   249→   249→        method: 'POST',
   250→   250→        headers: { 'Content-Type': 'application/json' },
   251→   251→        body: JSON.stringify({ symptoms, age, gender, modality }),
   252→   252→      })
   253→   253→      const data = await res.json()
   254→   254→      const result: SymptomCheckResult = data.data ?? data
   255→   255→      setResults(result)
   256→   256→      toast({
   257→   257→        title: 'Symptom Check Complete',
   258→   258→        description: `${result.matchedIssues?.length ?? 0} possible conditions found`,
   259→   259→      })
   260→   260→    } catch {
   261→   261→      toast({ title: 'Error', description: 'Failed to check symptoms', variant: 'destructive' })
   262→   262→    } finally {
   263→   263→      setLoading(false)
   264→   264→    }
   265→   265→  }
   266→   266→
   267→   267→  // ── Helpers ────────────────────────────────
   268→   268→
   269→   269→  const canSubmit = symptoms.length > 0 && age && gender
   270→   270→
   271→   271→  const getSeverityBadge = (severity: string) => {
   272→   272→    const config = SEVERITY_CONFIG[severity] ?? SEVERITY_CONFIG.MODERATE
   273→   273→    return (
   274→   274→      <Badge variant="outline" className={config.className}>
   275→   275→        {config.label}
   276→   276→      </Badge>
   277→   277→    )
   278→   278→  }
   279→   279→
   280→   280→  const getRelevanceColor = (score: number) => {
   281→   281→    if (score >= 80) return 'text-red-600 dark:text-red-400'
   282→   282→    if (score >= 60) return 'text-orange-600 dark:text-orange-400'
   283→   283→    if (score >= 40) return 'text-amber-600 dark:text-amber-400'
   284→   284→    return 'text-green-600 dark:text-green-400'
   285→   285→  }
   286→   286→
   287→   287→  const getWarningIcon = (severity: string) => {
   288→   288→    if (severity === 'CRITICAL') return <ShieldAlert className="h-4 w-4 text-red-500" />
   289→   289→    if (severity === 'WARNING') return <AlertTriangle className="h-4 w-4 text-amber-500" />
   290→   290→    return <Info className="h-4 w-4 text-teal-500" />
   291→   291→  }
   292→   292→
   293→   293→  // ── Render ─────────────────────────────────
   294→   294→
   295→   295→  return (
   296→   296→    <TooltipProvider>
   297→   297→      <motion.div {...fadeSlide} className="space-y-6">
   298→   298→        {/* ── Input Card ─────────────────────── */}
   299→   299→        <Card>
   300→   300→          <CardHeader>
   301→   301→            <CardTitle className="text-lg flex items-center gap-2">
   302→   302→              <Stethoscope className="h-5 w-5" />
   303→   303→              Symptom Checker
   304→   304→            </CardTitle>
   305→   305→            <CardDescription>
   306→   306→              Enter your symptoms to get potential health insights across modalities
   307→   307→            </CardDescription>
   308→   308→          </CardHeader>
   309→   309→          <CardContent className="space-y-5">
   310→   310→            {/* Symptom Tags Input */}
   311→   311→            <div className="space-y-2">
   312→   312→              <Label>Symptoms</Label>
   313→   313→              <div className="flex flex-wrap items-center gap-2 p-3 border rounded-lg bg-background min-h-[48px] focus-within:ring-2 focus-within:ring-ring focus-within:ring-offset-2">
   314→   314→                <AnimatePresence>
   315→   315→                  {symptoms.map((symptom) => (
   316→   316→                    <motion.span
   317→   317→                      key={symptom}
   318→   318→                      initial={{ opacity: 0, scale: 0.8 }}
   319→   319→                      animate={{ opacity: 1, scale: 1 }}
   320→   320→                      exit={{ opacity: 0, scale: 0.8 }}
   321→   321→                      className="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   322→   322→                    >
   323→   323→                      {symptom}
   324→   324→                      <button
   325→   325→                        type="button"
   326→   326→                        onClick={() => removeSymptom(symptom)}
   327→   327→                        className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   328→   328→                      >
   329→   329→                        <X className="h-3 w-3" />
   330→   330→                      </button>
   331→   331→                    </motion.span>
   332→   332→                  ))}
   333→   333→                </AnimatePresence>
   334→   334→                <div className="relative flex-1 min-w-[140px]">
   335→   335→                  <input
   336→   336→                    ref={inputRef}
   337→   337→                    type="text"
   338→   338→                    value={symptomInput}
   339→   339→                    onChange={(e) => handleSymInputChange(e.target.value)}
   340→   340→                    onKeyDown={handleSymptomKeyDown}
   341→   341→                    onFocus={() => {
   342→   342→                      if (symptomInput.trim()) handleSymInputChange(symptomInput)
   343→   343→                    }}
   344→   344→                    placeholder={symptoms.length === 0 ? 'Type a symptom (e.g. headache, fever, pet dard)...' : 'Add more...'}
   345→   345→                    className="w-full bg-transparent outline-none text-sm placeholder:text-muted-foreground"
   346→   346→                  />
   347→   347→                </div>
   348→   348→              </div>
   349→   349→              {/* Suggestions Dropdown */}
   350→   350→              <AnimatePresence>
   351→   351→                {showSuggestions && filteredSuggestions.length > 0 && (
   352→   352→                  <motion.div
   353→   353→                    ref={suggestionsRef}
   354→   354→                    initial={{ opacity: 0, y: -4 }}
   355→   355→                    animate={{ opacity: 1, y: 0 }}
   356→   356→                    exit={{ opacity: 0, y: -4 }}
   357→   357→                    className="border rounded-lg bg-popover shadow-md overflow-hidden z-10"
   358→   358→                  >
   359→   359→                    <ScrollArea className="max-h-48">
   360→   360→                      {filteredSuggestions.map((suggestion) => (
   361→   361→                        <button
   362→   362→                          key={suggestion}
   363→   363→                          type="button"
   364→   364→                          onClick={() => addSymptom(suggestion)}
   365→   365→                          className="w-full text-left px-3 py-2 text-sm hover:bg-accent transition-colors flex items-center gap-2"
   366→   366→                        >
   367→   367→                          <Plus className="h-3.5 w-3.5 text-muted-foreground" />
   368→   368→                          {suggestion}
   369→   369→                        </button>
   370→   370→                      ))}
   371→   371→                    </ScrollArea>
   372→   372→                  </motion.div>
   373→   373→                )}
   374→   374→              </AnimatePresence>
   375→   375→              {symptoms.length === 0 && (
   376→   376→                <p className="text-xs text-muted-foreground flex items-center gap-1">
   377→   377→                  <Info className="h-3 w-3" />
   378→   378→                  Press Enter or select from suggestions to add symptoms
   379→   379→                </p>
   380→   380→              )}
   381→   381→            </div>
   382→   382→
   383→   383→            {/* Age, Gender, Modality Row */}
   384→   384→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   385→   385→              {/* Age */}
   386→   386→              <div className="space-y-2">
   387→   387→                <Label>Age Group</Label>
   388→   388→                <Select value={age} onValueChange={setAge}>
   389→   389→                  <SelectTrigger>
   390→   390→                    <SelectValue placeholder="Select age..." />
   391→   391→                  </SelectTrigger>
   392→   392→                  <SelectContent>
   393→   393→                    {AGE_OPTIONS.map((opt) => (
   394→   394→                      <SelectItem key={opt.value} value={opt.value}>
   395→   395→                        {opt.label}
   396→   396→                      </SelectItem>
   397→   397→                    ))}
   398→   398→                  </SelectContent>
   399→   399→                </Select>
   400→   400→              </div>
   401→   401→
   402→   402→              {/* Gender */}
   403→   403→              <div className="space-y-2">
   404→   404→                <Label>Gender</Label>
   405→   405→                <Select value={gender} onValueChange={setGender}>
   406→   406→                  <SelectTrigger>
   407→   407→                    <SelectValue placeholder="Select gender..." />
   408→   408→                  </SelectTrigger>
   409→   409→                  <SelectContent>
   410→   410→                    {GENDER_OPTIONS.map((opt) => (
   411→   411→                      <SelectItem key={opt.value} value={opt.value}>
   412→   412→                        {opt.label}
   413→   413→                      </SelectItem>
   414→   414→                    ))}
   415→   415→                  </SelectContent>
   416→   416→                </Select>
   417→   417→              </div>
   418→   418→
   419→   419→              {/* Modality */}
   420→   420→              <div className="space-y-2">
   421→   421→                <Label>Treatment Modality</Label>
   422→   422→                <Select value={modality} onValueChange={(v) => setModality(v as Modality)}>
   423→   423→                  <SelectTrigger>
   424→   424→                    <SelectValue />
   425→   425→                  </SelectTrigger>
   426→   426→                  <SelectContent>
   427→   427→                    {Object.entries(MODALITY_CONFIG).map(([key, cfg]) => {
   428→   428→                      const Icon = cfg.icon
   429→   429→                      return (
   430→   430→                        <SelectItem key={key} value={key}>
   431→   431→                          <span className="flex items-center gap-2">
   432→   432→                            <Icon className={`h-4 w-4 ${cfg.color}`} />
   433→   433→                            {cfg.label}
   434→   434→                          </span>
   435→   435→                        </SelectItem>
   436→   436→                      )
   437→   437→                    })}
   438→   438→                  </SelectContent>
   439→   439→                </Select>
   440→   440→              </div>
   441→   441→            </div>
   442→   442→
   443→   443→            {/* Check Button */}
   444→   444→            <div className="flex items-center gap-3">
   445→   445→              <Button
   446→   446→                onClick={handleCheckSymptoms}
   447→   447→                disabled={!canSubmit || loading}
   448→   448→                className="gap-2 min-w-[180px]"
   449→   449→              >
   450→   450→                {loading ? (
   451→   451→                  <>
   452→   452→                    <Loader2 className="h-4 w-4 animate-spin" />
   453→   453→                    Checking...
   454→   454→                  </>
   455→   455→                ) : (
   456→   456→                  <>
   457→   457→                    <Search className="h-4 w-4" />
   458→   458→                    Check Symptoms
   459→   459→                  </>
   460→   460→                )}
   461→   461→              </Button>
   462→   462→              {symptoms.length > 0 && (
   463→   463→                <Button
   464→   464→                  variant="ghost"
   465→   465→                  size="sm"
   466→   466→                  onClick={() => {
   467→   467→                    setSymptoms([])
   468→   468→                    setSymptomInput('')
   469→   469→                    setResults(null)
   470→   470→                  }}
   471→   471→                  className="text-muted-foreground"
   472→   472→                >
   473→   473→                  Clear all

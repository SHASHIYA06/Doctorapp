   100→  warnings: string[]

   140→const SEVERITY_CELL_CONFIG: Record<string, string> = {
   141→  LOW: 'text-green-700 dark:text-green-300',
   142→  MODERATE: 'text-amber-700 dark:text-amber-300',
   143→  HIGH: 'text-orange-700 dark:text-orange-300',
   144→  CRITICAL: 'text-red-700 dark:text-red-300',
   145→}
   146→
   147→const SAFETY_SCORE_COLORS: Record<string, { bar: string; text: string; bg: string }> = {
   148→  green: { bar: 'bg-green-500', text: 'text-green-700 dark:text-green-300', bg: 'bg-green-50 dark:bg-green-950/30' },
   149→  amber: { bar: 'bg-amber-500', text: 'text-amber-700 dark:text-amber-300', bg: 'bg-amber-50 dark:bg-amber-950/30' },
   150→  orange: { bar: 'bg-orange-500', text: 'text-orange-700 dark:text-orange-300', bg: 'bg-orange-50 dark:bg-orange-950/30' },
   151→  red: { bar: 'bg-red-500', text: 'text-red-700 dark:text-red-300', bg: 'bg-red-50 dark:bg-red-950/30' },
   152→  critical: { bar: 'bg-red-700', text: 'text-red-900 dark:text-red-200', bg: 'bg-red-100 dark:bg-red-950/50' },
   153→}
   154→
   155→function getSafetyTier(score: number): keyof typeof SAFETY_SCORE_COLORS {
   156→  if (score >= 80) return 'green'
   157→  if (score >= 60) return 'amber'
   158→  if (score >= 40) return 'orange'
   159→  if (score >= 20) return 'red'
   160→  return 'critical'
   161→}
   162→
   163→const POPULATION_CONFIG: Record<string, { label: string; icon: React.ElementType }> = {
   164→  pregnancy: { label: 'Pregnancy', icon: Baby },
   165→  lactation: { label: 'Lactation', icon: Baby },
   166→  pediatric: { label: 'Pediatric', icon: Heart },
   167→  geriatric: { label: 'Geriatric', icon: Heart },
   168→  hepaticImpairment: { label: 'Hepatic Impairment', icon: Activity },
   169→  renalImpairment: { label: 'Renal Impairment', icon: Activity },
   170→}
   171→
   172→const fadeSlide = {
   173→  initial: { opacity: 0, y: 12 },
   174→  animate: { opacity: 1, y: 0 },
   175→  transition: { duration: 0.3 },
   176→}

   178→// ──────────────────────────────────────────────
   179→// Component
   180→// ──────────────────────────────────────────────
   181→
   182→export function DrugInteractionsSection() {
   183→  const { selectedPatientId } = useAppStore()
   184→
   185→  // Medicine selection state
   186→  const [searchQuery, setSearchQuery] = useState('')
   187→  const [searchResults, setSearchResults] = useState<MedicineSearchResult[]>([])
   188→  const [selectedMedicines, setSelectedMedicines] = useState<SelectedMedicine[]>([])
   189→  const [searching, setSearching] = useState(false)
   190→  const [showDropdown, setShowDropdown] = useState(false)
   191→  const searchInputRef = useRef<HTMLInputElement>(null)
   192→  const dropdownRef = useRef<HTMLDivElement>(null)
   193→
   194→  // Results state
   195→  const [results, setResults] = useState<DrugInteractionResult | null>(null)
   196→  const [loading, setLoading] = useState(false)
   197→  const [showAllInteractions, setShowAllInteractions] = useState(false)
   198→  const [patientMedsLoading, setPatientMedsLoading] = useState(false)
   199→
   200→  // CDSCO Advisory Data (well-known interactions)
   201→  const CDSCO_ADVISORIES: Record<string, { advisory: string; reference: string }> = {
   202→    'Warfarin-Aspirin': { advisory: 'CDSCO Advisory: Combined use significantly increases bleeding risk. INR monitoring mandatory. Avoid without specialist supervision.', reference: 'CDSCO/2024/DRG-INT/012' },
   203→    'Warfarin-Ibuprofen': { advisory: 'CDSCO Advisory: NSAIDs with Warfarin increase GI bleeding risk. Use Paracetamol as alternative analgesic.', reference: 'CDSCO/2024/DRG-INT/015' },
   204→    'Metformin-Cimetidine': { advisory: 'CDSCO Advisory: Cimetidine reduces Metformin renal clearance. Risk of lactic acidosis. Use alternative H2 blocker.', reference: 'CDSCO/2024/DRG-INT/023' },
   205→  }
   206→
   207→  const getCdScoAdvisory = (med1: string, med2: string): { advisory: string; reference: string } | null => {
   208→    const key1 = `${med1}-${med2}`
   209→    const key2 = `${med2}-${med1}`
   210→    return CDSCO_ADVISORIES[key1] ?? CDSCO_ADVISORIES[key2] ?? null
   211→  }
   212→
   213→  // ── Search medicines ───────────────────────
   214→
   215→  const searchMedicines = useCallback(async (query: string) => {
   216→    if (!query.trim() || query.length < 2) {
   217→      setSearchResults([])
   218→      setShowDropdown(false)
   219→      return
   220→    }
   221→
   222→    setSearching(true)
   223→    try {
   224→      const res = await fetch(`/api/medicines?search=${encodeURIComponent(query)}&limit=10`)
   225→      const data = await res.json()
   226→      const list: MedicineSearchResult[] = data.data ?? data.medicines ?? []
   227→      // Filter out already selected
   228→      const selectedIds = selectedMedicines.map((m) => m.id)
   229→      const filtered = list.filter((m) => !selectedIds.includes(m.id))
   230→      setSearchResults(filtered)
   231→      setShowDropdown(filtered.length > 0)
   232→    } catch {
   233→      setSearchResults([])
   234→    } finally {
   235→      setSearching(false)
   236→    }
   237→  }, [selectedMedicines])
   238→
   239→  // Debounced search
   240→  useEffect(() => {
   241→    const timer = setTimeout(() => {
   242→      searchMedicines(searchQuery)
   243→    }, 300)
   244→    return () => clearTimeout(timer)
   245→  }, [searchQuery, searchMedicines])
   246→
   247→  // Close dropdown on outside click
   248→  useEffect(() => {
   249→    const handler = (e: MouseEvent) => {
   250→      if (
   251→        dropdownRef.current &&
   252→        !dropdownRef.current.contains(e.target as Node) &&
   253→        searchInputRef.current &&
   254→        !searchInputRef.current.contains(e.target as Node)
   255→      ) {
   256→        setShowDropdown(false)
   257→      }
   258→    }
   259→    document.addEventListener('mousedown', handler)
   260→    return () => document.removeEventListener('mousedown', handler)
   261→  }, [])
   262→
   263→  // ── Medicine management ────────────────────
   264→
   265→  const addMedicine = (medicine: MedicineSearchResult) => {
   266→    if (!selectedMedicines.find((m) => m.id === medicine.id)) {
   267→      setSelectedMedicines((prev) => [...prev, { id: medicine.id, name: medicine.name }])
   268→    }
   269→    setSearchQuery('')
   270→    setSearchResults([])
   271→    setShowDropdown(false)
   272→    searchInputRef.current?.focus()
   273→  }
   274→
   275→  const removeMedicine = (id: string) => {
   276→    setSelectedMedicines((prev) => prev.filter((m) => m.id !== id))
   277→  }
   278→
   279→  // ── Check interactions ─────────────────────
   280→
   281→  const handleCheckInteractions = async () => {
   282→    if (selectedMedicines.length < 2) {
   283→      toast({
   284→        title: 'Add more medicines',
   285→        description: 'Please add at least 2 medicines to check interactions',
   286→        variant: 'destructive',
   287→      })
   288→      return
   289→    }
   290→
   291→    setLoading(true)
   292→    setResults(null)
   293→    setShowAllInteractions(false)
   294→
   295→    try {
   296→      const res = await fetch('/api/drug-interactions', {
   297→        method: 'POST',
   298→        headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({
   300→          medicineIds: selectedMedicines.map((m) => m.id),
   301→          medicineNames: selectedMedicines.map((m) => m.name),
   302→        }),
   303→      })
   304→      const data = await res.json()
   305→      const result: DrugInteractionResult = data.data ?? data
   306→      setResults(result)
   307→
   308→      const riskLabel = RISK_LEVEL_CONFIG[result.riskLevel]?.label ?? result.riskLevel
   309→      toast({
   310→        title: 'Interaction Check Complete',
   311→        description: `Safety score: ${result.safetyScore}/100 — ${riskLabel}`,
   312→      })
   313→    } catch {
   314→      toast({ title: 'Error', description: 'Failed to check drug interactions', variant: 'destructive' })
   315→    } finally {
   316→      setLoading(false)
   317→    }
   318→  }
   319→
   320→  // ── Helpers ────────────────────────────────
   321→
   322→  const safetyTier = results ? getSafetyTier(results.safetyScore) : null
   323→  const tierColors = safetyTier ? SAFETY_SCORE_COLORS[safetyTier] : null
   324→
   325→  const visibleInteractions = results?.pairwiseInteractions
   326→    ? showAllInteractions
   327→      ? results.pairwiseInteractions
   328→      : results.pairwiseInteractions.slice(0, 5)
   329→    : []
   330→
   331→  // ── Render ─────────────────────────────────
   332→
   333→  return (
   334→    <TooltipProvider>
   335→      <motion.div {...fadeSlide} className="space-y-6">
   336→        {/* ── Input Card ─────────────────────── */}
   337→        <Card>
   338→          <CardHeader>
   339→            <CardTitle className="text-lg flex items-center gap-2">
   340→              <Pill className="h-5 w-5" />
   341→              Drug Interaction Checker
   342→            </CardTitle>
   343→            <CardDescription>
   344→              Add medicines to check for potential drug-drug interactions and safety concerns
   345→            </CardDescription>
   346→          </CardHeader>
   347→          <CardContent className="space-y-5">
   348→            {/* Medicine Search Input */}
   349→            <div className="space-y-2">
   350→              <Label>Add Medicines</Label>
   351→              <div className="relative">
   352→                <div className="relative">
   353→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   354→                  <Input
   355→                    ref={searchInputRef}
   356→                    type="text"
   357→                    value={searchQuery}
   358→                    onChange={(e) => setSearchQuery(e.target.value)}
   359→                    onFocus={() => {
   360→                      if (searchResults.length > 0) setShowDropdown(true)
   361→                    }}
   362→                    placeholder="Search medicines by name..."
   363→                    className="pl-9"
   364→                  />
   365→                  {searching && (
   366→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 animate-spin text-muted-foreground" />
   367→                  )}
   368→                </div>
   369→
   370→                {/* Search Dropdown */}
   371→                <AnimatePresence>
   372→                  {showDropdown && searchResults.length > 0 && (
   373→                    <motion.div
   374→                      ref={dropdownRef}
   375→                      initial={{ opacity: 0, y: -4 }}
   376→                      animate={{ opacity: 1, y: 0 }}
   377→                      exit={{ opacity: 0, y: -4 }}
   378→                      className="absolute top-full left-0 right-0 mt-1 border rounded-lg bg-popover shadow-lg z-20 overflow-hidden"
   379→                    >
   380→                      <ScrollArea className="max-h-56">
   381→                        {searchResults.map((medicine) => (
   382→                          <button
   383→                            key={medicine.id}
   384→                            type="button"
   385→                            onClick={() => addMedicine(medicine)}
   386→                            className="w-full text-left px-3 py-2.5 text-sm hover:bg-accent transition-colors flex items-center gap-3"
   387→                          >
   388→                            <Plus className="h-4 w-4 text-muted-foreground shrink-0" />
   389→                            <div className="flex-1 min-w-0">
   390→                              <span className="font-medium">{medicine.name}</span>
   391→                              {medicine.genericName && (
   392→                                <span className="text-muted-foreground ml-1">({medicine.genericName})</span>
   393→                              )}
   394→                            </div>
   395→                            <Badge variant="outline" className="text-xs shrink-0">
   396→                              {medicine.modality}
   397→                            </Badge>
   398→                          </button>
   399→                        ))}
   400→                      </ScrollArea>
   401→                    </motion.div>
   402→                  )}
   403→                </AnimatePresence>
   404→              </div>
   405→            </div>
   406→
   407→            {/* Selected Medicine Tags */}
   408→            {selectedMedicines.length > 0 && (
   409→              <div className="space-y-2">
   410→                <Label className="text-muted-foreground">
   411→                  Selected Medicines ({selectedMedicines.length})
   412→                </Label>
   413→                <div className="flex flex-wrap gap-2">
   414→                  <AnimatePresence>
   415→                    {selectedMedicines.map((med) => (
   416→                      <motion.span
   417→                        key={med.id}
   418→                        initial={{ opacity: 0, scale: 0.8 }}
   419→                        animate={{ opacity: 1, scale: 1 }}
   420→                        exit={{ opacity: 0, scale: 0.8 }}
   421→                        className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   422→                      >
   423→                        <Pill className="h-3.5 w-3.5" />
   424→                        {med.name}
   425→                        <button
   426→                          type="button"
   427→                          onClick={() => removeMedicine(med.id)}
   428→                          className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   429→                        >
   430→                          <X className="h-3 w-3" />
   431→                        </button>
   432→                      </motion.span>
   433→                    ))}
   434→                  </AnimatePresence>
   435→                </div>
   436→              </div>
   437→            )}
   438→
   439→            {/* Action Buttons */}
   440→            <div className="flex flex-wrap items-center gap-3">
   441→              <Button
   442→                onClick={handleCheckInteractions}
   443→                disabled={selectedMedicines.length < 2 || loading}
   444→                className="gap-2 min-w-[200px]"
   445→              >
   446→                {loading ? (
   447→                  <>
   448→                    <Loader2 className="h-4 w-4 animate-spin" />
   449→                    Checking...
   450→                  </>
   451→                ) : (
   452→                  <>
   453→                    <ShieldAlert className="h-4 w-4" />
   454→                    Check Interactions
   455→                  </>
   456→                )}
   457→              </Button>
   458→              {/* Check with Patient Medications */}
   459→              <Button
   460→                variant="outline"
   461→                className="gap-2"
   462→                disabled={patientMedsLoading}
   463→                onClick={async () => {
   464→                  if (!selectedPatientId) {
   465→                    toast({ title: 'No Patient Selected', description: 'Select a patient from the Patients section first', variant: 'destructive' })
   466→                    return
   467→                  }
   468→                  setPatientMedsLoading(true)
   469→                  try {
   470→                    const res = await fetch(`/api/patients/${selectedPatientId}`)
   471→                    const data = await res.json()
   472→                    const patient = data.data ?? data.patient
   473→                    const activeMeds = patient?.medicationStatements?.filter((m: any) => m.isActive) ?? []
   474→                    if (activeMeds.length === 0) {
   475→                      toast({ title: 'No Active Medications', description: 'This patient has no active medications on file' })
   476→                    } else {
   477→                      const medNames = activeMeds.map((m: any) => ({ id: m.id, name: m.medication }))
   478→                      setSelectedMedicines(medNames)
   479→                      toast({ title: 'Patient Medications Loaded', description: `${medNames.length} active medications added` })
   480→                    }
   481→                  } catch {
   482→                    toast({ title: 'Error', description: 'Failed to load patient medications', variant: 'destructive' })
   483→                  } finally {
   484→                    setPatientMedsLoading(false)
   485→                  }
   486→                }}
   487→              >
   488→                {patientMedsLoading ? <Loader2 className="h-4 w-4 animate-spin" /> : <UserRound className="h-4 w-4" />}
   489→                Check with Patient Meds
   490→              </Button>
   491→              {selectedMedicines.length > 0 && (
   492→                <Button
   493→                  variant="ghost"
   494→                  size="sm"
   495→                  onClick={() => {
   496→                    setSelectedMedicines([])
   497→                    setSearchQuery('')
   498→                    setResults(null)
   499→                  }}
   500→                  className="text-muted-foreground"
   501→                >
   502→                  Clear all
   503→                </Button>
   504→              )}
   505→            </div>
   506→
   507→            {selectedMedicines.length > 0 && selectedMedicines.length < 2 && (
   508→              <p className="text-xs text-muted-foreground flex items-center gap-1">
   509→                <Info className="h-3 w-3" />
   510→                Add at least 2 medicines to check for interactions
   511→              </p>
   512→            )}
   513→          </CardContent>
   514→        </Card>
   515→
   516→        {/* ── Loading Skeleton ────────────────── */}
   517→        <AnimatePresence>
   518→          {loading && (
   519→            <motion.div
   520→              initial={{ opacity: 0 }}
   521→              animate={{ opacity: 1 }}
   522→              exit={{ opacity: 0 }}
   523→              className="space-y-4"
   524→            >
   525→              <Card>
   526→                <CardContent className="p-6 space-y-4">
   527→                  <Skeleton className="h-8 w-64" />
   528→                  <Skeleton className="h-4 w-full" />
   529→                  <Skeleton className="h-32 w-full" />
   530→                  <Skeleton className="h-24 w-full" />
   531→                </CardContent>
   532→              </Card>
   533→            </motion.div>
   534→          )}
   535→        </AnimatePresence>
   536→
   537→        {/* ── Results ─────────────────────────── */}
   538→        <AnimatePresence>
   539→          {results && !loading && (
   540→            <motion.div
   541→              initial={{ opacity: 0, y: 16 }}
   542→              animate={{ opacity: 1, y: 0 }}
   543→              transition={{ duration: 0.4 }}
   544→              className="space-y-4"
   545→            >
   546→              {/* ── Safety Score & Risk Meter ────── */}
   547→              <Card className={tierColors ? `border-2 ${tierColors.bg}` : ''}>
   548→                <CardContent className="p-6">
   549→                  <div className="flex flex-col sm:flex-row items-start sm:items-center gap-6">
   550→                    {/* Score Circle */}
   551→                    <div className="text-center shrink-0">
   552→                      <div className={`text-5xl font-bold tracking-tight ${tierColors?.text ?? ''}`}>
   553→                        {results.safetyScore}
   554→                      </div>
   555→                      <div className="text-sm text-muted-foreground mt-1">of 100</div>
   556→                    </div>
   557→
   558→                    {/* Risk Meter & Level */}

// ================================================================
// Clinician-Governed Multi-Modality Healthcare SaaS
// Three Care Tracks: ALLOPATHY, AYURVEDA, HOMEOPATHY (NEVER MERGED)
// Database: PostgreSQL (Neon Serverless)
// ================================================================

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DATABASE_URL_UNPOOLED")
}

  id          String   @id @default(cuid())

// ============================================================
// HEALTH ISSUE TAXONOMY
// Hierarchical: Body System → Clinical Domain → Symptom Group → Issue → Subtype
// ============================================================

model HealthIssue {
  id              String   @id @default(cuid())
  code            String?  // ICD-10 or modality-specific code
  name            String
  description     String?
  bodySystem      String   // RESPIRATORY, CARDIOVASCULAR, GASTROINTESTINAL, etc.
  clinicalDomain  String   // General Medicine, Pulmonology, Cardiology, etc.
  symptomGroup    String   // Cough, Chest Pain, Abdominal Pain, etc.
  severity        String   @default("MODERATE") // MILD, MODERATE, SEVERE, CRITICAL
  chronicity      String   @default("ACUTE") // ACUTE, CHRONIC, ACUTE_ON_CHRONIC
  prevalence      String?  // RARE, UNCOMMON, COMMON, VERY_COMMON
  isSubtype       Boolean  @default(false)
  parentIssueId   String?
  isActive        Boolean  @default(true)
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  aliases       HealthIssueAlias[]
  translations  HealthIssueTranslation[]
  wingApproaches HealthIssueWing[]
  indications   MedicineIndication[]
  patternIssues TreatmentPatternIssue[]
}

model HealthIssueAlias {
  id         String   @id @default(cuid())
  issueId    String
  alias      String   // "pet dard", "stomach ache", "abd pain"
  language   String   @default("en")
  createdAt  DateTime @default(now())

  issue HealthIssue @relation(fields: [issueId], references: [id])

  @@index([alias])
  @@index([language])
}

model HealthIssueTranslation {
  id          String   @id @default(cuid())
  issueId     String
  language    String   // en, hi, bn, ta, te, mr, gu, kn, ml, pa, or, ur
  name        String
  description String?
  createdAt   DateTime @default(now())

  issue HealthIssue @relation(fields: [issueId], references: [id])

  @@unique([issueId, language])
}

model HealthIssueWing {
  id                String   @id @default(cuid())
  issueId           String
  modality          String   // ALLOPATHY, AYURVEDA, HOMEOPATHY
  approach          String   // Brief clinical approach description
  commonMedicines   String?  // JSON array of medicine IDs
  lifestyleAdvice   String?
  whenToSeeDoctor   String?
  redFlags          String?  // JSON array of red flag symptoms
  evidenceLevel     String?  // STRONG, MODERATE, LIMITED, TRADITIONAL
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt

  issue HealthIssue @relation(fields: [issueId], references: [id])

  @@unique([issueId, modality])
}

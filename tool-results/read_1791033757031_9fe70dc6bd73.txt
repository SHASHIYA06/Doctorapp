   198→export function LabOrdersSection() {
   199→  const { selectedPatientId } = useAppStore()
   200→
   201→  const [labOrders, setLabOrders] = useState<LabOrder[]>([])
   202→  const [patients, setPatients] = useState<Patient[]>([])
   203→  const [loading, setLoading] = useState(true)
   204→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   205→  const [categoryTab, setCategoryTab] = useState<string>('all')
   206→
   207→  // Create dialog
   208→  const [createOpen, setCreateOpen] = useState(false)
   209→  const [formPatientId, setFormPatientId] = useState(selectedPatientId ?? '')
   210→  const [formPriority, setFormPriority] = useState<LabPriority>('ROUTINE')
   211→  const [formNotes, setFormNotes] = useState('')
   212→  const [selectedTests, setSelectedTests] = useState<Set<string>>(new Set())
   213→  const [submitting, setSubmitting] = useState(false)
   214→
   215→  // Detail dialog
   216→  const [detailOpen, setDetailOpen] = useState(false)
   217→  const [selectedOrder, setSelectedOrder] = useState<LabOrder | null>(null)
   218→
   219→  // ─── Fetch patients ───────────────────────────────────────────────
   220→
   221→  useEffect(() => {
   222→    fetch('/api/patients')
   223→      .then((r) => r.json())
   224→      .then((d) => setPatients(d.data ?? d.patients ?? []))
   225→      .catch(() => {})
   226→  }, [])
   227→
   228→  // ─── Fetch lab orders ─────────────────────────────────────────────
   229→
   230→  const loadLabOrders = useCallback(async () => {
   231→    setLoading(true)
   232→    try {
   233→      const params = new URLSearchParams()
   234→      if (statusFilter && statusFilter !== 'ALL') params.set('status', statusFilter)
   235→      if (selectedPatientId) params.set('patientId', selectedPatientId)
   236→      const res = await fetch(`/api/lab-orders?${params.toString()}`)
   237→      const json = await res.json()
   238→      setLabOrders((json.data ?? json.labOrders ?? []) as LabOrder[])
   239→    } catch {
   240→      toast({ title: 'Error', description: 'Failed to load lab orders', variant: 'destructive' })
   241→    } finally {
   242→      setLoading(false)
   243→    }
   244→  }, [statusFilter, selectedPatientId])
   245→
   246→  useEffect(() => { loadLabOrders() }, [loadLabOrders])
   247→
   248→  // ─── Test toggle ──────────────────────────────────────────────────
   249→
   250→  const toggleTest = (testName: string) => {
   251→    setSelectedTests((prev) => {
   252→      const next = new Set(prev)
   253→      if (next.has(testName)) next.delete(testName)
   254→      else next.add(testName)
   255→      return next
   256→    })
   257→  }
   258→
   259→  // ─── Create lab order ─────────────────────────────────────────────
   260→
   261→  const handleCreate = async () => {
   262→    if (!formPatientId) {
   263→      toast({ title: 'Validation', description: 'Select a patient', variant: 'destructive' })
   264→      return
   265→    }
   266→    if (selectedTests.size === 0) {
   267→      toast({ title: 'Validation', description: 'Select at least one test', variant: 'destructive' })
   268→      return
   269→    }
   270→    setSubmitting(true)
   271→    try {
   272→      const tests = Array.from(selectedTests).map((name) => {
   273→        const match = COMMON_TESTS.find((t) => t.name === name)
   274→        return {
   275→          testName: name,
   276→          testCode: match?.loinc ?? '',
   277→          category: match?.category ?? 'BIOCHEMISTRY',
   278→        }
   279→      })
   280→      const res = await fetch('/api/lab-orders', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          patientId: formPatientId,
   285→          priority: formPriority,
   286→          notes: formNotes || undefined,
   287→          tests,
   288→        }),
   289→      })
   290→      if (res.ok) {
   291→        toast({ title: 'Lab Order Created', description: `Order with ${tests.length} test(s) placed` })
   292→        setCreateOpen(false)
   293→        resetForm()
   294→        loadLabOrders()
   295→      } else {
   296→        toast({ title: 'Error', description: 'Failed to create lab order', variant: 'destructive' })
   297→      }
   298→    } catch {
   299→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   300→    } finally {
   301→      setSubmitting(false)
   302→    }
   303→  }
   304→
   305→  const resetForm = () => {
   306→    setFormPatientId(selectedPatientId ?? '')
   307→    setFormPriority('ROUTINE')
   308→    setFormNotes('')
   309→    setSelectedTests(new Set())
   310→  }
   311→
   312→  // ─── Filtered orders by category tab ──────────────────────────────
   313→
   314→  const filteredOrders = labOrders.filter((order) => {
   315→    if (categoryTab === 'all') return true
   316→    return order.tests?.some((t) => t.category === categoryTab)
   317→  })
   318→
   319→  // ─── Status timeline step index ───────────────────────────────────
   320→
   321→  const getStepIndex = (status: LabOrderStatus): number => {
   322→    if (status === 'CANCELLED') return -1
   323→    return STATUS_STEPS.indexOf(status)
   324→  }
   325→
   326→  // ─── Render ───────────────────────────────────────────────────────
   327→
   328→  if (loading) {
   329→    return (
   330→      <div className="space-y-3">
   331→        {[1, 2, 3].map((i) => (
   332→          <Skeleton key={i} className="h-24 w-full" />
   333→        ))}
   334→      </div>
   335→    )
   336→  }
   337→
   338→  return (
   339→    <motion.div {...fadeSlide} className="space-y-6">
   340→      {/* Header */}
   341→      <div className="flex items-center justify-between flex-wrap gap-3">
   342→        <div className="flex items-center gap-3">
   343→          <div className="p-2 rounded-lg bg-teal-100 dark:bg-teal-900/30">
   344→            <FlaskConical className="h-6 w-6 text-teal-600" />
   345→          </div>
   346→          <div>
   347→            <h2 className="text-2xl font-bold tracking-tight">Lab & Diagnostics</h2>
   348→            <p className="text-sm text-muted-foreground">Order tests, track results, and flag abnormal values</p>
   349→          </div>
   350→        </div>
   351→        <Button
   352→          className="gap-2 bg-teal-600 hover:bg-teal-700 text-white"
   353→          onClick={() => {
   354→            resetForm()
   355→            setCreateOpen(true)
   356→          }}
   357→        >
   358→          <Plus className="h-4 w-4" /> New Lab Order
   359→        </Button>
   360→      </div>
   361→
   362→      {/* Status Filter + Category Tabs */}
   363→      <Tabs
   364→        value={categoryTab}
   365→        onValueChange={setCategoryTab}
   366→        className="space-y-4"
   367→      >
   368→        <div className="flex items-center gap-4 flex-wrap">
   369→          <TabsList>
   370→            <TabsTrigger value="all" className="gap-1">
   371→              All
   372→            </TabsTrigger>
   373→            <TabsTrigger value="HEMATOLOGY" className="gap-1">
   374→              <Beaker className="h-3 w-3" /> Hematology
   375→            </TabsTrigger>
   376→            <TabsTrigger value="BIOCHEMISTRY" className="gap-1">
   377→              <FlaskConical className="h-3 w-3" /> Biochemistry
   378→            </TabsTrigger>
   379→            <TabsTrigger value="MICROBIOLOGY" className="gap-1">
   380→              <TestTube className="h-3 w-3" /> Microbiology
   381→            </TabsTrigger>
   382→            <TabsTrigger value="RADIOLOGY" className="gap-1">
   383→              Radiology
   384→            </TabsTrigger>
   385→            <TabsTrigger value="PATHOLOGY" className="gap-1">
   386→              Pathology
   387→            </TabsTrigger>
   388→          </TabsList>
   389→
   390→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   391→            <SelectTrigger className="w-40">
   392→              <SelectValue placeholder="All Status" />
   393→            </SelectTrigger>
   394→            <SelectContent>
   395→              <SelectItem value="ALL">All Status</SelectItem>
   396→              <SelectItem value="ORDERED">Ordered</SelectItem>
   397→              <SelectItem value="SAMPLE_COLLECTED">Sample Collected</SelectItem>
   398→              <SelectItem value="PROCESSING">Processing</SelectItem>
   399→              <SelectItem value="COMPLETED">Completed</SelectItem>
   400→              <SelectItem value="CANCELLED">Cancelled</SelectItem>
   401→            </SelectContent>
   402→          </Select>
   403→        </div>
   404→
   405→        {/* All tab contents share the same table */}
   406→        {['all', 'HEMATOLOGY', 'BIOCHEMISTRY', 'MICROBIOLOGY', 'RADIOLOGY', 'PATHOLOGY'].map(
   407→          (tab) => (
   408→            <TabsContent key={tab} value={tab} className="space-y-4">
   409→              {/* Summary cards */}
   410→              <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   411→                {(
   412→                  ['ORDERED', 'SAMPLE_COLLECTED', 'PROCESSING', 'COMPLETED', 'CANCELLED'] as LabOrderStatus[]
   413→                ).map((st) => {
   414→                  const count = filteredOrders.filter((o) => o.status === st).length
   415→                  return (
   416→                    <Card key={st} className="p-3">
   417→                      <div className="flex items-center justify-between">
   418→                        <span className="text-xs font-medium text-muted-foreground">
   419→                          {STATUS_CONFIG[st].label}
   420→                        </span>
   421→                        <Badge className={`text-xs ${STATUS_CONFIG[st].className}`}>
   422→                          {count}
   423→                        </Badge>
   424→                      </div>
   425→                    </Card>
   426→                  )
   427→                })}
   428→              </div>
   429→
   430→              {/* Lab orders table */}
   431→              {filteredOrders.length === 0 ? (
   432→                <Card>
   433→                  <CardContent className="py-8 text-center text-muted-foreground">
   434→                    No lab orders found
   435→                  </CardContent>
   436→                </Card>
   437→              ) : (
   438→                <Card>
   439→                  <CardContent className="p-0">
   440→                    <div className="overflow-x-auto">
   441→                      <Table>
   442→                        <TableHeader>
   443→                          <TableRow>
   444→                            <TableHead>Order #</TableHead>
   445→                            <TableHead>Patient</TableHead>
   446→                            <TableHead>Priority</TableHead>
   447→                            <TableHead>Tests</TableHead>
   448→                            <TableHead>Status</TableHead>
   449→                            <TableHead>Ordered</TableHead>
   450→                            <TableHead className="text-right">Action</TableHead>
   451→                          </TableRow>
   452→                        </TableHeader>
   453→                        <TableBody>
   454→                          {filteredOrders.map((order) => {
   455→                            const statusCfg = STATUS_CONFIG[order.status] ?? STATUS_CONFIG.ORDERED
   456→                            const priCfg = PRIORITY_CONFIG[order.priority] ?? PRIORITY_CONFIG.ROUTINE
   457→                            const hasAbnormal = order.tests?.some((t) => t.isAbnormal)
   458→                            return (
   459→                              <TableRow
   460→                                key={order.id}
   461→                                className="cursor-pointer hover:bg-muted/50"
   462→                                onClick={() => {
   463→                                  setSelectedOrder(order)
   464→                                  setDetailOpen(true)
   465→                                }}
   466→                              >
   467→                                <TableCell className="font-mono text-sm font-medium">
   468→                                  {order.orderNumber}
   469→                                </TableCell>
   470→                                <TableCell className="text-sm">
   471→                                  {order.patient
   472→                                    ? `${order.patient.firstName} ${order.patient.lastName}`
   473→                                    : '—'}
   474→                                </TableCell>
   475→                                <TableCell>
   476→                                  <Badge className={`text-xs ${priCfg.className}`}>
   477→                                    {priCfg.label}
   478→                                  </Badge>
   479→                                </TableCell>
   480→                                <TableCell className="text-sm">
   481→                                  {order.tests?.length ?? 0}
   482→                                </TableCell>
   483→                                <TableCell>
   484→                                  <div className="flex items-center gap-1">
   485→                                    <Badge className={`text-xs ${statusCfg.className}`}>
   486→                                      {statusCfg.label}
   487→                                    </Badge>
   488→                                    {hasAbnormal && (
   489→                                      <AlertCircle className="h-3 w-3 text-red-500" />
   490→                                    )}
   491→                                  </div>
   492→                                </TableCell>
   493→                                <TableCell className="text-sm text-muted-foreground">
   494→                                  {new Date(order.orderedAt).toLocaleDateString()}
   495→                                </TableCell>
   496→                                <TableCell className="text-right">
   497→                                  <Button
   498→                                    variant="ghost"
   499→                                    size="sm"
   500→                                    onClick={(e) => {
   501→                                      e.stopPropagation()
   502→                                      setSelectedOrder(order)
   503→                                      setDetailOpen(true)
   504→                                    }}
   505→                                  >
   506→                                    <FlaskConical className="h-4 w-4" />
   507→                                  </Button>
   508→                                </TableCell>
   509→                              </TableRow>
   510→                            )
   511→                          })}
   512→                        </TableBody>
   513→                      </Table>
   514→                    </div>
   515→                  </CardContent>
   516→                </Card>
   517→              )}
   518→            </TabsContent>
   519→          )
   520→        )}
   521→      </Tabs>
   522→
   523→      {/* ─── Create Lab Order Dialog ─────────────────────────────── */}
   524→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   525→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   526→          <DialogHeader>
   527→            <DialogTitle className="flex items-center gap-2">
   528→              <FlaskConical className="h-5 w-5 text-teal-600" /> New Lab Order
   529→            </DialogTitle>
   530→            <DialogDescription>
   531→              Select tests to order — LOINC codes are auto-assigned
   532→            </DialogDescription>
   533→          </DialogHeader>
   534→
   535→          <div className="grid gap-4 py-2">
   536→            {/* Patient & Priority */}
   537→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   538→              <div className="space-y-2">
   539→                <Label>Patient *</Label>
   540→                <Select value={formPatientId} onValueChange={setFormPatientId}>
   541→                  <SelectTrigger>
   542→                    <SelectValue placeholder="Select patient..." />
   543→                  </SelectTrigger>
   544→                  <SelectContent>
   545→                    {patients.map((p) => (
   546→                      <SelectItem key={p.id} value={p.id}>
   547→                        {p.firstName} {p.lastName}
   548→                      </SelectItem>
   549→                    ))}
   550→                  </SelectContent>
   551→                </Select>
   552→              </div>
   553→              <div className="space-y-2">
   554→                <Label>Priority *</Label>
   555→                <Select value={formPriority} onValueChange={(v) => setFormPriority(v as LabPriority)}>
   556→                  <SelectTrigger>
   557→                    <SelectValue />
   558→                  </SelectTrigger>
   559→                  <SelectContent>
   560→                    <SelectItem value="ROUTINE">Routine</SelectItem>

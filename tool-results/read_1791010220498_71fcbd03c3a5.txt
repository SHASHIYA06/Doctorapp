    68→  type: string

    74→  id: string
    75→  name: string
    76→  bodySystem: string
    77→  severity: 'LOW' | 'MODERATE' | 'HIGH' | 'CRITICAL'
    78→  relevanceScore: number
    79→  description: string
    80→  wingApproaches: WingApproach[]
    81→  topMedicines: { name: string; modality: string }[]
    82→  safetyWarnings: SafetyWarning[]
    83→}
    84→

    86→  matchedIssues: MatchedIssue[]

    88→}

    93→

    95→  'headache', 'fever', 'cough', 'cold', 'fatigue', 'body pain',
    96→  'nausea', 'dizziness', 'chest pain', 'breathing difficulty',
    97→  'stomach pain', 'back pain', 'joint pain', 'skin rash',
    98→  'sore throat', 'vomiting', 'diarrhea', 'constipation',
    99→  'anxiety', 'depression', 'insomnia', 'pet dard',
   100→  'sir dard', 'bukhar', 'khasi', 'dast',
   101→]
   102→
   103→const AGE_OPTIONS = [
   104→  { value: '0-2', label: '0–2 years (Infant)' },
   105→  { value: '3-12', label: '3–12 years (Child)' },
   106→  { value: '13-17', label: '13–17 years (Adolescent)' },
   107→  { value: '18-39', label: '18–39 years (Young Adult)' },
   108→  { value: '40-59', label: '40–59 years (Middle Age)' },

   110→  { value: '75+', label: '75+ years (Elderly)' },
   111→]

   114→  { value: 'MALE', label: 'Male' },
   115→  { value: 'FEMALE', label: 'Female' },
   116→  { value: 'OTHER', label: 'Other' },
   117→]
   118→
   119→const MODALITY_CONFIG: Record<Modality, { label: string; icon: React.ElementType; color: string }> = {
   120→  ALLOPATHY: { label: 'Allopathy', icon: Pill, color: 'text-teal-600 dark:text-teal-400' },
   121→  AYURVEDA: { label: 'Ayurveda', icon: Leaf, color: 'text-emerald-600 dark:text-emerald-400' },
   122→  HOMEOPATHY: { label: 'Homeopathy', icon: FlaskConical, color: 'text-violet-600 dark:text-violet-400' },
   123→}
   124→
   125→const SEVERITY_CONFIG: Record<string, { label: string; className: string }> = {
   126→  LOW: { label: 'Low', className: 'bg-green-100 text-green-800 border-green-300 dark:bg-green-900 dark:text-green-200 dark:border-green-700' },
   127→  MODERATE: { label: 'Moderate', className: 'bg-amber-100 text-amber-800 border-amber-300 dark:bg-amber-900 dark:text-amber-200 dark:border-amber-700' },
   128→  HIGH: { label: 'High', className: 'bg-orange-100 text-orange-800 border-orange-300 dark:bg-orange-900 dark:text-orange-200 dark:border-orange-700' },
   129→  CRITICAL: { label: 'Critical', className: 'bg-red-100 text-red-800 border-red-300 dark:bg-red-900 dark:text-red-200 dark:border-red-700' },
   130→}
   131→
   132→const BODY_SYSTEM_ICONS: Record<string, React.ElementType> = {
   133→  NERVOUS: Brain,
   134→  CARDIOVASCULAR: Heart,
   135→  RESPIRATORY: Activity,
   136→  MUSCULOSKELETAL: Zap,
   137→  DIGESTIVE: Thermometer,
   138→}
   139→
   140→const fadeSlide = {
   141→  initial: { opacity: 0, y: 12 },
   142→  animate: { opacity: 1, y: 0 },
   143→  transition: { duration: 0.3 },
   144→}

   146→// ──────────────────────────────────────────────
   147→// Component
   148→// ──────────────────────────────────────────────
   149→
   150→export function SymptomCheckerSection() {
   151→  const { activeModality } = useAppStore()
   152→
   153→  // Form state
   154→  const [symptoms, setSymptoms] = useState<string[]>([])
   155→  const [symptomInput, setSymptomInput] = useState('')
   156→  const [age, setAge] = useState('')
   157→  const [gender, setGender] = useState('')
   158→  const [modality, setModality] = useState<Modality>(activeModality)
   159→
   160→  // Suggestion state
   161→  const [showSuggestions, setShowSuggestions] = useState(false)
   162→  const [filteredSuggestions, setFilteredSuggestions] = useState<string[]>([])
   163→  const inputRef = useRef<HTMLInputElement>(null)
   164→  const suggestionsRef = useRef<HTMLDivElement>(null)
   165→
   166→  // Results state
   167→  const [results, setResults] = useState<SymptomCheckResult | null>(null)
   168→  const [loading, setLoading] = useState(false)
   169→  const [expandedIssue, setExpandedIssue] = useState<string | null>(null)
   170→
   171→  // ── Symptom management ─────────────────────
   172→
   173→  const addSymptom = (symptom: string) => {
   174→    const trimmed = symptom.trim().toLowerCase()
   175→    if (trimmed && !symptoms.includes(trimmed)) {
   176→      setSymptoms((prev) => [...prev, trimmed])
   177→    }
   178→    setSymptomInput('')
   179→    setShowSuggestions(false)
   180→    inputRef.current?.focus()
   181→  }
   182→
   183→  const removeSymptom = (symptom: string) => {
   184→    setSymptoms((prev) => prev.filter((s) => s !== symptom))
   185→  }
   186→
   187→  const handleSymptomKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   188→    if (e.key === 'Enter' && symptomInput.trim()) {
   189→      e.preventDefault()
   190→      addSymptom(symptomInput)
   191→    }
   192→    if (e.key === 'Backspace' && !symptomInput && symptoms.length > 0) {
   193→      setSymptoms((prev) => prev.slice(0, -1))
   194→    }
   195→  }
   196→
   197→  const handleSymInputChange = (value: string) => {
   198→    setSymptomInput(value)
   199→    if (value.trim()) {
   200→      const lower = value.toLowerCase()
   201→      const filtered = COMMON_SYMPTOMS.filter(
   202→        (s) => s.includes(lower) && !symptoms.includes(s)
   203→      )
   204→      setFilteredSuggestions(filtered.slice(0, 8))
   205→      setShowSuggestions(filtered.length > 0)
   206→    } else {
   207→      setShowSuggestions(false)
   208→    }
   209→  }
   210→
   211→  // Close suggestions on outside click
   212→  useEffect(() => {
   213→    const handler = (e: MouseEvent) => {
   214→      if (
   215→        suggestionsRef.current &&
   216→        !suggestionsRef.current.contains(e.target as Node) &&
   217→        inputRef.current &&
   218→        !inputRef.current.contains(e.target as Node)
   219→      ) {
   220→        setShowSuggestions(false)
   221→      }
   222→    }
   223→    document.addEventListener('mousedown', handler)
   224→    return () => document.removeEventListener('mousedown', handler)
   225→  }, [])
   226→
   227→  // ── Submit handler ─────────────────────────
   228→
   229→  const handleCheckSymptoms = async () => {
   230→    if (symptoms.length === 0) {
   231→      toast({ title: 'Add symptoms', description: 'Please add at least one symptom', variant: 'destructive' })
   232→      return
   233→    }
   234→    if (!age) {
   235→      toast({ title: 'Select age group', description: 'Please select your age range', variant: 'destructive' })
   236→      return
   237→    }
   238→    if (!gender) {
   239→      toast({ title: 'Select gender', description: 'Please select your gender', variant: 'destructive' })
   240→      return
   241→    }
   242→
   243→    setLoading(true)
   244→    setResults(null)
   245→    setExpandedIssue(null)
   246→
   247→    try {
   248→      const res = await fetch('/api/symptom-checker', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ symptoms, age, gender, modality }),
   252→      })
   253→      const data = await res.json()
   254→      const result: SymptomCheckResult = data.data ?? data
   255→      setResults(result)
   256→      toast({
   257→        title: 'Symptom Check Complete',
   258→        description: `${result.matchedIssues?.length ?? 0} possible conditions found`,
   259→      })
   260→    } catch {
   261→      toast({ title: 'Error', description: 'Failed to check symptoms', variant: 'destructive' })
   262→    } finally {
   263→      setLoading(false)
   264→    }
   265→  }
   266→
   267→  // ── Helpers ────────────────────────────────
   268→
   269→  const canSubmit = symptoms.length > 0 && age && gender
   270→
   271→  const getSeverityBadge = (severity: string) => {
   272→    const config = SEVERITY_CONFIG[severity] ?? SEVERITY_CONFIG.MODERATE
   273→    return (
   274→      <Badge variant="outline" className={config.className}>
   275→        {config.label}
   276→      </Badge>
   277→    )
   278→  }
   279→
   280→  const getRelevanceColor = (score: number) => {
   281→    if (score >= 80) return 'text-red-600 dark:text-red-400'
   282→    if (score >= 60) return 'text-orange-600 dark:text-orange-400'
   283→    if (score >= 40) return 'text-amber-600 dark:text-amber-400'
   284→    return 'text-green-600 dark:text-green-400'
   285→  }
   286→
   287→  const getWarningIcon = (severity: string) => {
   288→    if (severity === 'CRITICAL') return <ShieldAlert className="h-4 w-4 text-red-500" />
   289→    if (severity === 'WARNING') return <AlertTriangle className="h-4 w-4 text-amber-500" />
   290→    return <Info className="h-4 w-4 text-teal-500" />
   291→  }
   292→
   293→  // ── Render ─────────────────────────────────
   294→
   295→  return (
   296→    <TooltipProvider>
   297→      <motion.div {...fadeSlide} className="space-y-6">
   298→        {/* ── Input Card ─────────────────────── */}
   299→        <Card>
   300→          <CardHeader>
   301→            <CardTitle className="text-lg flex items-center gap-2">
   302→              <Stethoscope className="h-5 w-5" />
   303→              Symptom Checker
   304→            </CardTitle>
   305→            <CardDescription>
   306→              Enter your symptoms to get potential health insights across modalities
   307→            </CardDescription>
   308→          </CardHeader>
   309→          <CardContent className="space-y-5">
   310→            {/* Symptom Tags Input */}
   311→            <div className="space-y-2">
   312→              <Label>Symptoms</Label>
   313→              <div className="flex flex-wrap items-center gap-2 p-3 border rounded-lg bg-background min-h-[48px] focus-within:ring-2 focus-within:ring-ring focus-within:ring-offset-2">
   314→                <AnimatePresence>
   315→                  {symptoms.map((symptom) => (
   316→                    <motion.span
   317→                      key={symptom}
   318→                      initial={{ opacity: 0, scale: 0.8 }}
   319→                      animate={{ opacity: 1, scale: 1 }}
   320→                      exit={{ opacity: 0, scale: 0.8 }}
   321→                      className="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   322→                    >
   323→                      {symptom}
   324→                      <button
   325→                        type="button"
   326→                        onClick={() => removeSymptom(symptom)}
   327→                        className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   328→                      >
   329→                        <X className="h-3 w-3" />
   330→                      </button>
   331→                    </motion.span>
   332→                  ))}
   333→                </AnimatePresence>
   334→                <div className="relative flex-1 min-w-[140px]">
   335→                  <input
   336→                    ref={inputRef}
   337→                    type="text"
   338→                    value={symptomInput}
   339→                    onChange={(e) => handleSymInputChange(e.target.value)}
   340→                    onKeyDown={handleSymptomKeyDown}
   341→                    onFocus={() => {
   342→                      if (symptomInput.trim()) handleSymInputChange(symptomInput)
   343→                    }}
   344→                    placeholder={symptoms.length === 0 ? 'Type a symptom (e.g. headache, fever, pet dard)...' : 'Add more...'}
   345→                    className="w-full bg-transparent outline-none text-sm placeholder:text-muted-foreground"
   346→                  />
   347→                </div>
   348→              </div>
   349→              {/* Suggestions Dropdown */}
   350→              <AnimatePresence>
   351→                {showSuggestions && filteredSuggestions.length > 0 && (
   352→                  <motion.div
   353→                    ref={suggestionsRef}
   354→                    initial={{ opacity: 0, y: -4 }}
   355→                    animate={{ opacity: 1, y: 0 }}
   356→                    exit={{ opacity: 0, y: -4 }}
   357→                    className="border rounded-lg bg-popover shadow-md overflow-hidden z-10"
   358→                  >
   359→                    <ScrollArea className="max-h-48">
   360→                      {filteredSuggestions.map((suggestion) => (
   361→                        <button
   362→                          key={suggestion}
   363→                          type="button"
   364→                          onClick={() => addSymptom(suggestion)}
   365→                          className="w-full text-left px-3 py-2 text-sm hover:bg-accent transition-colors flex items-center gap-2"
   366→                        >
   367→                          <Plus className="h-3.5 w-3.5 text-muted-foreground" />
   368→                          {suggestion}
   369→                        </button>
   370→                      ))}
   371→                    </ScrollArea>
   372→                  </motion.div>
   373→                )}
   374→              </AnimatePresence>
   375→              {symptoms.length === 0 && (
   376→                <p className="text-xs text-muted-foreground flex items-center gap-1">
   377→                  <Info className="h-3 w-3" />
   378→                  Press Enter or select from suggestions to add symptoms
   379→                </p>
   380→              )}
   381→            </div>
   382→
   383→            {/* Age, Gender, Modality Row */}
   384→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   385→              {/* Age */}
   386→              <div className="space-y-2">
   387→                <Label>Age Group</Label>
   388→                <Select value={age} onValueChange={setAge}>
   389→                  <SelectTrigger>
   390→                    <SelectValue placeholder="Select age..." />
   391→                  </SelectTrigger>
   392→                  <SelectContent>
   393→                    {AGE_OPTIONS.map((opt) => (
   394→                      <SelectItem key={opt.value} value={opt.value}>
   395→                        {opt.label}
   396→                      </SelectItem>
   397→                    ))}
   398→                  </SelectContent>
   399→                </Select>
   400→              </div>
   401→
   402→              {/* Gender */}
   403→              <div className="space-y-2">
   404→                <Label>Gender</Label>
   405→                <Select value={gender} onValueChange={setGender}>
   406→                  <SelectTrigger>
   407→                    <SelectValue placeholder="Select gender..." />
   408→                  </SelectTrigger>
   409→                  <SelectContent>
   410→                    {GENDER_OPTIONS.map((opt) => (
   411→                      <SelectItem key={opt.value} value={opt.value}>
   412→                        {opt.label}
   413→                      </SelectItem>
   414→                    ))}
   415→                  </SelectContent>
   416→                </Select>
   417→              </div>
   418→
   419→              {/* Modality */}
   420→              <div className="space-y-2">
   421→                <Label>Treatment Modality</Label>
   422→                <Select value={modality} onValueChange={(v) => setModality(v as Modality)}>
   423→                  <SelectTrigger>
   424→                    <SelectValue />
   425→                  </SelectTrigger>
   426→                  <SelectContent>
   427→                    {Object.entries(MODALITY_CONFIG).map(([key, cfg]) => {
   428→                      const Icon = cfg.icon
   429→                      return (
   430→                        <SelectItem key={key} value={key}>
   431→                          <span className="flex items-center gap-2">
   432→                            <Icon className={`h-4 w-4 ${cfg.color}`} />
   433→                            {cfg.label}
   434→                          </span>
   435→                        </SelectItem>
   436→                      )
   437→                    })}
   438→                  </SelectContent>
   439→                </Select>
   440→              </div>
   441→            </div>
   442→
   443→            {/* Check Button */}
   444→            <div className="flex items-center gap-3">
   445→              <Button
   446→                onClick={handleCheckSymptoms}
   447→                disabled={!canSubmit || loading}
   448→                className="gap-2 min-w-[180px]"
   449→              >
   450→                {loading ? (
   451→                  <>
   452→                    <Loader2 className="h-4 w-4 animate-spin" />
   453→                    Checking...
   454→                  </>
   455→                ) : (
   456→                  <>
   457→                    <Search className="h-4 w-4" />
   458→                    Check Symptoms
   459→                  </>
   460→                )}
   461→              </Button>
   462→              {symptoms.length > 0 && (
   463→                <Button
   464→                  variant="ghost"
   465→                  size="sm"
   466→                  onClick={() => {
   467→                    setSymptoms([])
   468→                    setSymptomInput('')
   469→                    setResults(null)
   470→                  }}
   471→                  className="text-muted-foreground"
   472→                >
   473→                  Clear all
   474→                </Button>
   475→              )}
   476→            </div>
   477→          </CardContent>
   478→        </Card>
   479→
   480→        {/* ── Loading Skeleton ────────────────── */}
   481→        <AnimatePresence>
   482→          {loading && (
   483→            <motion.div
   484→              initial={{ opacity: 0 }}
   485→              animate={{ opacity: 1 }}
   486→              exit={{ opacity: 0 }}
   487→              className="space-y-4"
   488→            >
   489→              <Card>
   490→                <CardContent className="p-6 space-y-4">
   491→                  <Skeleton className="h-6 w-48" />
   492→                  <Skeleton className="h-24 w-full" />
   493→                  <Skeleton className="h-24 w-full" />
   494→                  <Skeleton className="h-24 w-full" />
   495→                </CardContent>
   496→              </Card>
   497→            </motion.div>
   498→          )}
   499→        </AnimatePresence>
   500→
   501→        {/* ── Results ─────────────────────────── */}
   502→        <AnimatePresence>
   503→          {results && !loading && (
   504→            <motion.div
   505→              initial={{ opacity: 0, y: 16 }}
   506→              animate={{ opacity: 1, y: 0 }}
   507→              transition={{ duration: 0.4 }}
   508→              className="space-y-4"
   509→            >
   510→              {/* Summary */}
   511→              {results.summary && (
   512→                <div className="p-3 rounded-lg bg-muted/50 border text-sm text-muted-foreground">
   513→                  {results.summary}
   514→                </div>
   515→              )}
   516→
   517→              {/* Matched Issues */}
   518→              {results.matchedIssues && results.matchedIssues.length > 0 ? (
   519→                <div className="space-y-3">
   520→                  <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wider">
   521→                    Possible Conditions ({results.matchedIssues.length})
   522→                  </h3>
   523→                  {results.matchedIssues.map((issue, index) => {
   524→                    const SystemIcon = BODY_SYSTEM_ICONS[issue.bodySystem?.toUpperCase()] ?? Activity
   525→                    const isExpanded = expandedIssue === issue.id
   526→                    const wingForModality = issue.wingApproaches?.find(
   527→                      (w) => w.modality === modality
   528→                    )
   529→
   530→                    return (
   531→                      <motion.div
   532→                        key={issue.id}
   533→                        initial={{ opacity: 0, y: 8 }}
   534→                        animate={{ opacity: 1, y: 0 }}
   535→                        transition={{ duration: 0.25, delay: index * 0.05 }}
   536→                      >
   537→                        <Card className="overflow-hidden hover:shadow-md transition-shadow">
   538→                          <CardContent className="p-0">
   539→                            {/* Issue Header */}
   540→                            <button
   541→                              type="button"
   542→                              onClick={() => setExpandedIssue(isExpanded ? null : issue.id)}
   543→                              className="w-full text-left p-4 flex items-start gap-4 hover:bg-accent/30 transition-colors"
   544→                            >
   545→                              <div className="shrink-0 mt-0.5">
   546→                                <div className="h-10 w-10 rounded-lg bg-primary/10 flex items-center justify-center">
   547→                                  <SystemIcon className="h-5 w-5 text-primary" />
   548→                                </div>
   549→                              </div>
   550→                              <div className="flex-1 min-w-0">
   551→                                <div className="flex items-center gap-2 flex-wrap">
   552→                                  <span className="font-semibold text-base">{issue.name}</span>
   553→                                  {getSeverityBadge(issue.severity)}
   554→                                  <Badge variant="outline" className="text-xs bg-muted text-muted-foreground">
   555→                                    {issue.bodySystem}
   556→                                  </Badge>
   557→                                </div>
   558→                                {issue.description && (
   559→                                  <p className="mt-1 text-sm text-muted-foreground line-clamp-2">
   560→                                    {issue.description}
   561→                                  </p>
   562→                                )}
   563→                                <div className="mt-2 flex items-center gap-3">
   564→                                  <div className="flex items-center gap-1.5">
   565→                                    <span className="text-xs text-muted-foreground">Relevance</span>
   566→                                    <div className="w-24">
   567→                                      <Progress
   568→                                        value={issue.relevanceScore}

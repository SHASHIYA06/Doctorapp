    98→// ── Helpers ────────────────────────────────────────────────────
    99→
   100→const getAvailabilityStatus = (item: InventoryItem): AvailabilityStatus => {
   101→  if (item.currentStock === 0) return 'OUT_OF_STOCK'
   102→  if (item.currentStock < item.threshold) return 'LOW_STOCK'
   103→  return 'IN_STOCK'
   104→}
   105→
   106→const isExpiringSoon = (expiryDate: string, days: number = 30): boolean => {
   107→  const expiry = new Date(expiryDate)
   108→  const now = new Date()
   109→  const diff = (expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
   110→  return diff > 0 && diff <= days
   111→}
   112→
   113→const isExpired = (expiryDate: string): boolean => {
   114→  return new Date(expiryDate) < new Date()
   115→}
   116→
   117→const formatDate = (dateStr: string) => {
   118→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   119→}
   120→
   121→const statusBadge = (status: AvailabilityStatus) => {
   122→  switch (status) {
   123→    case 'IN_STOCK':
   124→      return <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100">In Stock</Badge>
   125→    case 'LOW_STOCK':
   126→      return <Badge className="bg-red-100 text-red-800 hover:bg-red-100">Low Stock</Badge>
   127→    case 'OUT_OF_STOCK':
   128→      return <Badge className="bg-gray-100 text-gray-800 hover:bg-gray-100">Out of Stock</Badge>
   129→  }
   130→}

   276→  }
   277→
   278→  if (loading) {

   398→                  ))}
   399→                </div>
   400→              </CardContent>
   401→            </Card>
   402→          )}
   403→        </motion.div>
   404→      )}
   405→
   406→      {/* Search & Filter */}
   407→      <motion.div {...fadeSlide}>
   408→        <Card>
   409→          <CardContent className="p-4">
   410→            <div className="flex flex-col sm:flex-row gap-3">

   479→                    return (
   480→                      <TableRow key={item.id}>
   481→                        <TableCell>
   482→                          <div>
   483→                            <span className="font-medium">{item.medicineName}</span>
   484→                            <span className="text-muted-foreground text-xs block">{item.genericName} • {item.category}</span>
   485→                          </div>
   486→                        </TableCell>
   487→                        <TableCell>
   488→                          <span className="font-mono text-xs">{item.batchNumber}</span>
   489→                        </TableCell>
   490→                        <TableCell>
   491→                          <span className={status === 'LOW_STOCK' ? 'text-red-600 font-semibold' : status === 'OUT_OF_STOCK' ? 'text-gray-500' : ''}>
   492→                            {item.currentStock} {item.unit}{item.currentStock !== 1 ? 's' : ''}
   493→                          </span>
   494→                        </TableCell>
   495→                        <TableCell>
   496→                          <span className="flex items-center gap-0.5">
   497→                            <IndianRupee className="h-3 w-3" />{item.mrp}
   498→                          </span>
   499→                        </TableCell>
   500→                        <TableCell>
   501→                          <div className="flex items-center gap-1">
   502→                            <span className={expired ? 'text-red-600 font-semibold' : expiring ? 'text-amber-600 font-semibold' : ''}>
   503→                              {formatDate(item.expiryDate)}
   504→                            </span>
   505→                            {expired && <Badge variant="destructive" className="text-[10px] px-1 py-0">EXPIRED</Badge>}
   506→                            {!expired && expiring && <Badge className="bg-amber-100 text-amber-800 text-[10px] px-1 py-0 hover:bg-amber-100">EXPIRING</Badge>}
   507→                          </div>
   508→                        </TableCell>
   509→                        <TableCell>{statusBadge(status)}</TableCell>
   510→                        <TableCell className="text-right">
   511→                          <Dialog open={updateDialogOpen && selectedItem?.id === item.id} onOpenChange={(open) => {
   512→                            if (open) {
   513→                              setSelectedItem(item)
   514→                              setUpdateForm({ itemId: item.id, action: 'add', quantity: 0, reason: '' })
   515→                            } else {
   516→                              setUpdateDialogOpen(false)
   517→                              setSelectedItem(null)
   518→                            }
   519→                          }}>
   520→                            <DialogTrigger asChild>
   521→                              <Button variant="ghost" size="sm" onClick={() => {
   522→                                setSelectedItem(item)
   523→                                setUpdateForm({ itemId: item.id, action: 'add', quantity: 0, reason: '' })
   524→                                setUpdateDialogOpen(true)
   525→                              }}>
   526→                                <Plus className="h-4 w-4" />
   527→                              </Button>
   528→                            </DialogTrigger>
   529→                            <DialogContent>
   530→                              <DialogHeader>
   531→                                <DialogTitle>Update Stock — {item.medicineName}</DialogTitle>
   532→                                <DialogDescription>
   533→                                  Current stock: {item.currentStock} {item.unit}s | Batch: {item.batchNumber}
   534→                                </DialogDescription>
   535→                              </DialogHeader>
   536→                              <div className="space-y-4 py-2">
   537→                                <Select value={updateForm.action} onValueChange={(v) => setUpdateForm(prev => ({ ...prev, action: v as 'add' | 'reduce' }))}>
   538→                                  <SelectTrigger>
   539→                                    <SelectValue placeholder="Action" />
   540→                                  </SelectTrigger>
   541→                                  <SelectContent>
   542→                                    <SelectItem value="add">Add Stock</SelectItem>
   543→                                    <SelectItem value="reduce">Reduce Stock</SelectItem>
   544→                                  </SelectContent>
   545→                                </Select>
   546→                                <Input
   547→                                  type="number"
   548→                                  min={1}
   549→                                  placeholder="Quantity"
   550→                                  value={updateForm.quantity || ''}
   551→                                  onChange={e => setUpdateForm(prev => ({ ...prev, quantity: parseInt(e.target.value) || 0 }))}
   552→                                />
   553→                                <Input
   554→                                  placeholder="Reason (e.g., New shipment, Damaged, Returned)"
   555→                                  value={updateForm.reason}
   556→                                  onChange={e => setUpdateForm(prev => ({ ...prev, reason: e.target.value }))}
   557→                                />
   558→                              </div>
   559→                              <DialogFooter>
   560→                                <Button variant="outline" onClick={() => setUpdateDialogOpen(false)}>Cancel</Button>
   561→                                <Button
   562→                                  onClick={handleStockUpdate}
   563→                                  disabled={updateForm.quantity <= 0}
   564→                                  className={updateForm.action === 'add' ? 'bg-emerald-600 hover:bg-emerald-700' : 'bg-red-600 hover:bg-red-700'}
   565→                                >
   566→                                  {updateForm.action === 'add' ? <Plus className="h-4 w-4 mr-1" /> : <Minus className="h-4 w-4 mr-1" />}
   567→                                  {updateForm.action === 'add' ? 'Add' : 'Reduce'} {updateForm.quantity || 0} Units
   568→                                </Button>
   569→                              </DialogFooter>
   570→                            </DialogContent>
   571→                          </Dialog>
   572→                        </TableCell>
   573→                      </TableRow>
   574→                    )
   575→                  })
   576→                )}
   577→              </TableBody>
   578→            </Table>
   579→          </CardContent>
   580→        </Card>
   581→      </motion.div>
   582→
   583→      {/* Out of Stock Summary */}
   584→      {outOfStockItems.length > 0 && (
   585→        <motion.div {...fadeSlide}>
   586→          <Card className="border-red-200">
   587→            <CardHeader>
   588→              <CardTitle className="text-lg flex items-center gap-2 text-red-700">
   589→                <XCircle className="h-5 w-5" /> Out of Stock Items
   590→              </CardTitle>
   591→              <CardDescription>These items need immediate restocking</CardDescription>
   592→            </CardHeader>
   593→            <CardContent>
   594→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   595→                {outOfStockItems.map(item => (
   596→                  <div key={item.id} className="flex items-center justify-between p-3 rounded-lg border border-red-100 bg-red-50/30">
   597→                    <div>
   598→                      <p className="font-medium text-sm">{item.medicineName}</p>
   599→                      <p className="text-xs text-muted-foreground">{item.genericName} • {item.manufacturer}</p>
   600→                    </div>
   601→                    <Badge variant="destructive" className="text-[10px]">OUT</Badge>
   602→                  </div>
   603→                ))}
   604→              </div>
   605→            </CardContent>
   606→          </Card>
   607→        </motion.div>
   608→      )}
   609→    </div>
   610→  )
   611→}

    16→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    17→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'

    25→// ── Types ────────────────────────────────────────────────────────────────
    26→
    27→type NoteType = 'SOAP' | 'PROGRESS' | 'DISCHARGE' | 'REFERRAL' | 'PROCEDURE'
    28→
    29→interface VitalSigns {
    30→  bp?: string
    31→  hr?: string
    32→  temp?: string
    33→  spo2?: string
    34→  rr?: string
    35→  weight?: string
    36→}
    37→
    38→interface SOAPData {
    39→  subjective: string
    40→  objective: string
    41→  assessment: string
    42→  plan: string
    43→}
    44→
    45→interface ClinicalNote {
    46→  id: string
    47→  patientId: string
    48→  patientName: string
    49→  type: NoteType
    50→  modality: Modality
    51→  soap: SOAPData
    52→  vitals: VitalSigns
    53→  signed: boolean
    54→  signedBy: string | null
    55→  signedAt: string | null
    56→  practitionerName: string
    57→  summary: string | null
    58→  createdAt: string
    59→  updatedAt: string
    60→}
    61→
    62→interface Patient {
    63→  id: string
    64→  firstName: string
    65→  lastName: string
    66→}

    69→

    71→
    72→const SOAP_COLORS = {
    73→  subjective: 'border-l-blue-500 bg-blue-50/50 dark:bg-blue-950/20',
    74→  objective: 'border-l-green-500 bg-green-50/50 dark:bg-green-950/20',
    75→  assessment: 'border-l-amber-500 bg-amber-50/50 dark:bg-amber-950/20',

    77→}
    78→

    80→  subjective: { letter: 'S', label: 'Subjective', color: 'text-blue-700 dark:text-blue-300', icon: User },
    81→  objective: { letter: 'O', label: 'Objective', color: 'text-green-700 dark:text-green-300', icon: Activity },
    82→  assessment: { letter: 'A', label: 'Assessment', color: 'text-amber-700 dark:text-amber-300', icon: Stethoscope },
    83→  plan: { letter: 'P', label: 'Plan', color: 'text-teal-700 dark:text-teal-300', icon: Heart },

    88→  { key: 'hr', label: 'HR', unit: 'bpm', placeholder: '72' },
    89→  { key: 'temp', label: 'Temp', unit: '°C', placeholder: '37.0' },
    90→  { key: 'spo2', label: 'SpO2', unit: '%', placeholder: '98' },
    91→  { key: 'rr', label: 'RR', unit: '/min', placeholder: '16' },
    92→  { key: 'weight', label: 'Weight', unit: 'kg', placeholder: '70' },
    93→]
    94→
    95→const COMMON_DIAGNOSES = [
    96→  'Hypertension', 'Type 2 Diabetes', 'Acute Bronchitis', 'Migraine',
    97→  'Gastroesophageal Reflux', 'Anxiety Disorder', 'Osteoarthritis', 'Asthma',
    98→  'Urinary Tract Infection', 'Iron Deficiency Anemia',
    99→]
   100→
   101→const fadeSlide = {
   102→  initial: { opacity: 0, y: 12 },
   103→  animate: { opacity: 1, y: 0 },
   104→  transition: { duration: 0.3 },
   105→}

   107→// ── Component ────────────────────────────────────────────────────────────
   108→
   109→export function ClinicalNotesSection() {
   110→  const { activeModality, selectedPatientId } = useAppStore()
   111→
   112→  // Data
   113→  const [notes, setNotes] = useState<ClinicalNote[]>([])
   114→  const [patients, setPatients] = useState<Patient[]>([])
   115→  const [loading, setLoading] = useState(true)
   116→
   117→  // Modality filter
   118→  const [modalityFilter, setModalityFilter] = useState<Modality>(activeModality)
   119→
   120→  // Search & filter
   121→  const [searchQuery, setSearchQuery] = useState('')
   122→  const [typeFilter, setTypeFilter] = useState<NoteType | 'ALL'>('ALL')
   123→
   124→  // New note form
   125→  const [isCreating, setIsCreating] = useState(false)
   126→  const [formPatientId, setFormPatientId] = useState(selectedPatientId ?? '')
   127→  const [formType, setFormType] = useState<NoteType>('SOAP')
   128→  const [formModality, setFormModality] = useState<Modality>(activeModality)
   129→  const [formSOAP, setFormSOAP] = useState<SOAPData>({ subjective: '', objective: '', assessment: '', plan: '' })
   130→  const [formVitals, setFormVitals] = useState<VitalSigns>({})
   131→  const [formPractitioner, setFormPractitioner] = useState('Dr. Sharma')
   132→  const [formSubmitting, setFormSubmitting] = useState(false)
   133→
   134→  // AI summary
   135→  const [aiGenerating, setAiGenerating] = useState(false)
   136→
   137→  // Detail view
   138→  const [selectedNote, setSelectedNote] = useState<ClinicalNote | null>(null)
   139→
   140→  // Sign dialog
   141→  const [signDialogNote, setSignDialogNote] = useState<ClinicalNote | null>(null)
   142→  const [signSubmitting, setSignSubmitting] = useState(false)
   143→
   144→  // Diagnosis suggestions
   145→  const [showDiagnosisSuggestions, setShowDiagnosisSuggestions] = useState(false)
   146→
   147→  // ── Load Data ──────────────────────────────────────────────────────────
   148→
   149→  const loadData = useCallback(async () => {
   150→    try {
   151→      const [notesRes, patRes] = await Promise.all([
   152→        fetch('/api/clinical-notes').then((r) => r.json()),
   153→        fetch('/api/patients').then((r) => r.json()),
   154→      ])
   155→      const noteList = (notesRes.data ?? notesRes.notes ?? []).map((n: Record<string, unknown>) => ({
   156→        id: n.id as string,
   157→        patientId: (n.patientId ?? '') as string,
   158→        patientName: (n.patientName ?? (n.patient ? `${(n.patient as Record<string, string>).firstName} ${(n.patient as Record<string, string>).lastName}` : 'Unknown')) as string,
   159→        type: (n.type ?? 'SOAP') as NoteType,
   160→        modality: (n.modality ?? 'ALLOPATHY') as Modality,
   161→        soap: (n.soap ?? { subjective: '', objective: '', assessment: '', plan: '' }) as SOAPData,
   162→        vitals: (n.vitals ?? {}) as VitalSigns,
   163→        signed: (n.signed ?? false) as boolean,
   164→        signedBy: (n.signedBy ?? null) as string | null,
   165→        signedAt: (n.signedAt ?? null) as string | null,
   166→        practitionerName: (n.practitionerName ?? 'Unknown') as string,
   167→        summary: (n.summary ?? null) as string | null,
   168→        createdAt: (n.createdAt ?? new Date().toISOString()) as string,
   169→        updatedAt: (n.updatedAt ?? new Date().toISOString()) as string,
   170→      }))
   171→      setNotes(noteList)
   172→      setPatients(patRes.data ?? patRes.patients ?? [])
   173→    } catch {
   174→      toast({ title: 'Error', description: 'Failed to load clinical notes', variant: 'destructive' })
   175→    } finally {
   176→      setLoading(false)
   177→    }
   178→  }, [])
   179→
   180→  useEffect(() => { loadData() }, [loadData])
   181→
   182→  // ── Filtered Notes ─────────────────────────────────────────────────────
   183→
   184→  const filteredNotes = useMemo(() => {
   185→    let list = notes.filter((n) => n.modality === modalityFilter)
   186→
   187→    if (typeFilter !== 'ALL') {
   188→      list = list.filter((n) => n.type === typeFilter)
   189→    }
   190→
   191→    if (searchQuery.trim()) {
   192→      const q = searchQuery.toLowerCase()
   193→      list = list.filter(
   194→        (n) =>
   195→          n.patientName.toLowerCase().includes(q) ||
   196→          n.soap.subjective.toLowerCase().includes(q) ||
   197→          n.soap.assessment.toLowerCase().includes(q) ||
   198→          n.practitionerName.toLowerCase().includes(q)
   199→      )
   200→    }
   201→
   202→    return list.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   203→  }, [notes, modalityFilter, typeFilter, searchQuery])
   204→
   205→  // ── SOAP Update Helper ─────────────────────────────────────────────────
   206→
   207→  const updateSOAP = (field: keyof SOAPData, value: string) => {
   208→    setFormSOAP((prev) => ({ ...prev, [field]: value }))
   209→  }
   210→
   211→  const updateVital = (key: keyof VitalSigns, value: string) => {
   212→    setFormVitals((prev) => ({ ...prev, [key]: value }))
   213→  }
   214→
   215→  // ── AI Summary ─────────────────────────────────────────────────────────
   216→
   217→  const handleAISummary = async () => {
   218→    if (!formSOAP.subjective && !formSOAP.objective && !formSOAP.assessment) {
   219→      toast({ title: 'No Content', description: 'Add some clinical content first', variant: 'destructive' })
   220→      return
   221→    }
   222→    setAiGenerating(true)
   223→    try {
   224→      const res = await fetch('/api/ai-draft', {
   225→        method: 'POST',
   226→        headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify({
   228→          type: 'CLINICAL_SUMMARY',
   229→          context: {
   230→            subjective: formSOAP.subjective,
   231→            objective: formSOAP.objective,
   232→            assessment: formSOAP.assessment,
   233→            plan: formSOAP.plan,
   234→            vitals: formVitals,
   235→            modality: formModality,
   236→          },
   237→        }),
   238→      })
   239→      if (res.ok) {
   240→        const data = await res.json()
   241→        const summary = data.summary ?? data.content ?? ''
   242→        if (summary) {
   243→          // Append AI summary to the plan field
   244→          updateSOAP('plan', formSOAP.plan ? `${formSOAP.plan}\n\n[AI Summary]: ${summary}` : `[AI Summary]: ${summary}`)
   245→          toast({ title: 'AI Summary Generated', description: 'Added to Plan section' })
   246→        }
   247→      } else {
   248→        toast({ title: 'AI Error', description: 'Failed to generate summary', variant: 'destructive' })
   249→      }
   250→    } catch {
   251→      // Fallback: generate a local summary
   252→      const localSummary = `Patient presents with ${formSOAP.subjective.slice(0, 100) || 'reported symptoms'}. Assessment: ${formSOAP.assessment.slice(0, 100) || 'pending'}.`
   253→      updateSOAP('plan', formSOAP.plan ? `${formSOAP.plan}\n\n[AI Summary]: ${localSummary}` : `[AI Summary]: ${localSummary}`)
   254→      toast({ title: 'AI Summary (Local)', description: 'Generated locally as fallback' })
   255→    } finally {
   256→      setAiGenerating(false)
   257→    }
   258→  }
   259→
   260→  // ── Save Note ──────────────────────────────────────────────────────────
   261→
   262→  const handleSave = async () => {
   263→    if (!formPatientId) {
   264→      toast({ title: 'Validation', description: 'Select a patient', variant: 'destructive' })
   265→      return
   266→    }
   267→    setFormSubmitting(true)
   268→    try {
   269→      const patient = patients.find((p) => p.id === formPatientId)
   270→      const res = await fetch('/api/clinical-notes', {
   271→        method: 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify({
   274→          patientId: formPatientId,
   275→          patientName: patient ? `${patient.firstName} ${patient.lastName}` : 'Unknown',
   276→          type: formType,
   277→          modality: formModality,
   278→          soap: formSOAP,
   279→          vitals: formVitals,
   280→          practitionerName: formPractitioner,
   281→        }),
   282→      })
   283→      if (res.ok) {
   284→        toast({ title: 'Note Saved', description: `${formType} note created` })
   285→        resetForm()
   286→        loadData()
   287→      } else {
   288→        const err = await res.json()
   289→        toast({ title: 'Error', description: err.error ?? 'Failed to save note', variant: 'destructive' })
   290→      }
   291→    } catch {
   292→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   293→    } finally {
   294→      setFormSubmitting(false)
   295→    }
   296→  }
   297→
   298→  // ── Sign Note ──────────────────────────────────────────────────────────
   299→
   300→  const handleSign = async () => {
   301→    if (!signDialogNote) return
   302→    setSignInSubmitting(true)
   303→    try {
   304→      const res = await fetch(`/api/clinical-notes/${signDialogNote.id}`, {
   305→        method: 'PATCH',
   306→        headers: { 'Content-Type': 'application/json' },
   307→        body: JSON.stringify({
   308→          signed: true,
   309→          signedBy: signDialogNote.practitionerName,
   310→          signedAt: new Date().toISOString(),
   311→        }),
   312→      })
   313→      if (res.ok) {
   314→        toast({ title: 'Note Signed', description: 'Note is now locked' })
   315→        setSignDialogNote(null)
   316→        setSelectedNote(null)
   317→        loadData()
   318→      } else {
   319→        toast({ title: 'Error', description: 'Failed to sign note', variant: 'destructive' })
   320→      }
   321→    } catch {
   322→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   323→    } finally {
   324→      setSignInSubmitting(false)
   325→    }
   326→  }
   327→
   328→  // ── Reset Form ─────────────────────────────────────────────────────────
   329→
   330→  const resetForm = () => {
   331→    setIsCreating(false)
   332→    setFormPatientId(selectedPatientId ?? '')
   333→    setFormType('SOAP')
   334→    setFormModality(activeModality)
   335→    setFormSOAP({ subjective: '', objective: '', assessment: '', plan: '' })
   336→    setFormVitals({})
   337→    setFormPractitioner('Dr. Sharma')
   338→    setShowDiagnosisSuggestions(false)
   339→  }
   340→
   341→  // ── Loading ────────────────────────────────────────────────────────────
   342→
   343→  if (loading) {
   344→    return (
   345→      <div className="space-y-4">
   346→        <Skeleton className="h-10 w-full max-w-sm" />
   347→        <div className="grid gap-4 lg:grid-cols-3">
   348→          <div className="lg:col-span-2 space-y-3">
   349→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-20 w-full rounded-lg" />)}
   350→          </div>
   351→          <Skeleton className="h-64 rounded-lg" />
   352→        </div>
   353→      </div>
   354→    )
   355→  }
   356→
   357→  // ── Render ─────────────────────────────────────────────────────────────
   358→
   359→  return (
   360→    <motion.div {...fadeSlide} className="space-y-4">
   361→      {/* Header */}
   362→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   363→        <div className="flex items-center gap-2">
   364→          <FileText className="h-5 w-5" />
   365→          <h2 className="text-lg font-semibold">Clinical Notes</h2>
   366→        </div>
   367→        <div className="flex items-center gap-2 flex-wrap">
   368→          <div className="relative">
   369→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   370→            <Input
   371→              placeholder="Search notes..."
   372→              value={searchQuery}
   373→              onChange={(e) => setSearchQuery(e.target.value)}
   374→              className="pl-8 h-8 w-48"
   375→            />
   376→          </div>
   377→          <Select value={typeFilter} onValueChange={(v) => setTypeFilter(v as NoteType | 'ALL')}>
   378→            <SelectTrigger className="h-8 w-32">
   379→              <SelectValue />
   380→            </SelectTrigger>
   381→            <SelectContent>
   382→              <SelectItem value="ALL">All Types</SelectItem>
   383→              {NOTE_TYPES.map((t) => (
   384→                <SelectItem key={t} value={t}>{t}</SelectItem>
   385→              ))}
   386→            </SelectContent>
   387→          </Select>
   388→          <Button size="sm" className="gap-1.5" onClick={() => { setIsCreating(true); resetForm(); setIsCreating(true) }}>
   389→            <Plus className="h-4 w-4" />
   390→            New Note
   391→          </Button>
   392→        </div>
   393→      </div>
   394→
   395→      {/* Modality Tabs (NEVER merge) */}
   396→      <div className="flex gap-2">
   397→        {(['ALLOPATHY', 'AYURVEDA', 'HOMEOPATHY'] as Modality[]).map((m) => (
   398→          <Button
   399→            key={m}
   400→            variant={modalityFilter === m ? 'default' : 'outline'}
   401→            size="sm"
   402→            onClick={() => setModalityFilter(m)}
   403→            className="gap-1.5"
   404→          >
   405→            <ModalityBadge modality={m} className="border-0 p-0" />
   406→          </Button>
   407→        ))}
   408→      </div>
   409→
   410→      {/* Main Layout */}
   411→      <div className="grid gap-4 lg:grid-cols-3">
   412→        {/* Left: Note List or Create Form */}
   413→        <div className={selectedNote ? 'lg:col-span-2' : 'lg:col-span-3'}>
   414→          {isCreating ? (
   415→            /* ── Create Form ─────────────────────────────────────────────── */
   416→            <Card>
   417→              <CardHeader>
   418→                <div className="flex items-center justify-between">
   419→                  <CardTitle className="text-base flex items-center gap-2">
   420→                    <PenLine className="h-4 w-4" />
   421→                    New Clinical Note
   422→                  </CardTitle>
   423→                  <Button variant="ghost" size="icon" className="h-7 w-7" onClick={resetForm}>
   424→                    <X className="h-4 w-4" />
   425→                  </Button>
   426→                </div>
   427→              </CardHeader>
   428→              <CardContent className="space-y-5">
   429→                {/* Patient, Type, Modality Row */}
   430→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   431→                  <div className="space-y-2">
   432→                    <Label>Patient *</Label>
   433→                    <Select value={formPatientId} onValueChange={setFormPatientId}>
   434→                      <SelectTrigger><SelectValue placeholder="Select..." /></SelectTrigger>
   435→                      <SelectContent>
   436→                        {patients.map((p) => (
   437→                          <SelectItem key={p.id} value={p.id}>{p.firstName} {p.lastName}</SelectItem>
   438→                        ))}
   439→                      </SelectContent>
   440→                    </Select>
   441→                  </div>
   442→                  <div className="space-y-2">
   443→                    <Label>Note Type</Label>
   444→                    <Select value={formType} onValueChange={(v) => setFormType(v as NoteType)}>
   445→                      <SelectTrigger><SelectValue /></SelectTrigger>
   446→                      <SelectContent>
   447→                        {NOTE_TYPES.map((t) => (
   448→                          <SelectItem key={t} value={t}>{t}</SelectItem>
   449→                        ))}
   450→                      </SelectContent>
   451→                    </Select>
   452→                  </div>
   453→                  <div className="space-y-2">
   454→                    <Label>Care Modality</Label>
   455→                    <div className="flex gap-2 flex-wrap h-9 items-center">
   456→                      {(['ALLOPATHY', 'AYURVEDA', 'HOMEOPATHY'] as Modality[]).map((m) => (
   457→                        <Button
   458→                          key={m}
   459→                          type="button"
   460→                          variant={formModality === m ? 'default' : 'outline'}
   461→                          size="sm"
   462→                          onClick={() => setFormModality(m)}
   463→                        >
   464→                          <ModalityBadge modality={m} className="border-0 p-0" />
   465→                        </Button>
   466→                      ))}
   467→                    </div>
   468→                  </div>
   469→                </div>
   470→
   471→                {/* Practitioner */}
   472→                <div className="space-y-2">
   473→                  <Label>Practitioner</Label>
   474→                  <Input
   475→                    value={formPractitioner}
   476→                    onChange={(e) => setFormPractitioner(e.target.value)}
   477→                    placeholder="Dr. Name"
   478→                    className="max-w-xs"
   479→                  />
   480→                </div>
   481→
   482→                <Separator />
   483→
   484→                {/* Vital Signs Quick-Add (in Objective section) */}
   485→                <div>
   486→                  <div className="flex items-center gap-2 mb-3">
   487→                    <Activity className="h-4 w-4 text-green-600" />
   488→                    <Label className="text-sm font-semibold">Vital Signs</Label>
   489→                  </div>
   490→                  <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-6 gap-3">
   491→                    {VITAL_FIELDS.map(({ key, label, unit, placeholder }) => (
   492→                      <div key={key} className="space-y-1">
   493→                        <Label className="text-xs text-muted-foreground">{label} ({unit})</Label>
   494→                        <Input
   495→                          placeholder={placeholder}
   496→                          value={formVitals[key] ?? ''}
   497→                          onChange={(e) => updateVital(key, e.target.value)}
   498→                          className="h-8 text-sm"
   499→                        />
   500→                      </div>
   501→                    ))}
   502→                  </div>
   503→                </div>
   504→
   505→                <Separator />
   506→
   507→                {/* SOAP Fields */}
   508→                {(['subjective', 'objective', 'assessment', 'plan'] as const).map((field) => {
   509→                  const config = SOAP_LABELS[field]
   510→                  const IconComp = config.icon
   511→                  return (
   512→                    <div
   513→                      key={field}
   514→                      className={`border-l-4 rounded-r-md p-4 ${SOAP_COLORS[field]}`}
   515→                    >
   516→                      <div className="flex items-center justify-between mb-2">
   517→                        <div className="flex items-center gap-2">
   518→                          <span className={`text-lg font-bold ${config.color}`}>{config.letter}</span>
   519→                          <span className={`text-sm font-semibold ${config.color}`}>
   520→                            {config.label}
   521→                          </span>

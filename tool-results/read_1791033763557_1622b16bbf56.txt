   287→// ── Helpers ──────────────────────────────────────────────────────────────
   288→
   289→function formatCurrency(amount: number): string {
   290→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(amount)
   291→}
   292→
   293→function formatDate(iso: string): string {
   294→  return new Date(iso).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
   295→}
   296→
   297→function isOverdue(dueDate: string, status: InvoiceStatus): boolean {
   298→  return status !== 'PAID' && status !== 'CANCELLED' && new Date(dueDate) < new Date()
   299→}
   300→
   301→function claimStatusFlow(current: ClaimStatus): ClaimStatus[] {
   302→  const flow: ClaimStatus[] = ['SUBMITTED', 'UNDER_REVIEW', 'APPROVED', 'SETTLED']
   303→  const altFlow: ClaimStatus[] = ['SUBMITTED', 'UNDER_REVIEW', 'PARTIALLY_APPROVED', 'SETTLED']
   304→  const rejectFlow: ClaimStatus[] = ['SUBMITTED', 'UNDER_REVIEW', 'REJECTED']
   305→
   306→  if (current === 'PARTIALLY_APPROVED') return altFlow
   307→  if (current === 'REJECTED') return rejectFlow
   308→  return flow
   309→}

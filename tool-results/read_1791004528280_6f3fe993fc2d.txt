   142→// ─── Safe JSON fetch helper ───────────────────────────────────────
   143→
   144→async function safeFetch<T>(url: string, init?: RequestInit): Promise<T | null> {
   145→  try {
   146→    const res = await fetch(url, init)
   147→    if (!res.ok) return null
   148→    return (await res.json()) as T
   149→  } catch {
   150→    return null
   151→  }
   152→}

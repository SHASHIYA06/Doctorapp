   197→
   198→export function LabOrdersSection() {
   199→  const { selectedPatientId } = useAppStore()
   200→
   201→  const [labOrders, setLabOrders] = useState<LabOrder[]>([])
   202→  const [patients, setPatients] = useState<Patient[]>([])
   203→  const [loading, setLoading] = useState(true)
   204→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   205→  const [categoryTab, setCategoryTab] = useState<string>('all')
   206→
   207→  // Create dialog
   208→  const [createOpen, setCreateOpen] = useState(false)
   209→  const [formPatientId, setFormPatientId] = useState(selectedPatientId ?? '')
   210→  const [formPriority, setFormPriority] = useState<LabPriority>('ROUTINE')
   211→  const [formNotes, setFormNotes] = useState('')
   212→  const [selectedTests, setSelectedTests] = useState<Set<string>>(new Set())
   213→  const [submitting, setSubmitting] = useState(false)
   214→
   215→  // Detail dialog
   216→  const [detailOpen, setDetailOpen] = useState(false)
   217→  const [selectedOrder, setSelectedOrder] = useState<LabOrder | null>(null)
   218→
   219→  // ─── Fetch patients ───────────────────────────────────────────────
   220→
   221→  useEffect(() => {
   222→    fetch('/api/patients')
   223→      .then((r) => r.json())
   224→      .then((d) => setPatients(d.data ?? d.patients ?? []))
   225→      .catch(() => {
   226→        toast({ title: 'Error', description: 'Failed to load patients', variant: 'destructive' })
   227→      })
   228→  }, [])
   229→
   230→  // ─── Fetch lab orders ─────────────────────────────────────────────
   231→
   232→  const loadLabOrders = useCallback(async () => {
   233→    setLoading(true)
   234→    try {
   235→      const params = new URLSearchParams()
   236→      if (statusFilter && statusFilter !== 'ALL') params.set('status', statusFilter)
   237→      if (selectedPatientId) params.set('patientId', selectedPatientId)
   238→      const res = await fetch(`/api/lab-orders?${params.toString()}`)
   239→      const json = await res.json()
   240→      const raw = json.data ?? json.labOrders ?? []
   241→      const mapped = raw.map((o: Record<string, unknown>) => {
   242→        const patient = o.patient as { id: string; firstName: string; lastName: string } | undefined
   243→        const practitioner = o.practitioner as { id: string; name: string } | undefined
   244→        const tests = (Array.isArray(o.tests) ? o.tests : []) as LabTest[]
   245→        return {
   246→          id: o.id as string,
   247→          orderNumber: (o.orderNumber ?? '') as string,
   248→          patientId: (o.patientId ?? '') as string,
   249→          patient,
   250→          practitioner,
   251→          status: (o.status ?? 'ORDERED') as LabOrderStatus,
   252→          priority: (o.priority ?? 'ROUTINE') as LabPriority,
   253→          modality: (o.modality ?? 'ALLOPATHY') as string,
   254→          notes: (o.notes ?? null) as string | null,
   255→          orderedAt: (o.orderedAt ?? o.createdAt ?? new Date().toISOString()) as string,
   256→          sampleCollectedAt: (o.sampleCollectedAt ?? null) as string | null,
   257→          completedAt: (o.completedAt ?? null) as string | null,
   258→          createdAt: (o.createdAt ?? new Date().toISOString()) as string,
   259→          tests,
   260→        } as LabOrder
   261→      })
   262→      setLabOrders(mapped)
   263→    } catch {
   264→      toast({ title: 'Error', description: 'Failed to load lab orders', variant: 'destructive' })
   265→    } finally {
   266→      setLoading(false)
   267→    }
   268→  }, [statusFilter, selectedPatientId])
   269→
   270→  useEffect(() => { loadLabOrders() }, [loadLabOrders])
   271→
   272→  // ─── Test toggle ──────────────────────────────────────────────────
   273→
   274→  const toggleTest = (testName: string) => {
   275→    setSelectedTests((prev) => {
   276→      const next = new Set(prev)
   277→      if (next.has(testName)) next.delete(testName)
   278→      else next.add(testName)
   279→      return next
   280→    })
   281→  }
   282→
   283→  // ─── Create lab order ─────────────────────────────────────────────
   284→
   285→  const handleCreate = async () => {
   286→    if (!formPatientId) {
   287→      toast({ title: 'Validation', description: 'Select a patient', variant: 'destructive' })
   288→      return
   289→    }
   290→    if (selectedTests.size === 0) {
   291→      toast({ title: 'Validation', description: 'Select at least one test', variant: 'destructive' })
   292→      return
   293→    }
   294→    setSubmitting(true)
   295→    try {
   296→      const tests = Array.from(selectedTests).map((name) => {
   297→        const match = COMMON_TESTS.find((t) => t.name === name)
   298→        return {
   299→          testName: name,
   300→          testCode: match?.loinc ?? '',
   301→          category: match?.category ?? 'BIOCHEMISTRY',
   302→        }
   303→      })
   304→      const res = await fetch('/api/lab-orders', {
   305→        method: 'POST',
   306→        headers: { 'Content-Type': 'application/json' },
   307→        body: JSON.stringify({
   308→          patientId: formPatientId,
   309→          priority: formPriority,
   310→          notes: formNotes || undefined,
   311→          tests,
   312→        }),
   313→      })
   314→      if (res.ok) {
   315→        toast({ title: 'Lab Order Created', description: `Order with ${tests.length} test(s) placed` })
   316→        setCreateOpen(false)
   317→        resetForm()
   318→        loadLabOrders()
   319→      } else {
   320→        toast({ title: 'Error', description: 'Failed to create lab order', variant: 'destructive' })
   321→      }
   322→    } catch {
   323→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   324→    } finally {
   325→      setSubmitting(false)
   326→    }
   327→  }
   328→
   329→  const resetForm = () => {
   330→    setFormPatientId(selectedPatientId ?? '')
   331→    setFormPriority('ROUTINE')
   332→    setFormNotes('')
   333→    setSelectedTests(new Set())
   334→  }
   335→
   336→  // ─── Filtered orders by category tab ──────────────────────────────
   337→
   338→  const filteredOrders = labOrders.filter((order) => {
   339→    if (categoryTab === 'all') return true
   340→    return order.tests?.some((t) => t.category === categoryTab)
   341→  })
   342→
   343→  // ─── Status timeline step index ───────────────────────────────────
   344→
   345→  const getStepIndex = (status: LabOrderStatus): number => {
   346→    if (status === 'CANCELLED') return -1
   347→    return STATUS_STEPS.indexOf(status)
   348→  }
   349→
   350→  // ─── Render ───────────────────────────────────────────────────────
   351→
   352→  if (loading) {
   353→    return (
   354→      <div className="space-y-3">
   355→        {[1, 2, 3].map((i) => (
   356→          <Skeleton key={i} className="h-24 w-full" />
   357→        ))}
   358→      </div>
   359→    )
   360→  }
   361→
   362→  return (
   363→    <motion.div {...fadeSlide} className="space-y-6">
   364→      {/* Header */}
   365→      <div className="flex items-center justify-between flex-wrap gap-3">
   366→        <div className="flex items-center gap-3">
   367→          <div className="p-2 rounded-lg bg-teal-100 dark:bg-teal-900/30">
   368→            <FlaskConical className="h-6 w-6 text-teal-600" />
   369→          </div>
   370→          <div>
   371→            <h2 className="text-2xl font-bold tracking-tight">Lab & Diagnostics</h2>
   372→            <p className="text-sm text-muted-foreground">Order tests, track results, and flag abnormal values</p>
   373→          </div>
   374→        </div>
   375→        <Button
   376→          className="gap-2 bg-teal-600 hover:bg-teal-700 text-white"
   377→          onClick={() => {
   378→            resetForm()
   379→            setCreateOpen(true)
   380→          }}
   381→        >
   382→          <Plus className="h-4 w-4" /> New Lab Order
   383→        </Button>
   384→      </div>
   385→
   386→      {/* Status Filter + Category Tabs */}
   387→      <Tabs
   388→        value={categoryTab}
   389→        onValueChange={setCategoryTab}
   390→        className="space-y-4"
   391→      >
   392→        <div className="flex items-center gap-4 flex-wrap">
   393→          <TabsList>
   394→            <TabsTrigger value="all" className="gap-1">
   395→              All
   396→            </TabsTrigger>
   397→            <TabsTrigger value="HEMATOLOGY" className="gap-1">
   398→              <Beaker className="h-3 w-3" /> Hematology
   399→            </TabsTrigger>
   400→            <TabsTrigger value="BIOCHEMISTRY" className="gap-1">
   401→              <FlaskConical className="h-3 w-3" /> Biochemistry
   402→            </TabsTrigger>
   403→            <TabsTrigger value="MICROBIOLOGY" className="gap-1">
   404→              <TestTube className="h-3 w-3" /> Microbiology
   405→            </TabsTrigger>
   406→            <TabsTrigger value="RADIOLOGY" className="gap-1">
   407→              Radiology
   408→            </TabsTrigger>
   409→            <TabsTrigger value="PATHOLOGY" className="gap-1">
   410→              Pathology
   411→            </TabsTrigger>
   412→          </TabsList>
   413→
   414→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   415→            <SelectTrigger className="w-40">
   416→              <SelectValue placeholder="All Status" />
   417→            </SelectTrigger>
   418→            <SelectContent>
   419→              <SelectItem value="ALL">All Status</SelectItem>
   420→              <SelectItem value="ORDERED">Ordered</SelectItem>
   421→              <SelectItem value="SAMPLE_COLLECTED">Sample Collected</SelectItem>
   422→              <SelectItem value="PROCESSING">Processing</SelectItem>
   423→              <SelectItem value="COMPLETED">Completed</SelectItem>
   424→              <SelectItem value="CANCELLED">Cancelled</SelectItem>
   425→            </SelectContent>
   426→          </Select>
   427→        </div>
   428→
   429→        {/* All tab contents share the same table */}
   430→        {['all', 'HEMATOLOGY', 'BIOCHEMISTRY', 'MICROBIOLOGY', 'RADIOLOGY', 'PATHOLOGY'].map(
   431→          (tab) => (
   432→            <TabsContent key={tab} value={tab} className="space-y-4">
   433→              {/* Summary cards */}
   434→              <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   435→                {(
   436→                  ['ORDERED', 'SAMPLE_COLLECTED', 'PROCESSING', 'COMPLETED', 'CANCELLED'] as LabOrderStatus[]
   437→                ).map((st) => {
   438→                  const count = filteredOrders.filter((o) => o.status === st).length
   439→                  return (
   440→                    <Card key={st} className="p-3">
   441→                      <div className="flex items-center justify-between">
   442→                        <span className="text-xs font-medium text-muted-foreground">
   443→                          {STATUS_CONFIG[st].label}
   444→                        </span>
   445→                        <Badge className={`text-xs ${STATUS_CONFIG[st].className}`}>
   446→                          {count}
   447→                        </Badge>
   448→                      </div>
   449→                    </Card>
   450→                  )
   451→                })}
   452→              </div>
   453→
   454→              {/* Lab orders table */}
   455→              {filteredOrders.length === 0 ? (
   456→                <Card>
   457→                  <CardContent className="py-8 text-center text-muted-foreground">
   458→                    No lab orders found
   459→                  </CardContent>
   460→                </Card>
   461→              ) : (
   462→                <Card>
   463→                  <CardContent className="p-0">
   464→                    <div className="overflow-x-auto">
   465→                      <Table>
   466→                        <TableHeader>
   467→                          <TableRow>
   468→                            <TableHead>Order #</TableHead>
   469→                            <TableHead>Patient</TableHead>
   470→                            <TableHead>Priority</TableHead>
   471→                            <TableHead>Tests</TableHead>
   472→                            <TableHead>Status</TableHead>
   473→                            <TableHead>Ordered</TableHead>
   474→                            <TableHead className="text-right">Action</TableHead>
   475→                          </TableRow>
   476→                        </TableHeader>
   477→                        <TableBody>
   478→                          {filteredOrders.map((order) => {
   479→                            const statusCfg = STATUS_CONFIG[order.status] ?? STATUS_CONFIG.ORDERED
   480→                            const priCfg = PRIORITY_CONFIG[order.priority] ?? PRIORITY_CONFIG.ROUTINE
   481→                            const hasAbnormal = order.tests?.some((t) => t.isAbnormal)
   482→                            return (
   483→                              <TableRow
   484→                                key={order.id}
   485→                                className="cursor-pointer hover:bg-muted/50"
   486→                                onClick={() => {
   487→                                  setSelectedOrder(order)
   488→                                  setDetailOpen(true)
   489→                                }}
   490→                              >
   491→                                <TableCell className="font-mono text-sm font-medium">
   492→                                  {order.orderNumber}
   493→                                </TableCell>
   494→                                <TableCell className="text-sm">
   495→                                  {order.patient
   496→                                    ? `${order.patient.firstName} ${order.patient.lastName}`
   497→                                    : '—'}
   498→                                </TableCell>
   499→                                <TableCell>
   500→                                  <Badge className={`text-xs ${priCfg.className}`}>
   501→                                    {priCfg.label}
   502→                                  </Badge>
   503→                                </TableCell>
   504→                                <TableCell className="text-sm">
   505→                                  {order.tests?.length ?? 0}
   506→                                </TableCell>
   507→                                <TableCell>
   508→                                  <div className="flex items-center gap-1">
   509→                                    <Badge className={`text-xs ${statusCfg.className}`}>
   510→                                      {statusCfg.label}
   511→                                    </Badge>
   512→                                    {hasAbnormal && (
   513→                                      <AlertCircle className="h-3 w-3 text-red-500" />
   514→                                    )}
   515→                                  </div>
   516→                                </TableCell>
   517→                                <TableCell className="text-sm text-muted-foreground">
   518→                                  {new Date(order.orderedAt).toLocaleDateString()}
   519→                                </TableCell>
   520→                                <TableCell className="text-right">
   521→                                  <Button
   522→                                    variant="ghost"
   523→                                    size="sm"
   524→                                    onClick={(e) => {
   525→                                      e.stopPropagation()
   526→                                      setSelectedOrder(order)
   527→                                      setDetailOpen(true)
   528→                                    }}
   529→                                  >
   530→                                    <FlaskConical className="h-4 w-4" />
   531→                                  </Button>
   532→                                </TableCell>
   533→                              </TableRow>
   534→                            )
   535→                          })}
   536→                        </TableBody>
   537→                      </Table>
   538→                    </div>
   539→                  </CardContent>
   540→                </Card>
   541→              )}
   542→            </TabsContent>
   543→          )
   544→        )}
   545→      </Tabs>
   546→
   547→      {/* ─── Create Lab Order Dialog ─────────────────────────────── */}
   548→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   549→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   550→          <DialogHeader>
   551→            <DialogTitle className="flex items-center gap-2">
   552→              <FlaskConical className="h-5 w-5 text-teal-600" /> New Lab Order

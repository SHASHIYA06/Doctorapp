    61→
    62→interface MedicineSearchResult {
    63→  id: string
    64→  name: string
    65→  modality: string
    66→  genericName?: string
    67→}
    68→
    69→interface SelectedMedicine {
    70→  id: string
    71→  name: string
    72→}
    73→
    74→interface PairwiseInteraction {
    75→  medicine1: string
    76→  medicine2: string
    77→  interactionType: string
    78→  severity: 'LOW' | 'MODERATE' | 'HIGH' | 'CRITICAL'
    79→  effect: string
    80→  recommendation: string
    81→}
    82→
    83→interface PopulationSafety {
    84→  pregnancy: { safe: boolean; note: string }
    85→  lactation: { safe: boolean; note: string }
    86→  pediatric: { safe: boolean; note: string }
    87→  geriatric: { safe: boolean; note: string }
    88→  hepaticImpairment: { safe: boolean; note: string }
    89→  renalImpairment: { safe: boolean; note: string }
    90→}
    91→
    92→interface DrugInteractionResult {
    93→  safetyScore: number
    94→  riskLevel: 'LOW' | 'MODERATE' | 'HIGH' | 'CRITICAL'
    95→  pairwiseInteractions: PairwiseInteraction[]
    96→  contraindications: string[]
    97→  warnings: string[]
    98→  populationSafety: PopulationSafety
    99→}

   106→  label: string
   107→  className: string
   108→  icon: React.ElementType
   109→  iconColor: string
   110→}> = {
   111→  LOW: {
   112→    label: 'Low Risk',

   137→const SEVERITY_CELL_CONFIG: Record<string, string> = {
   138→  LOW: 'text-green-700 dark:text-green-300',
   139→  MODERATE: 'text-amber-700 dark:text-amber-300',
   140→  HIGH: 'text-orange-700 dark:text-orange-300',
   141→  CRITICAL: 'text-red-700 dark:text-red-300',
   142→}
   143→
   144→const SAFETY_SCORE_COLORS: Record<string, { bar: string; text: string; bg: string }> = {
   145→  green: { bar: 'bg-green-500', text: 'text-green-700 dark:text-green-300', bg: 'bg-green-50 dark:bg-green-950/30' },
   146→  amber: { bar: 'bg-amber-500', text: 'text-amber-700 dark:text-amber-300', bg: 'bg-amber-50 dark:bg-amber-950/30' },
   147→  orange: { bar: 'bg-orange-500', text: 'text-orange-700 dark:text-orange-300', bg: 'bg-orange-50 dark:bg-orange-950/30' },
   148→  red: { bar: 'bg-red-500', text: 'text-red-700 dark:text-red-300', bg: 'bg-red-50 dark:bg-red-950/30' },
   149→  critical: { bar: 'bg-red-700', text: 'text-red-900 dark:text-red-200', bg: 'bg-red-100 dark:bg-red-950/50' },
   150→}
   151→
   152→function getSafetyTier(score: number): keyof typeof SAFETY_SCORE_COLORS {
   153→  if (score >= 80) return 'green'
   154→  if (score >= 60) return 'amber'
   155→  if (score >= 40) return 'orange'
   156→  if (score >= 20) return 'red'
   157→  return 'critical'
   158→}
   159→
   160→const POPULATION_CONFIG: Record<string, { label: string; icon: React.ElementType }> = {
   161→  pregnancy: { label: 'Pregnancy', icon: Baby },
   162→  lactation: { label: 'Lactation', icon: Baby },
   163→  pediatric: { label: 'Pediatric', icon: Heart },
   164→  geriatric: { label: 'Geriatric', icon: Heart },
   165→  hepaticImpairment: { label: 'Hepatic Impairment', icon: Activity },
   166→  renalImpairment: { label: 'Renal Impairment', icon: Activity },
   167→}
   168→
   169→const fadeSlide = {
   170→  initial: { opacity: 0, y: 12 },
   171→  animate: { opacity: 1, y: 0 },
   172→  transition: { duration: 0.3 },
   173→}

   178→
   179→export function DrugInteractionsSection() {
   180→  // Medicine selection state
   181→  const [searchQuery, setSearchQuery] = useState('')
   182→  const [searchResults, setSearchResults] = useState<MedicineSearchResult[]>([])
   183→  const [selectedMedicines, setSelectedMedicines] = useState<SelectedMedicine[]>([])
   184→  const [searching, setSearching] = useState(false)
   185→  const [showDropdown, setShowDropdown] = useState(false)
   186→  const searchInputRef = useRef<HTMLInputElement>(null)
   187→  const dropdownRef = useRef<HTMLDivElement>(null)
   188→
   189→  // Results state
   190→  const [results, setResults] = useState<DrugInteractionResult | null>(null)
   191→  const [loading, setLoading] = useState(false)
   192→  const [showAllInteractions, setShowAllInteractions] = useState(false)
   193→
   194→  // ── Search medicines ───────────────────────
   195→
   196→  const searchMedicines = useCallback(async (query: string) => {
   197→    if (!query.trim() || query.length < 2) {
   198→      setSearchResults([])
   199→      setShowDropdown(false)
   200→      return
   201→    }
   202→
   203→    setSearching(true)
   204→    try {
   205→      const res = await fetch(`/api/medicines?search=${encodeURIComponent(query)}&limit=10`)
   206→      const data = await res.json()
   207→      const list: MedicineSearchResult[] = data.data ?? data.medicines ?? []
   208→      // Filter out already selected
   209→      const selectedIds = selectedMedicines.map((m) => m.id)
   210→      const filtered = list.filter((m) => !selectedIds.includes(m.id))
   211→      setSearchResults(filtered)
   212→      setShowDropdown(filtered.length > 0)
   213→    } catch {
   214→      setSearchResults([])
   215→    } finally {
   216→      setSearching(false)
   217→    }
   218→  }, [selectedMedicines])
   219→
   220→  // Debounced search
   221→  useEffect(() => {
   222→    const timer = setTimeout(() => {
   223→      searchMedicines(searchQuery)
   224→    }, 300)
   225→    return () => clearTimeout(timer)
   226→  }, [searchQuery, searchMedicines])
   227→
   228→  // Close dropdown on outside click
   229→  useEffect(() => {
   230→    const handler = (e: MouseEvent) => {
   231→      if (
   232→        dropdownRef.current &&
   233→        !dropdownRef.current.contains(e.target as Node) &&
   234→        searchInputRef.current &&
   235→        !searchInputRef.current.contains(e.target as Node)
   236→      ) {
   237→        setShowDropdown(false)
   238→      }
   239→    }
   240→    document.addEventListener('mousedown', handler)
   241→    return () => document.removeEventListener('mousedown', handler)
   242→  }, [])
   243→
   244→  // ── Medicine management ────────────────────
   245→
   246→  const addMedicine = (medicine: MedicineSearchResult) => {
   247→    if (!selectedMedicines.find((m) => m.id === medicine.id)) {
   248→      setSelectedMedicines((prev) => [...prev, { id: medicine.id, name: medicine.name }])
   249→    }
   250→    setSearchQuery('')
   251→    setSearchResults([])
   252→    setShowDropdown(false)
   253→    searchInputRef.current?.focus()
   254→  }
   255→
   256→  const removeMedicine = (id: string) => {
   257→    setSelectedMedicines((prev) => prev.filter((m) => m.id !== id))
   258→  }
   259→
   260→  // ── Check interactions ─────────────────────
   261→
   262→  const handleCheckInteractions = async () => {
   263→    if (selectedMedicines.length < 2) {
   264→      toast({
   265→        title: 'Add more medicines',
   266→        description: 'Please add at least 2 medicines to check interactions',
   267→        variant: 'destructive',
   268→      })
   269→      return
   270→    }
   271→
   272→    setLoading(true)
   273→    setResults(null)
   274→    setShowAllInteractions(false)
   275→
   276→    try {
   277→      const res = await fetch('/api/drug-interactions', {
   278→        method: 'POST',
   279→        headers: { 'Content-Type': 'application/json' },
   280→        body: JSON.stringify({
   281→          medicineIds: selectedMedicines.map((m) => m.id),
   282→          medicineNames: selectedMedicines.map((m) => m.name),
   283→        }),
   284→      })
   285→      const data = await res.json()
   286→      const result: DrugInteractionResult = data.data ?? data
   287→      setResults(result)
   288→
   289→      const riskLabel = RISK_LEVEL_CONFIG[result.riskLevel]?.label ?? result.riskLevel
   290→      toast({
   291→        title: 'Interaction Check Complete',
   292→        description: `Safety score: ${result.safetyScore}/100 — ${riskLabel}`,
   293→      })
   294→    } catch {
   295→      toast({ title: 'Error', description: 'Failed to check drug interactions', variant: 'destructive' })
   296→    } finally {
   297→      setLoading(false)
   298→    }
   299→  }
   300→
   301→  // ── Helpers ────────────────────────────────
   302→
   303→  const safetyTier = results ? getSafetyTier(results.safetyScore) : null
   304→  const tierColors = safetyTier ? SAFETY_SCORE_COLORS[safetyTier] : null
   305→
   306→  const visibleInteractions = results?.pairwiseInteractions
   307→    ? showAllInteractions
   308→      ? results.pairwiseInteractions
   309→      : results.pairwiseInteractions.slice(0, 5)
   310→    : []
   311→
   312→  // ── Render ─────────────────────────────────
   313→
   314→  return (
   315→    <TooltipProvider>
   316→      <motion.div {...fadeSlide} className="space-y-6">
   317→        {/* ── Input Card ─────────────────────── */}
   318→        <Card>
   319→          <CardHeader>
   320→            <CardTitle className="text-lg flex items-center gap-2">
   321→              <Pill className="h-5 w-5" />
   322→              Drug Interaction Checker
   323→            </CardTitle>
   324→            <CardDescription>
   325→              Add medicines to check for potential drug-drug interactions and safety concerns
   326→            </CardDescription>
   327→          </CardHeader>
   328→          <CardContent className="space-y-5">
   329→            {/* Medicine Search Input */}
   330→            <div className="space-y-2">
   331→              <Label>Add Medicines</Label>
   332→              <div className="relative">
   333→                <div className="relative">
   334→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   335→                  <Input
   336→                    ref={searchInputRef}
   337→                    type="text"
   338→                    value={searchQuery}
   339→                    onChange={(e) => setSearchQuery(e.target.value)}
   340→                    onFocus={() => {
   341→                      if (searchResults.length > 0) setShowDropdown(true)
   342→                    }}
   343→                    placeholder="Search medicines by name..."
   344→                    className="pl-9"
   345→                  />
   346→                  {searching && (
   347→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 animate-spin text-muted-foreground" />
   348→                  )}
   349→                </div>
   350→
   351→                {/* Search Dropdown */}
   352→                <AnimatePresence>
   353→                  {showDropdown && searchResults.length > 0 && (
   354→                    <motion.div
   355→                      ref={dropdownRef}
   356→                      initial={{ opacity: 0, y: -4 }}
   357→                      animate={{ opacity: 1, y: 0 }}
   358→                      exit={{ opacity: 0, y: -4 }}
   359→                      className="absolute top-full left-0 right-0 mt-1 border rounded-lg bg-popover shadow-lg z-20 overflow-hidden"
   360→                    >
   361→                      <ScrollArea className="max-h-56">
   362→                        {searchResults.map((medicine) => (
   363→                          <button
   364→                            key={medicine.id}
   365→                            type="button"
   366→                            onClick={() => addMedicine(medicine)}
   367→                            className="w-full text-left px-3 py-2.5 text-sm hover:bg-accent transition-colors flex items-center gap-3"
   368→                          >
   369→                            <Plus className="h-4 w-4 text-muted-foreground shrink-0" />
   370→                            <div className="flex-1 min-w-0">
   371→                              <span className="font-medium">{medicine.name}</span>
   372→                              {medicine.genericName && (
   373→                                <span className="text-muted-foreground ml-1">({medicine.genericName})</span>
   374→                              )}
   375→                            </div>
   376→                            <Badge variant="outline" className="text-xs shrink-0">
   377→                              {medicine.modality}
   378→                            </Badge>
   379→                          </button>
   380→                        ))}
   381→                      </ScrollArea>
   382→                    </motion.div>
   383→                  )}
   384→                </AnimatePresence>
   385→              </div>
   386→            </div>
   387→
   388→            {/* Selected Medicine Tags */}
   389→            {selectedMedicines.length > 0 && (
   390→              <div className="space-y-2">
   391→                <Label className="text-muted-foreground">
   392→                  Selected Medicines ({selectedMedicines.length})
   393→                </Label>
   394→                <div className="flex flex-wrap gap-2">
   395→                  <AnimatePresence>
   396→                    {selectedMedicines.map((med) => (
   397→                      <motion.span
   398→                        key={med.id}
   399→                        initial={{ opacity: 0, scale: 0.8 }}
   400→                        animate={{ opacity: 1, scale: 1 }}
   401→                        exit={{ opacity: 0, scale: 0.8 }}
   402→                        className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   403→                      >
   404→                        <Pill className="h-3.5 w-3.5" />
   405→                        {med.name}
   406→                        <button
   407→                          type="button"
   408→                          onClick={() => removeMedicine(med.id)}
   409→                          className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   410→                        >
   411→                          <X className="h-3 w-3" />
   412→                        </button>
   413→                      </motion.span>
   414→                    ))}
   415→                  </AnimatePresence>
   416→                </div>
   417→              </div>
   418→            )}
   419→
   420→            {/* Action Buttons */}
   421→            <div className="flex items-center gap-3">
   422→              <Button
   423→                onClick={handleCheckInteractions}
   424→                disabled={selectedMedicines.length < 2 || loading}
   425→                className="gap-2 min-w-[200px]"
   426→              >
   427→                {loading ? (
   428→                  <>
   429→                    <Loader2 className="h-4 w-4 animate-spin" />
   430→                    Checking...
   431→                  </>
   432→                ) : (
   433→                  <>
   434→                    <ShieldAlert className="h-4 w-4" />
   435→                    Check Interactions
   436→                  </>
   437→                )}
   438→              </Button>
   439→              {selectedMedicines.length > 0 && (
   440→                <Button
   441→                  variant="ghost"
   442→                  size="sm"
   443→                  onClick={() => {
   444→                    setSelectedMedicines([])
   445→                    setSearchQuery('')
   446→                    setResults(null)
   447→                  }}
   448→                  className="text-muted-foreground"
   449→                >
   450→                  Clear all
   451→                </Button>
   452→              )}
   453→            </div>
   454→
   455→            {selectedMedicines.length > 0 && selectedMedicines.length < 2 && (
   456→              <p className="text-xs text-muted-foreground flex items-center gap-1">
   457→                <Info className="h-3 w-3" />
   458→                Add at least 2 medicines to check for interactions
   459→              </p>
   460→            )}
   461→          </CardContent>
   462→        </Card>
   463→
   464→        {/* ── Loading Skeleton ────────────────── */}
   465→        <AnimatePresence>
   466→          {loading && (
   467→            <motion.div
   468→              initial={{ opacity: 0 }}
   469→              animate={{ opacity: 1 }}
   470→              exit={{ opacity: 0 }}
   471→              className="space-y-4"
   472→            >
   473→              <Card>
   474→                <CardContent className="p-6 space-y-4">
   475→                  <Skeleton className="h-8 w-64" />
   476→                  <Skeleton className="h-4 w-full" />
   477→                  <Skeleton className="h-32 w-full" />
   478→                  <Skeleton className="h-24 w-full" />
   479→                </CardContent>
   480→              </Card>
   481→            </motion.div>
   482→          )}
   483→        </AnimatePresence>
   484→
   485→        {/* ── Results ─────────────────────────── */}
   486→        <AnimatePresence>
   487→          {results && !loading && (
   488→            <motion.div
   489→              initial={{ opacity: 0, y: 16 }}
   490→              animate={{ opacity: 1, y: 0 }}
   491→              transition={{ duration: 0.4 }}
   492→              className="space-y-4"
   493→            >
   494→              {/* ── Safety Score & Risk Meter ────── */}
   495→              <Card className={tierColors ? `border-2 ${tierColors.bg}` : ''}>
   496→                <CardContent className="p-6">
   497→                  <div className="flex flex-col sm:flex-row items-start sm:items-center gap-6">
   498→                    {/* Score Circle */}
   499→                    <div className="text-center shrink-0">
   500→                      <div className={`text-5xl font-bold tracking-tight ${tierColors?.text ?? ''}`}>
   501→                        {results.safetyScore}
   502→                      </div>
   503→                      <div className="text-sm text-muted-foreground mt-1">of 100</div>
   504→                    </div>
   505→
   506→                    {/* Risk Meter & Level */}
   507→                    <div className="flex-1 min-w-0 space-y-3 w-full">
   508→                      <div className="flex items-center gap-3 flex-wrap">
   509→                        <span className="text-sm font-semibold">Safety Score</span>
   510→                        {(() => {
   511→                          const riskCfg = RISK_LEVEL_CONFIG[results.riskLevel]
   512→                          const RiskIcon = riskCfg?.icon ?? ShieldCheck
   513→                          return (
   514→                            <Badge variant="outline" className={`gap-1.5 ${riskCfg?.className ?? ''}`}>
   515→                              <RiskIcon className={`h-3.5 w-3.5 ${riskCfg?.iconColor ?? ''}`} />
   516→                              {riskCfg?.label ?? results.riskLevel}
   517→                            </Badge>
   518→                          )
   519→                        })()}
   520→                      </div>
   521→
   522→                      {/* Visual Risk Meter */}
   523→                      <div className="relative w-full">
   524→                        <div className="flex h-4 w-full overflow-hidden rounded-full bg-muted">
   525→                          {/* Background gradient segments */}
   526→                          <div className="flex-1 bg-red-700/30" />
   527→                          <div className="flex-1 bg-red-500/30" />
   528→                          <div className="flex-1 bg-orange-500/30" />
   529→                          <div className="flex-1 bg-amber-500/30" />
   530→                          <div className="flex-1 bg-green-500/30" />
   531→                        </div>
   532→                        {/* Score indicator */}
   533→                        <div
   534→                          className="absolute top-0 left-0 h-4 rounded-full transition-all duration-700"
   535→                          style={{ width: `${results.safetyScore}%` }}
   536→                        >
   537→                          <div className={`h-full w-full rounded-full ${tierColors?.bar ?? 'bg-green-500'}`} />
   538→                        </div>
   539→                      </div>
   540→
   541→                      <div className="flex justify-between text-xs text-muted-foreground">
   542→                        <span>Critical</span>
   543→                        <span>High</span>
   544→                        <span>Moderate</span>
   545→                        <span>Low</span>
   546→                        <span>Safe</span>
   547→                      </div>
   548→                    </div>
   549→                  </div>
   550→                </CardContent>
   551→              </Card>
   552→
   553→              {/* ── Pairwise Interactions Table ──── */}
   554→              {results.pairwiseInteractions && results.pairwiseInteractions.length > 0 && (
   555→                <Card>
   556→                  <CardHeader>
   557→                    <CardTitle className="text-base flex items-center gap-2">
   558→                      <Activity className="h-5 w-5" />
   559→                      Pairwise Interactions
   560→                    </CardTitle>
   561→                    <CardDescription>
   562→                      {results.pairwiseInteractions.length} interaction{results.pairwiseInteractions.length !== 1 ? 's' : ''} detected
   563→                    </CardDescription>
   564→                  </CardHeader>
   565→                  <CardContent>
   566→                    <div className="overflow-x-auto -mx-6 px-6">
   567→                      <Table>
   568→                        <TableHeader>
   569→                          <TableRow>

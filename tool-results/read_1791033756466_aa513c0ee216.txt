   177→// ─── Simple QR Code Data URL Generator ───────────────────────────────────
   178→// Generates a visual placeholder SVG that encodes the prescription data
   179→
   180→function generateQrDataUrl(data: string): string {
   181→  const size = 128
   182→  const cellSize = 4
   183→  const gridCount = size / cellSize
   184→  // Simple hash-based pattern for visual representation
   185→  let hash = 0
   186→  for (let i = 0; i < data.length; i++) {
   187→    hash = ((hash << 5) - hash + data.charCodeAt(i)) | 0
   188→  }
   189→  let svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">`
   190→  svg += `<rect width="${size}" height="${size}" fill="white"/>`
   191→  // Finder patterns (3 corners)
   192→  for (const [ox, oy] of [[0, 0], [size - 7 * cellSize, 0], [0, size - 7 * cellSize]]) {
   193→    svg += `<rect x="${ox}" y="${oy}" width="${7 * cellSize}" height="${7 * cellSize}" fill="black"/>`
   194→    svg += `<rect x="${ox + cellSize}" y="${oy + cellSize}" width="${5 * cellSize}" height="${5 * cellSize}" fill="white"/>`
   195→    svg += `<rect x="${ox + 2 * cellSize}" y="${oy + 2 * cellSize}" width="${3 * cellSize}" height="${3 * cellSize}" fill="black"/>`
   196→  }
   197→  // Data pattern
   198→  for (let row = 0; row < gridCount; row++) {
   199→    for (let col = 0; col < gridCount; col++) {
   200→      if (row < 8 && col < 8) continue
   201→      if (row < 8 && col >= gridCount - 8) continue
   202→      if (row >= gridCount - 8 && col < 8) continue
   203→      const bit = ((hash * (row * gridCount + col + 1)) >>> 0) % 2
   204→      if (bit) {
   205→        svg += `<rect x="${col * cellSize}" y="${row * cellSize}" width="${cellSize}" height="${cellSize}" fill="black"/>`
   206→      }
   207→    }
   208→  }
   209→  svg += '</svg>'
   210→  return `data:image/svg+xml;base64,${btoa(svg)}`
   211→}

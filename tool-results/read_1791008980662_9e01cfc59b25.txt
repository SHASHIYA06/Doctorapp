   143→// ─── Safe JSON fetch helper ───────────────────────────────────────
   144→
   145→async function safeFetch<T>(url: string, init?: RequestInit): Promise<T | null> {
   146→  try {
   147→    const res = await fetch(url, init)
   148→    if (!res.ok) return null
   149→    return (await res.json()) as T
   150→  } catch {
   151→    return null
   152→  }
   153→}

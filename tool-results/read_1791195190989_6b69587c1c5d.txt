   107→function formatDate(d: Date): string {
   108→  return d.toISOString().slice(0, 10)
   109→}
   110→
   111→function getMonthDays(year: number, month: number) {
   112→  const firstDay = new Date(year, month, 1).getDay()
   113→  const daysInMonth = new Date(year, month + 1, 0).getDate()
   114→  const daysInPrevMonth = new Date(year, month, 0).getDate()
   115→  const cells: { date: Date; isCurrentMonth: boolean }[] = []
   116→
   117→  for (let i = firstDay - 1; i >= 0; i--) {
   118→    cells.push({
   119→      date: new Date(year, month - 1, daysInPrevMonth - i),
   120→      isCurrentMonth: false,
   121→    })
   122→  }
   123→  for (let d = 1; d <= daysInMonth; d++) {
   124→    cells.push({ date: new Date(year, month, d), isCurrentMonth: true })
   125→  }
   126→  const remaining = 42 - cells.length
   127→  for (let d = 1; d <= remaining; d++) {
   128→    cells.push({ date: new Date(year, month + 1, d), isCurrentMonth: false })
   129→  }
   130→  return cells
   131→}
   132→
   133→function getWeekDays(baseDate: Date): Date[] {
   134→  const day = baseDate.getDay()
   135→  const start = new Date(baseDate)
   136→  start.setDate(start.getDate() - day)
   137→  return Array.from({ length: 7 }, (_, i) => {
   138→    const d = new Date(start)
   139→    d.setDate(start.getDate() + i)
   140→    return d
   141→  })
   142→}
   143→
   144→function addMinutes(time: string, mins: number): string {
   145→  const [h, m] = time.split(':').map(Number)
   146→  const total = h * 60 + m + mins
   147→  const nh = Math.floor(total / 60)
   148→  const nm = total % 60
   149→  return `${String(nh).padStart(2, '0')}:${String(nm).padStart(2, '0')}`
   150→}
   151→
   152→function isToday(dateStr: string): boolean {
   153→  return dateStr === formatDate(new Date())
   154→}
   155→
   156→function isThisWeek(dateStr: string): boolean {
   157→  const d = new Date(dateStr)
   158→  const now = new Date()
   159→  const startOfWeek = new Date(now)
   160→  startOfWeek.setDate(now.getDate() - now.getDay())
   161→  startOfWeek.setHours(0, 0, 0, 0)
   162→  const endOfWeek = new Date(startOfWeek)
   163→  endOfWeek.setDate(startOfWeek.getDate() + 7)
   164→  return d >= startOfWeek && d < endOfWeek
   165→}

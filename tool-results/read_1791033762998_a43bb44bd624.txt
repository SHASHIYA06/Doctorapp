    98→// ── Helpers ────────────────────────────────────────────────────
    99→
   100→const getAvailabilityStatus = (item: InventoryItem): AvailabilityStatus => {
   101→  if (item.currentStock === 0) return 'OUT_OF_STOCK'
   102→  if (item.currentStock < item.threshold) return 'LOW_STOCK'
   103→  return 'IN_STOCK'
   104→}
   105→
   106→const isExpiringSoon = (expiryDate: string, days: number = 30): boolean => {
   107→  const expiry = new Date(expiryDate)
   108→  const now = new Date()
   109→  const diff = (expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
   110→  return diff > 0 && diff <= days
   111→}
   112→
   113→const isExpired = (expiryDate: string): boolean => {
   114→  return new Date(expiryDate) < new Date()
   115→}
   116→
   117→const formatDate = (dateStr: string) => {
   118→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   119→}
   120→
   121→const statusBadge = (status: AvailabilityStatus) => {
   122→  switch (status) {
   123→    case 'IN_STOCK':
   124→      return <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100">In Stock</Badge>
   125→    case 'LOW_STOCK':
   126→      return <Badge className="bg-red-100 text-red-800 hover:bg-red-100">Low Stock</Badge>
   127→    case 'OUT_OF_STOCK':
   128→      return <Badge className="bg-gray-100 text-gray-800 hover:bg-gray-100">Out of Stock</Badge>
   129→  }
   130→}

   240→    }

   246→  }

   266→    setUpdateDialogOpen(false)

   269→  }
   270→
   271→  if (loading) {

   306→      {/* Stats Cards */}

   383→              </CardHeader>

   385→                <div className="space-y-1 max-h-32 overflow-y-auto">

   392→                </div>

   479→                        </TableCell>

   481→                          <span className="font-mono text-xs">{item.batchNumber}</span>
   482→                        </TableCell>
   483→                        <TableCell>
   484→                          <span className={status === 'LOW_STOCK' ? 'text-red-600 font-semibold' : status === 'OUT_OF_STOCK' ? 'text-gray-500' : ''}>
   485→                            {item.currentStock} {item.unit}{item.currentStock !== 1 ? 's' : ''}
   486→                          </span>
   487→                        </TableCell>
   488→                        <TableCell>
   489→                          <span className="flex items-center gap-0.5">
   490→                            <IndianRupee className="h-3 w-3" />{item.mrp}
   491→                          </span>
   492→                        </TableCell>
   493→                        <TableCell>
   494→                          <div className="flex items-center gap-1">
   495→                            <span className={expired ? 'text-red-600 font-semibold' : expiring ? 'text-amber-600 font-semibold' : ''}>
   496→                              {formatDate(item.expiryDate)}
   497→                            </span>
   498→                            {expired && <Badge variant="destructive" className="text-[10px] px-1 py-0">EXPIRED</Badge>}
   499→                            {!expired && expiring && <Badge className="bg-amber-100 text-amber-800 text-[10px] px-1 py-0 hover:bg-amber-100">EXPIRING</Badge>}
   500→                          </div>
   501→                        </TableCell>
   502→                        <TableCell>{statusBadge(status)}</TableCell>
   503→                        <TableCell className="text-right">
   504→                          <Dialog open={updateDialogOpen && selectedItem?.id === item.id} onOpenChange={(open) => {
   505→                            if (open) {
   506→                              setSelectedItem(item)
   507→                              setUpdateForm({ itemId: item.id, action: 'add', quantity: 0, reason: '' })
   508→                            } else {
   509→                              setUpdateDialogOpen(false)
   510→                              setSelectedItem(null)
   511→                            }
   512→                          }}>
   513→                            <DialogTrigger asChild>
   514→                              <Button variant="ghost" size="sm" onClick={() => {
   515→                                setSelectedItem(item)
   516→                                setUpdateForm({ itemId: item.id, action: 'add', quantity: 0, reason: '' })
   517→                                setUpdateDialogOpen(true)
   518→                              }}>
   519→                                <Plus className="h-4 w-4" />
   520→                              </Button>
   521→                            </DialogTrigger>
   522→                            <DialogContent>
   523→                              <DialogHeader>
   524→                                <DialogTitle>Update Stock — {item.medicineName}</DialogTitle>
   525→                                <DialogDescription>
   526→                                  Current stock: {item.currentStock} {item.unit}s | Batch: {item.batchNumber}
   527→                                </DialogDescription>
   528→                              </DialogHeader>
   529→                              <div className="space-y-4 py-2">
   530→                                <Select value={updateForm.action} onValueChange={(v) => setUpdateForm(prev => ({ ...prev, action: v as 'add' | 'reduce' }))}>
   531→                                  <SelectTrigger>
   532→                                    <SelectValue placeholder="Action" />
   533→                                  </SelectTrigger>
   534→                                  <SelectContent>
   535→                                    <SelectItem value="add">Add Stock</SelectItem>
   536→                                    <SelectItem value="reduce">Reduce Stock</SelectItem>
   537→                                  </SelectContent>
   538→                                </Select>
   539→                                <Input
   540→                                  type="number"
   541→                                  min={1}
   542→                                  placeholder="Quantity"
   543→                                  value={updateForm.quantity || ''}
   544→                                  onChange={e => setUpdateForm(prev => ({ ...prev, quantity: parseInt(e.target.value) || 0 }))}
   545→                                />
   546→                                <Input
   547→                                  placeholder="Reason (e.g., New shipment, Damaged, Returned)"
   548→                                  value={updateForm.reason}
   549→                                  onChange={e => setUpdateForm(prev => ({ ...prev, reason: e.target.value }))}
   550→                                />
   551→                              </div>
   552→                              <DialogFooter>
   553→                                <Button variant="outline" onClick={() => setUpdateDialogOpen(false)}>Cancel</Button>
   554→                                <Button
   555→                                  onClick={handleStockUpdate}
   556→                                  disabled={updateForm.quantity <= 0}
   557→                                  className={updateForm.action === 'add' ? 'bg-emerald-600 hover:bg-emerald-700' : 'bg-red-600 hover:bg-red-700'}
   558→                                >
   559→                                  {updateForm.action === 'add' ? <Plus className="h-4 w-4 mr-1" /> : <Minus className="h-4 w-4 mr-1" />}
   560→                                  {updateForm.action === 'add' ? 'Add' : 'Reduce'} {updateForm.quantity || 0} Units
   561→                                </Button>
   562→                              </DialogFooter>
   563→                            </DialogContent>
   564→                          </Dialog>
   565→                        </TableCell>
   566→                      </TableRow>
   567→                    )
   568→                  })
   569→                )}
   570→              </TableBody>
   571→            </Table>
   572→          </CardContent>
   573→        </Card>
   574→      </motion.div>
   575→
   576→      {/* Out of Stock Summary */}
   577→      {outOfStockItems.length > 0 && (
   578→        <motion.div {...fadeSlide}>
   579→          <Card className="border-red-200">
   580→            <CardHeader>
   581→              <CardTitle className="text-lg flex items-center gap-2 text-red-700">
   582→                <XCircle className="h-5 w-5" /> Out of Stock Items
   583→              </CardTitle>
   584→              <CardDescription>These items need immediate restocking</CardDescription>
   585→            </CardHeader>
   586→            <CardContent>
   587→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   588→                {outOfStockItems.map(item => (
   589→                  <div key={item.id} className="flex items-center justify-between p-3 rounded-lg border border-red-100 bg-red-50/30">
   590→                    <div>
   591→                      <p className="font-medium text-sm">{item.medicineName}</p>
   592→                      <p className="text-xs text-muted-foreground">{item.genericName} • {item.manufacturer}</p>
   593→                    </div>
   594→                    <Badge variant="destructive" className="text-[10px]">OUT</Badge>
   595→                  </div>
   596→                ))}
   597→              </div>
   598→            </CardContent>
   599→          </Card>
   600→        </motion.div>
   601→      )}
   602→    </div>
   603→  )

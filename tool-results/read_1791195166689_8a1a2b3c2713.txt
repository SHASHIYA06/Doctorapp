   240→// ── Helpers ──────────────────────────────────────────────────────────────
   241→
   242→function formatDate(iso: string): string {
   243→  return new Date(iso).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
   244→}
   245→
   246→function formatTime(iso: string): string {
   247→  return new Date(iso).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true })
   248→}
   249→
   250→function formatRelative(iso: string): string {
   251→  const diff = Date.now() - new Date(iso).getTime()
   252→  const days = Math.floor(diff / 86400000)
   253→  if (days === 0) return 'Today'
   254→  if (days === 1) return 'Yesterday'
   255→  if (days < 30) return `${days}d ago`
   256→  if (days < 365) return `${Math.floor(days / 30)}mo ago`
   257→  return `${Math.floor(days / 365)}y ago`
   258→}

   288→    try {

   290→      const json = await res.json()
   291→      const list = json.data ?? json.patients ?? []
   292→      setPatients(list)
   293→    } catch {
   294→      toast({ title: 'Error', description: 'Failed to load patients', variant: 'destructive' })

   296→      setPatientsLoading(false)

   308→        const json = await res.json()

   310→        setEvents(Array.isArray(list) && list.length > 0 ? list : SAMPLE_EVENTS)
   311→      } else {
   312→        setEvents(SAMPLE_EVENTS)
   313→      }
   314→    } catch {

   324→  const filteredEvents = useMemo(() => {
   325→    let result = [...events]
   326→
   327→    if (filterEventType !== 'ALL') {
   328→      result = result.filter((e) => e.eventType === filterEventType)
   329→    }
   330→    if (filterModality !== 'ALL') {
   331→      result = result.filter((e) => e.modality === filterModality)
   332→    }
   333→    if (filterDateFrom) {
   334→      result = result.filter((e) => new Date(e.timestamp) >= new Date(filterDateFrom))
   335→    }
   336→    if (filterDateTo) {
   337→      const to = new Date(filterDateTo)
   338→      to.setHours(23, 59, 59, 999)
   339→      result = result.filter((e) => new Date(e.timestamp) <= to)
   340→    }
   341→
   342→    result.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime())
   343→    return result
   344→  }, [events, filterEventType, filterModality, filterDateFrom, filterDateTo])
   345→
   346→  // ── Stats ────────────────────────────────────────────────────────────

   351→      .filter((e) => e.eventType === 'FOLLOW_UP' && new Date(e.timestamp) >= new Date())
   352→      .sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime())[0]
   353→
   354→    return {
   355→      totalEvents: events.length,
   356→      lastVisit: lastEncounter ? formatDate(lastEncounter.timestamp) : 'N/A',
   357→      nextFollowUp: nextFollowUp ? formatDate(nextFollowUp.timestamp) : 'None scheduled',
   358→    }
   359→  }, [events])
   360→
   361→  // ── Render ───────────────────────────────────────────────────────────

   369→          <p className="text-sm text-muted-foreground">Visual event history across all modalities</p>
   370→        </div>
   371→        <div className="flex items-center gap-2">
   372→          <Select
   373→            value={localPatientId}

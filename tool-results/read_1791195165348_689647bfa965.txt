    33→  spo2?: string

    39→  subjective: string
    40→  objective: string
    41→  assessment: string
    42→  plan: string

    44→

    46→  id: string
    47→  patientId: string
    48→  patientName: string
    49→  type: NoteType
    50→  modality: Modality
    51→  soap: SOAPData
    52→  vitals: VitalSigns
    53→  signed: boolean
    54→  signedBy: string | null
    55→  signedAt: string | null
    56→  practitionerName: string
    57→  summary: string | null
    58→  createdAt: string
    59→  updatedAt: string
    60→}
    61→
    62→interface Patient {
    63→  id: string
    64→  firstName: string
    65→  lastName: string
    66→}

    93→]

    97→  'Gastroesophageal Reflux', 'Anxiety Disorder', 'Osteoarthritis', 'Asthma',
    98→  'Urinary Tract Infection', 'Iron Deficiency Anemia',
    99→]

   102→  initial: { opacity: 0, y: 12 },

   107→// ── Component ────────────────────────────────────────────────────────────
   108→
   109→export function ClinicalNotesSection() {
   110→  const { activeModality, selectedPatientId } = useAppStore()
   111→
   112→  // Data
   113→  const [notes, setNotes] = useState<ClinicalNote[]>([])
   114→  const [patients, setPatients] = useState<Patient[]>([])
   115→  const [loading, setLoading] = useState(true)
   116→
   117→  // Modality filter
   118→  const [modalityFilter, setModalityFilter] = useState<Modality>(activeModality)
   119→
   120→  // Search & filter
   121→  const [searchQuery, setSearchQuery] = useState('')
   122→  const [typeFilter, setTypeFilter] = useState<NoteType | 'ALL'>('ALL')
   123→
   124→  // New note form
   125→  const [isCreating, setIsCreating] = useState(false)
   126→  const [formPatientId, setFormPatientId] = useState(selectedPatientId ?? '')
   127→  const [formType, setFormType] = useState<NoteType>('SOAP')
   128→  const [formModality, setFormModality] = useState<Modality>(activeModality)
   129→  const [formSOAP, setFormSOAP] = useState<SOAPData>({ subjective: '', objective: '', assessment: '', plan: '' })
   130→  const [formVitals, setFormVitals] = useState<VitalSigns>({})
   131→  const [formPractitioner, setFormPractitioner] = useState('Dr. Sharma')
   132→  const [formSubmitting, setFormSubmitting] = useState(false)
   133→
   134→  // AI summary
   135→  const [aiGenerating, setAiGenerating] = useState(false)
   136→
   137→  // Detail view
   138→  const [selectedNote, setSelectedNote] = useState<ClinicalNote | null>(null)
   139→
   140→  // Sign dialog
   141→  const [signDialogNote, setSignDialogNote] = useState<ClinicalNote | null>(null)
   142→  const [signSubmitting, setSignSubmitting] = useState(false)
   143→
   144→  // Diagnosis suggestions
   145→  const [showDiagnosisSuggestions, setShowDiagnosisSuggestions] = useState(false)
   146→
   147→  // ── Load Data ──────────────────────────────────────────────────────────
   148→
   149→  const loadData = useCallback(async () => {
   150→    try {
   151→      const [notesRes, patRes] = await Promise.all([
   152→        fetch('/api/clinical-notes').then((r) => r.json()),
   153→        fetch('/api/patients').then((r) => r.json()),
   154→      ])
   155→      const rawNotes = notesRes.data ?? notesRes.notes ?? []
   156→      const noteList = rawNotes.map((n: Record<string, unknown>) => {
   157→        const patient = n.patient as Record<string, string> | undefined
   158→        const practitioner = n.practitioner as Record<string, string> | undefined
   159→        return {
   160→          id: n.id as string,
   161→          patientId: (n.patientId ?? '') as string,
   162→          patientName: (patient ? `${patient.firstName} ${patient.lastName}` : (n.patientName ?? 'Unknown')) as string,
   163→          type: ((n.noteType ?? n.type ?? 'SOAP') as string) as NoteType,
   164→          modality: (n.modality ?? 'ALLOPATHY') as Modality,
   165→          soap: {
   166→            subjective: ((n.subjective ?? (n.soap as Record<string, string>)?.subjective ?? '') as string),
   167→            objective: ((n.objective ?? (n.soap as Record<string, string>)?.objective ?? '') as string),
   168→            assessment: ((n.assessment ?? (n.soap as Record<string, string>)?.assessment ?? '') as string),
   169→            plan: ((n.plan ?? (n.soap as Record<string, string>)?.plan ?? '') as string),
   170→          } as SOAPData,
   171→          vitals: (n.vitals ?? {}) as VitalSigns,
   172→          signed: (n.isSigned ?? n.signed ?? false) as boolean,
   173→          signedBy: (n.signedBy ?? (practitioner ? practitioner.name : null)) as string | null,
   174→          signedAt: (n.signedAt ?? null) as string | null,
   175→          practitionerName: (practitioner ? practitioner.name : (n.practitionerName ?? 'Unknown')) as string,
   176→          summary: (n.summary ?? null) as string | null,
   177→          createdAt: (n.createdAt ?? new Date().toISOString()) as string,
   178→          updatedAt: (n.updatedAt ?? new Date().toISOString()) as string,
   179→        }
   180→      })
   181→      setNotes(noteList)
   182→      setPatients(patRes.data ?? patRes.patients ?? [])
   183→    } catch {
   184→      toast({ title: 'Error', description: 'Failed to load clinical notes', variant: 'destructive' })
   185→    } finally {
   186→      setLoading(false)
   187→    }
   188→  }, [])
   189→
   190→  useEffect(() => { loadData() }, [loadData])
   191→
   192→  // ── Filtered Notes ─────────────────────────────────────────────────────
   193→
   194→  const filteredNotes = useMemo(() => {
   195→    let list = notes.filter((n) => n.modality === modalityFilter)
   196→
   197→    if (typeFilter !== 'ALL') {
   198→      list = list.filter((n) => n.type === typeFilter)
   199→    }
   200→
   201→    if (searchQuery.trim()) {
   202→      const q = searchQuery.toLowerCase()
   203→      list = list.filter(
   204→        (n) =>
   205→          n.patientName.toLowerCase().includes(q) ||
   206→          n.soap.subjective.toLowerCase().includes(q) ||
   207→          n.soap.assessment.toLowerCase().includes(q) ||
   208→          n.practitionerName.toLowerCase().includes(q)
   209→      )
   210→    }
   211→
   212→    return list.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   213→  }, [notes, modalityFilter, typeFilter, searchQuery])
   214→
   215→  // ── SOAP Update Helper ─────────────────────────────────────────────────
   216→
   217→  const updateSOAP = (field: keyof SOAPData, value: string) => {
   218→    setFormSOAP((prev) => ({ ...prev, [field]: value }))
   219→  }
   220→
   221→  const updateVital = (key: keyof VitalSigns, value: string) => {
   222→    setFormVitals((prev) => ({ ...prev, [key]: value }))
   223→  }
   224→
   225→  // ── AI Summary ─────────────────────────────────────────────────────────
   226→
   227→  const handleAISummary = async () => {
   228→    if (!formSOAP.subjective && !formSOAP.objective && !formSOAP.assessment) {
   229→      toast({ title: 'No Content', description: 'Add some clinical content first', variant: 'destructive' })
   230→      return
   231→    }
   232→    setAiGenerating(true)
   233→    try {
   234→      const res = await fetch('/api/ai-draft', {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify({
   238→          type: 'CLINICAL_SUMMARY',
   239→          context: {
   240→            subjective: formSOAP.subjective,
   241→            objective: formSOAP.objective,
   242→            assessment: formSOAP.assessment,
   243→            plan: formSOAP.plan,
   244→            vitals: formVitals,
   245→            modality: formModality,
   246→          },
   247→        }),
   248→      })
   249→      if (res.ok) {
   250→        const data = await res.json()
   251→        const summary = data.summary ?? data.content ?? ''
   252→        if (summary) {
   253→          // Append AI summary to the plan field
   254→          updateSOAP('plan', formSOAP.plan ? `${formSOAP.plan}\n\n[AI Summary]: ${summary}` : `[AI Summary]: ${summary}`)
   255→          toast({ title: 'AI Summary Generated', description: 'Added to Plan section' })
   256→        }
   257→      } else {
   258→        toast({ title: 'AI Error', description: 'Failed to generate summary', variant: 'destructive' })
   259→      }
   260→    } catch {
   261→      // Fallback: generate a local summary
   262→      const localSummary = `Patient presents with ${formSOAP.subjective.slice(0, 100) || 'reported symptoms'}. Assessment: ${formSOAP.assessment.slice(0, 100) || 'pending'}.`
   263→      updateSOAP('plan', formSOAP.plan ? `${formSOAP.plan}\n\n[AI Summary]: ${localSummary}` : `[AI Summary]: ${localSummary}`)
   264→      toast({ title: 'AI Summary (Local)', description: 'Generated locally as fallback' })
   265→    } finally {
   266→      setAiGenerating(false)
   267→    }
   268→  }
   269→
   270→  // ── Save Note ──────────────────────────────────────────────────────────
   271→
   272→  const handleSave = async () => {
   273→    if (!formPatientId) {
   274→      toast({ title: 'Validation', description: 'Select a patient', variant: 'destructive' })
   275→      return
   276→    }
   277→    setFormSubmitting(true)
   278→    try {
   279→      const patient = patients.find((p) => p.id === formPatientId)
   280→      const res = await fetch('/api/clinical-notes', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          patientId: formPatientId,
   285→          patientName: patient ? `${patient.firstName} ${patient.lastName}` : 'Unknown',
   286→          type: formType,
   287→          modality: formModality,
   288→          soap: formSOAP,
   289→          vitals: formVitals,
   290→          practitionerName: formPractitioner,
   291→        }),
   292→      })
   293→      if (res.ok) {
   294→        toast({ title: 'Note Saved', description: `${formType} note created` })
   295→        resetForm()
   296→        loadData()
   297→      } else {
   298→        const err = await res.json()
   299→        toast({ title: 'Error', description: err.error ?? 'Failed to save note', variant: 'destructive' })
   300→      }
   301→    } catch {
   302→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   303→    } finally {
   304→      setFormSubmitting(false)
   305→    }
   306→  }
   307→
   308→  // ── Sign Note ──────────────────────────────────────────────────────────
   309→
   310→  const handleSign = async () => {
   311→    if (!signDialogNote) return
   312→    setSignInSubmitting(true)
   313→    try {
   314→      const res = await fetch(`/api/clinical-notes/${signDialogNote.id}`, {
   315→        method: 'PATCH',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify({
   318→          signed: true,
   319→          signedBy: signDialogNote.practitionerName,
   320→          signedAt: new Date().toISOString(),
   321→        }),
   322→      })
   323→      if (res.ok) {
   324→        toast({ title: 'Note Signed', description: 'Note is now locked' })
   325→        setSignDialogNote(null)
   326→        setSelectedNote(null)
   327→        loadData()
   328→      } else {
   329→        toast({ title: 'Error', description: 'Failed to sign note', variant: 'destructive' })
   330→      }
   331→    } catch {
   332→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   333→    } finally {
   334→      setSignInSubmitting(false)
   335→    }
   336→  }
   337→
   338→  // ── Reset Form ─────────────────────────────────────────────────────────
   339→
   340→  const resetForm = () => {
   341→    setIsCreating(false)
   342→    setFormPatientId(selectedPatientId ?? '')
   343→    setFormType('SOAP')
   344→    setFormModality(activeModality)
   345→    setFormSOAP({ subjective: '', objective: '', assessment: '', plan: '' })
   346→    setFormVitals({})
   347→    setFormPractitioner('Dr. Sharma')
   348→    setShowDiagnosisSuggestions(false)
   349→  }
   350→
   351→  // ── Loading ────────────────────────────────────────────────────────────
   352→
   353→  if (loading) {
   354→    return (
   355→      <div className="space-y-4">
   356→        <Skeleton className="h-10 w-full max-w-sm" />
   357→        <div className="grid gap-4 lg:grid-cols-3">
   358→          <div className="lg:col-span-2 space-y-3">
   359→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-20 w-full rounded-lg" />)}
   360→          </div>
   361→          <Skeleton className="h-64 rounded-lg" />
   362→        </div>
   363→      </div>
   364→    )
   365→  }
   366→
   367→  // ── Render ─────────────────────────────────────────────────────────────
   368→
   369→  return (
   370→    <motion.div {...fadeSlide} className="space-y-4">
   371→      {/* Header */}
   372→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   373→        <div className="flex items-center gap-2">
   374→          <FileText className="h-5 w-5" />
   375→          <h2 className="text-lg font-semibold">Clinical Notes</h2>
   376→        </div>
   377→        <div className="flex items-center gap-2 flex-wrap">
   378→          <div className="relative">
   379→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   380→            <Input
   381→              placeholder="Search notes..."
   382→              value={searchQuery}
   383→              onChange={(e) => setSearchQuery(e.target.value)}
   384→              className="pl-8 h-8 w-48"
   385→            />
   386→          </div>
   387→          <Select value={typeFilter} onValueChange={(v) => setTypeFilter(v as NoteType | 'ALL')}>
   388→            <SelectTrigger className="h-8 w-32">
   389→              <SelectValue />
   390→            </SelectTrigger>
   391→            <SelectContent>
   392→              <SelectItem value="ALL">All Types</SelectItem>
   393→              {NOTE_TYPES.map((t) => (
   394→                <SelectItem key={t} value={t}>{t}</SelectItem>
   395→              ))}
   396→            </SelectContent>
   397→          </Select>
   398→          <Button size="sm" className="gap-1.5" onClick={() => { setIsCreating(true); resetForm(); setIsCreating(true) }}>
   399→            <Plus className="h-4 w-4" />
   400→            New Note
   401→          </Button>
   402→        </div>
   403→      </div>
   404→
   405→      {/* Modality Tabs (NEVER merge) */}
   406→      <div className="flex gap-2">
   407→        {(['ALLOPATHY', 'AYURVEDA', 'HOMEOPATHY'] as Modality[]).map((m) => (
   408→          <Button
   409→            key={m}
   410→            variant={modalityFilter === m ? 'default' : 'outline'}
   411→            size="sm"
   412→            onClick={() => setModalityFilter(m)}
   413→            className="gap-1.5"
   414→          >
   415→            <ModalityBadge modality={m} className="border-0 p-0" />
   416→          </Button>
   417→        ))}
   418→      </div>
   419→
   420→      {/* Main Layout */}
   421→      <div className="grid gap-4 lg:grid-cols-3">
   422→        {/* Left: Note List or Create Form */}
   423→        <div className={selectedNote ? 'lg:col-span-2' : 'lg:col-span-3'}>
   424→          {isCreating ? (
   425→            /* ── Create Form ─────────────────────────────────────────────── */
   426→            <Card>
   427→              <CardHeader>
   428→                <div className="flex items-center justify-between">
   429→                  <CardTitle className="text-base flex items-center gap-2">
   430→                    <PenLine className="h-4 w-4" />
   431→                    New Clinical Note
   432→                  </CardTitle>
   433→                  <Button variant="ghost" size="icon" className="h-7 w-7" onClick={resetForm}>
   434→                    <X className="h-4 w-4" />
   435→                  </Button>
   436→                </div>
   437→              </CardHeader>
   438→              <CardContent className="space-y-5">
   439→                {/* Patient, Type, Modality Row */}
   440→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   441→                  <div className="space-y-2">
   442→                    <Label>Patient *</Label>
   443→                    <Select value={formPatientId} onValueChange={setFormPatientId}>
   444→                      <SelectTrigger><SelectValue placeholder="Select..." /></SelectTrigger>
   445→                      <SelectContent>
   446→                        {patients.map((p) => (
   447→                          <SelectItem key={p.id} value={p.id}>{p.firstName} {p.lastName}</SelectItem>
   448→                        ))}
   449→                      </SelectContent>
   450→                    </Select>
   451→                  </div>
   452→                  <div className="space-y-2">
   453→                    <Label>Note Type</Label>
   454→                    <Select value={formType} onValueChange={(v) => setFormType(v as NoteType)}>
   455→                      <SelectTrigger><SelectValue /></SelectTrigger>
   456→                      <SelectContent>
   457→                        {NOTE_TYPES.map((t) => (
   458→                          <SelectItem key={t} value={t}>{t}</SelectItem>
   459→                        ))}
   460→                      </SelectContent>
   461→                    </Select>
   462→                  </div>
   463→                  <div className="space-y-2">
   464→                    <Label>Care Modality</Label>
   465→                    <div className="flex gap-2 flex-wrap h-9 items-center">
   466→                      {(['ALLOPATHY', 'AYURVEDA', 'HOMEOPATHY'] as Modality[]).map((m) => (
   467→                        <Button
   468→                          key={m}
   469→                          type="button"
   470→                          variant={formModality === m ? 'default' : 'outline'}
   471→                          size="sm"
   472→                          onClick={() => setFormModality(m)}
   473→                        >
   474→                          <ModalityBadge modality={m} className="border-0 p-0" />
   475→                        </Button>
   476→                      ))}
   477→                    </div>
   478→                  </div>
   479→                </div>
   480→
   481→                {/* Practitioner */}
   482→                <div className="space-y-2">
   483→                  <Label>Practitioner</Label>
   484→                  <Input
   485→                    value={formPractitioner}
   486→                    onChange={(e) => setFormPractitioner(e.target.value)}
   487→                    placeholder="Dr. Name"
   488→                    className="max-w-xs"
   489→                  />
   490→                </div>
   491→
   492→                <Separator />
   493→
   494→                {/* Vital Signs Quick-Add (in Objective section) */}
   495→                <div>
   496→                  <div className="flex items-center gap-2 mb-3">
   497→                    <Activity className="h-4 w-4 text-green-600" />
   498→                    <Label className="text-sm font-semibold">Vital Signs</Label>
   499→                  </div>
   500→                  <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-6 gap-3">
   501→                    {VITAL_FIELDS.map(({ key, label, unit, placeholder }) => (
   502→                      <div key={key} className="space-y-1">
   503→                        <Label className="text-xs text-muted-foreground">{label} ({unit})</Label>
   504→                        <Input
   505→                          placeholder={placeholder}
   506→                          value={formVitals[key] ?? ''}
   507→                          onChange={(e) => updateVital(key, e.target.value)}
   508→                          className="h-8 text-sm"
   509→                        />
   510→                      </div>
   511→                    ))}
   512→                  </div>
   513→                </div>
   514→
   515→                <Separator />
   516→
   517→                {/* SOAP Fields */}
   518→                {(['subjective', 'objective', 'assessment', 'plan'] as const).map((field) => {
   519→                  const config = SOAP_LABELS[field]

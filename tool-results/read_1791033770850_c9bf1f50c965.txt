     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import { motion } from 'framer-motion'
     5→import {
     6→  ArrowRightLeft,
     7→  UserPlus,
     8→  CheckCircle2,
     9→  XCircle,
    10→  AlertTriangle,
    11→  Clock,
    12→  Stethoscope,
    13→  Plus,
    14→  Filter,
    15→  Eye,
    16→} from 'lucide-react'
    17→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
    18→import { Button } from '@/components/ui/button'
    19→import { Badge } from '@/components/ui/badge'
    20→import { Input } from '@/components/ui/input'
    21→import { Label } from '@/components/ui/label'
    22→import { Textarea } from '@/components/ui/textarea'
    23→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    24→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog'
    25→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    26→import { Skeleton } from '@/components/ui/skeleton'
    27→import { Separator } from '@/components/ui/separator'
    28→import { useAppStore } from '@/lib/store'
    29→import { ModalityBadge } from '@/components/clinical/modality-badge'
    30→import { toast } from '@/hooks/use-toast'
    31→
    32→// ─── Types ────────────────────────────────────────────────────────────
    33→
    34→type ReferralUrgency = 'ROUTINE' | 'URGENT' | 'EMERGENCY'
    35→type ReferralStatus = 'PENDING' | 'ACCEPTED' | 'REJECTED' | 'COMPLETED' | 'CANCELLED'
    36→type Modality = 'ALLOPATHY' | 'AYURVEDA' | 'HOMEOPATHY'
    37→
    38→const SPECIALTIES = [
    39→  'Cardiology', 'Neurology', 'Orthopedics', 'Dermatology', 'Ophthalmology',
    40→  'ENT', 'Psychiatry', 'Oncology', 'Pulmonology', 'Gastroenterology',
    41→  'Nephrology', 'Endocrinology', 'Urology', 'General Surgery', 'Pediatrics', 'OB-GYN',
    42→] as const
    43→
    44→interface Practitioner {
    45→  id: string
    46→  name: string
    47→  specialty: string
    48→  modality: Modality
    49→}
    50→
    51→interface Referral {
    52→  id: string
    53→  patientId: string
    54→  patientName: string
    55→  fromPractitionerId: string
    56→  fromPractitionerName: string
    57→  fromModality: Modality
    58→  toPractitionerId: string
    59→  toPractitionerName: string
    60→  toModality: Modality
    61→  specialty: string
    62→  urgency: ReferralUrgency
    63→  status: ReferralStatus
    64→  clinicalSummary: string
    65→  reason: string
    66→  notes: string
    67→  createdAt: string
    68→  updatedAt: string
    69→}
    70→
    71→interface Patient {
    72→  id: string
    73→  firstName: string
    74→  lastName: string
    75→}
    76→
    77→// ─── Style Maps ───────────────────────────────────────────────────────
    78→
    79→const urgencyStyles: Record<ReferralUrgency, string> = {
    80→  ROUTINE: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300',
    81→  URGENT: 'bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200',
    82→  EMERGENCY: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
    83→}
    84→
    85→const statusStyles: Record<ReferralStatus, string> = {
    86→  PENDING: 'bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200',
    87→  ACCEPTED: 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200',
    88→  REJECTED: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
    89→  COMPLETED: 'bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200',
    90→  CANCELLED: 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400',
    91→}
    92→
    93→const urgencyIcons: Record<ReferralUrgency, React.ReactNode> = {
    94→  ROUTINE: <Clock className="h-3 w-3" />,
    95→  URGENT: <AlertTriangle className="h-3 w-3" />,
    96→  EMERGENCY: <AlertTriangle className="h-3 w-3" />,
    97→}
    98→
    99→const fadeSlide = {
   100→  initial: { opacity: 0, y: 12 },
   101→  animate: { opacity: 1, y: 0 },
   102→  transition: { duration: 0.3 },
   103→}
   104→
   105→// ─── Component ────────────────────────────────────────────────────────
   106→
   107→export function ReferralsSection() {
   108→  const { activeModality } = useAppStore()
   109→
   110→  const [patients, setPatients] = useState<Patient[]>([])
   111→  const [referrals, setReferrals] = useState<Referral[]>([])
   112→  const [practitioners, setPractitioners] = useState<Practitioner[]>([])
   113→  const [loading, setLoading] = useState(true)
   114→
   115→  // Filters
   116→  const [filterStatus, setFilterStatus] = useState<string>('ALL')
   117→  const [filterUrgency, setFilterUrgency] = useState<string>('ALL')
   118→  const [filterSpecialty, setFilterSpecialty] = useState<string>('ALL')
   119→
   120→  // New referral form
   121→  const [addOpen, setAddOpen] = useState(false)
   122→  const [formPatientId, setFormPatientId] = useState('')
   123→  const [formToPractitionerId, setFormToPractitionerId] = useState('')
   124→  const [formSpecialty, setFormSpecialty] = useState(SPECIALTIES[0])
   125→  const [formUrgency, setFormUrgency] = useState<ReferralUrgency>('ROUTINE')
   126→  const [formToModality, setFormToModality] = useState<Modality>('ALLOPATHY')
   127→  const [formClinicalSummary, setFormClinicalSummary] = useState('')
   128→  const [formReason, setFormReason] = useState('')
   129→  const [formNotes, setFormNotes] = useState('')
   130→  const [submitting, setSubmitting] = useState(false)
   131→
   132→  // Detail dialog
   133→  const [detailReferral, setDetailReferral] = useState<Referral | null>(null)
   134→
   135→  // ─── Data Loading ─────────────────────────────────────────────────
   136→
   137→  useEffect(() => {
   138→    fetch('/api/patients')
   139→      .then((r) => r.json())
   140→      .then((d) => {
   141→        const list = d.data ?? d.patients ?? []
   142→        setPatients(list)
   143→        if (list.length > 0) setFormPatientId(list[0].id)
   144→      })
   145→      .catch(() => {})
   146→  }, [])
   147→
   148→  useEffect(() => {
   149→    fetch('/api/referrals?practitioners=true')
   150→      .then((r) => r.json())
   151→      .then((d) => {
   152→        setPractitioners(d.practitioners ?? [])
   153→      })
   154→      .catch(() => {})
   155→  }, [])
   156→
   157→  const loadReferrals = useCallback(async () => {
   158→    setLoading(true)
   159→    try {
   160→      const params = new URLSearchParams()
   161→      if (filterStatus !== 'ALL') params.set('status', filterStatus)
   162→      if (filterUrgency !== 'ALL') params.set('urgency', filterUrgency)
   163→      if (filterSpecialty !== 'ALL') params.set('specialty', filterSpecialty)
   164→      const res = await fetch(`/api/referrals?${params.toString()}`)
   165→      const data = await res.json()
   166→      setReferrals(data.data ?? data.referrals ?? [])
   167→    } catch {
   168→      toast({ title: 'Error', description: 'Failed to load referrals', variant: 'destructive' })
   169→    } finally {
   170→      setLoading(false)
   171→    }
   172→  }, [filterStatus, filterUrgency, filterSpecialty])
   173→
   174→  useEffect(() => { loadReferrals() }, [loadReferrals])
   175→
   176→  // ─── Stats ────────────────────────────────────────────────────────
   177→
   178→  const stats = {
   179→    total: referrals.length,
   180→    pending: referrals.filter((r) => r.status === 'PENDING').length,
   181→    accepted: referrals.filter((r) => r.status === 'ACCEPTED').length,
   182→    completed: referrals.filter((r) => r.status === 'COMPLETED').length,
   183→  }
   184→
   185→  // ─── Actions ──────────────────────────────────────────────────────
   186→
   187→  const handleCreate = async () => {
   188→    if (!formPatientId || !formToPractitionerId) {
   189→      toast({ title: 'Validation', description: 'Select patient and to-practitioner', variant: 'destructive' })
   190→      return
   191→    }
   192→    setSubmitting(true)
   193→    try {
   194→      const res = await fetch('/api/referrals', {
   195→        method: 'POST',
   196→        headers: { 'Content-Type': 'application/json' },
   197→        body: JSON.stringify({
   198→          patientId: formPatientId,
   199→          toPractitionerId: formToPractitionerId,
   200→          specialty: formSpecialty,
   201→          urgency: formUrgency,
   202→          toModality: formToModality,
   203→          fromModality: activeModality,
   204→          clinicalSummary: formClinicalSummary,
   205→          reason: formReason,
   206→          notes: formNotes,
   207→        }),
   208→      })
   209→      if (res.ok) {
   210→        toast({ title: 'Referral Created', description: 'Referral has been submitted successfully' })
   211→        setAddOpen(false)
   212→        setFormClinicalSummary('')
   213→        setFormReason('')
   214→        setFormNotes('')
   215→        loadReferrals()
   216→      } else {
   217→        toast({ title: 'Error', description: 'Failed to create referral', variant: 'destructive' })
   218→      }
   219→    } catch {
   220→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   221→    } finally {
   222→      setSubmitting(false)
   223→    }
   224→  }
   225→
   226→  const handleStatusUpdate = async (id: string, newStatus: ReferralStatus) => {
   227→    try {
   228→      const res = await fetch('/api/referrals', {
   229→        method: 'PUT',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({ id, status: newStatus }),
   232→      })
   233→      if (res.ok) {
   234→        toast({ title: `Referral ${newStatus}`, description: `Status updated to ${newStatus}` })
   235→        loadReferrals()
   236→        setDetailReferral(null)
   237→      }
   238→    } catch {
   239→      toast({ title: 'Error', description: 'Failed to update referral', variant: 'destructive' })
   240→    }
   241→  }
   242→
   243→  // ─── Render ───────────────────────────────────────────────────────
   244→
   245→  return (
   246→    <motion.div {...fadeSlide} className="space-y-4">
   247→      {/* Stats Cards */}
   248→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   249→        {[
   250→          { label: 'Total', value: stats.total, icon: ArrowRightLeft, color: 'text-slate-600 dark:text-slate-400' },
   251→          { label: 'Pending', value: stats.pending, icon: Clock, color: 'text-amber-600 dark:text-amber-400' },
   252→          { label: 'Accepted', value: stats.accepted, icon: CheckCircle2, color: 'text-green-600 dark:text-green-400' },
   253→          { label: 'Completed', value: stats.completed, icon: Stethoscope, color: 'text-teal-600 dark:text-teal-400' },
   254→        ].map((s) => (
   255→          <Card key={s.label}>
   256→            <CardContent className="p-4 flex items-center gap-3">
   257→              <s.icon className={`h-5 w-5 ${s.color}`} />
   258→              <div>
   259→                <p className="text-2xl font-bold">{s.value}</p>
   260→                <p className="text-xs text-muted-foreground">{s.label}</p>
   261→              </div>
   262→            </CardContent>
   263→          </Card>
   264→        ))}
   265→      </div>
   266→
   267→      {/* Toolbar */}
   268→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   269→        <div className="flex items-center gap-2 flex-wrap">
   270→          <Filter className="h-4 w-4 text-muted-foreground" />
   271→          <Select value={filterStatus} onValueChange={setFilterStatus}>
   272→            <SelectTrigger className="w-32 h-8"><SelectValue placeholder="Status" /></SelectTrigger>
   273→            <SelectContent>
   274→              <SelectItem value="ALL">All Status</SelectItem>
   275→              <SelectItem value="PENDING">Pending</SelectItem>
   276→              <SelectItem value="ACCEPTED">Accepted</SelectItem>
   277→              <SelectItem value="REJECTED">Rejected</SelectItem>
   278→              <SelectItem value="COMPLETED">Completed</SelectItem>
   279→              <SelectItem value="CANCELLED">Cancelled</SelectItem>
   280→            </SelectContent>
   281→          </Select>
   282→          <Select value={filterUrgency} onValueChange={setFilterUrgency}>
   283→            <SelectTrigger className="w-32 h-8"><SelectValue placeholder="Urgency" /></SelectTrigger>
   284→            <SelectContent>
   285→              <SelectItem value="ALL">All Urgency</SelectItem>
   286→              <SelectItem value="ROUTINE">Routine</SelectItem>
   287→              <SelectItem value="URGENT">Urgent</SelectItem>
   288→              <SelectItem value="EMERGENCY">Emergency</SelectItem>
   289→            </SelectContent>
   290→          </Select>
   291→          <Select value={filterSpecialty} onValueChange={setFilterSpecialty}>
   292→            <SelectTrigger className="w-36 h-8"><SelectValue placeholder="Specialty" /></SelectTrigger>
   293→            <SelectContent>
   294→              <SelectItem value="ALL">All Specialties</SelectItem>
   295→              {SPECIALTIES.map((s) => (
   296→                <SelectItem key={s} value={s}>{s}</SelectItem>
   297→              ))}
   298→            </SelectContent>
   299→          </Select>
   300→        </div>
   301→
   302→        <div className="sm:ml-auto">
   303→          <Dialog open={addOpen} onOpenChange={setAddOpen}>
   304→            <DialogTrigger asChild>
   305→              <Button className="gap-2">
   306→                <Plus className="h-4 w-4" />
   307→                New Referral
   308→              </Button>
   309→            </DialogTrigger>
   310→            <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   311→              <DialogHeader>
   312→                <DialogTitle className="flex items-center gap-2">
   313→                  <ArrowRightLeft className="h-5 w-5" />
   314→                  Create Referral
   315→                </DialogTitle>
   316→                <DialogDescription>Cross-practitioner referral with cross-modality support</DialogDescription>
   317→              </DialogHeader>
   318→              <div className="grid gap-4 py-4">
   319→                {/* Patient */}
   320→                <div className="space-y-2">
   321→                  <Label>Patient</Label>
   322→                  <Select value={formPatientId} onValueChange={setFormPatientId}>
   323→                    <SelectTrigger><SelectValue placeholder="Select patient" /></SelectTrigger>
   324→                    <SelectContent>
   325→                      {patients.map((p) => (
   326→                        <SelectItem key={p.id} value={p.id}>
   327→                          {p.firstName} {p.lastName}
   328→                        </SelectItem>
   329→                      ))}
   330→                    </SelectContent>
   331→                  </Select>
   332→                </div>
   333→
   334→                {/* Cross-modality row */}
   335→                <div className="grid grid-cols-2 gap-3">
   336→                  <div className="space-y-2">
   337→                    <Label>From Modality</Label>
   338→                    <div className="flex items-center h-9 px-3 rounded-md border bg-muted/50">
   339→                      <ModalityBadge modality={activeModality} />
   340→                    </div>
   341→                  </div>
   342→                  <div className="space-y-2">
   343→                    <Label>To Modality</Label>
   344→                    <Select value={formToModality} onValueChange={(v) => setFormToModality(v as Modality)}>
   345→                      <SelectTrigger><SelectValue /></SelectTrigger>
   346→                      <SelectContent>
   347→                        <SelectItem value="ALLOPATHY">Allopathy</SelectItem>
   348→                        <SelectItem value="AYURVEDA">Ayurveda</SelectItem>
   349→                        <SelectItem value="HOMEOPATHY">Homeopathy</SelectItem>
   350→                      </SelectContent>
   351→                    </Select>
   352→                  </div>
   353→                </div>
   354→
   355→                {/* To Practitioner */}
   356→                <div className="space-y-2">
   357→                  <Label>To Practitioner</Label>
   358→                  <Select value={formToPractitionerId} onValueChange={setFormToPractitionerId}>
   359→                    <SelectTrigger><SelectValue placeholder="Select practitioner" /></SelectTrigger>
   360→                    <SelectContent>
   361→                      {practitioners
   362→                        .filter((p) => p.modality === formToModality)
   363→                        .map((p) => (
   364→                          <SelectItem key={p.id} value={p.id}>
   365→                            {p.name} — {p.specialty}
   366→                          </SelectItem>
   367→                        ))}
   368→                      {practitioners.filter((p) => p.modality === formToModality).length === 0 && (
   369→                        <SelectItem value="__none" disabled>No practitioners for this modality</SelectItem>
   370→                      )}
   371→                    </SelectContent>
   372→                  </Select>
   373→                </div>
   374→
   375→                {/* Specialty + Urgency */}
   376→                <div className="grid grid-cols-2 gap-3">
   377→                  <div className="space-y-2">
   378→                    <Label>Specialty</Label>
   379→                    <Select value={formSpecialty} onValueChange={(v) => setFormSpecialty(v as typeof SPECIALTIES[number])}>
   380→                      <SelectTrigger><SelectValue /></SelectTrigger>
   381→                      <SelectContent>
   382→                        {SPECIALTIES.map((s) => (
   383→                          <SelectItem key={s} value={s}>{s}</SelectItem>
   384→                        ))}
   385→                      </SelectContent>
   386→                    </Select>
   387→                  </div>
   388→                  <div className="space-y-2">
   389→                    <Label>Urgency</Label>
   390→                    <Select value={formUrgency} onValueChange={(v) => setFormUrgency(v as ReferralUrgency)}>
   391→                      <SelectTrigger><SelectValue /></SelectTrigger>
   392→                      <SelectContent>
   393→                        <SelectItem value="ROUTINE">Routine</SelectItem>
   394→                        <SelectItem value="URGENT">Urgent</SelectItem>
   395→                        <SelectItem value="EMERGENCY">Emergency</SelectItem>
   396→                      </SelectContent>
   397→                    </Select>
   398→                  </div>
   399→                </div>
   400→
   401→                {/* Clinical Summary */}
   402→                <div className="space-y-2">
   403→                  <Label>Clinical Summary</Label>
   404→                  <Textarea
   405→                    value={formClinicalSummary}
   406→                    onChange={(e) => setFormClinicalSummary(e.target.value)}
   407→                    placeholder="Brief clinical summary for the receiving practitioner..."
   408→                    rows={3}
   409→                  />
   410→                </div>
   411→
   412→                {/* Reason */}
   413→                <div className="space-y-2">
   414→                  <Label>Reason for Referral</Label>
   415→                  <Input
   416→                    value={formReason}
   417→                    onChange={(e) => setFormReason(e.target.value)}
   418→                    placeholder="e.g., Second opinion, specialized care..."
   419→                  />
   420→                </div>
   421→
   422→                {/* Notes */}
   423→                <div className="space-y-2">
   424→                  <Label>Notes</Label>
   425→                  <Textarea
   426→                    value={formNotes}
   427→                    onChange={(e) => setFormNotes(e.target.value)}
   428→                    placeholder="Additional notes..."
   429→                    rows={2}
   430→                  />
   431→                </div>
   432→              </div>
   433→              <DialogFooter>
   434→                <Button variant="outline" onClick={() => setAddOpen(false)}>Cancel</Button>
   435→                <Button onClick={handleCreate} disabled={submitting}>
   436→                  {submitting ? 'Submitting...' : 'Submit Referral'}
   437→                </Button>
   438→              </DialogFooter>
   439→            </DialogContent>
   440→          </Dialog>
   441→        </div>
   442→      </div>
   443→
   444→      {/* Referral Table */}
   445→      <Card>
   446→        <CardHeader>
   447→          <CardTitle className="text-base flex items-center gap-2">
   448→            <ArrowRightLeft className="h-5 w-5" />
   449→            Referrals
   450→          </CardTitle>
   451→          <CardDescription>Cross-practitioner referral management with cross-modality support</CardDescription>
   452→        </CardHeader>
   453→        <CardContent>
   454→          {loading ? (
   455→            <div className="space-y-2">
   456→              {[1, 2, 3, 4].map((i) => (
   457→                <Skeleton key={i} className="h-12 w-full" />
   458→              ))}
   459→            </div>
   460→          ) : (
   461→            <div className="overflow-x-auto">
   462→              <Table>
   463→                <TableHeader>
   464→                  <TableRow>
   465→                    <TableHead>Patient</TableHead>
   466→                    <TableHead>From → To</TableHead>
   467→                    <TableHead>Specialty</TableHead>
   468→                    <TableHead>Urgency</TableHead>
   469→                    <TableHead>Status</TableHead>
   470→                    <TableHead>Date</TableHead>
   471→                    <TableHead className="text-right">Actions</TableHead>
   472→                  </TableRow>
   473→                </TableHeader>
   474→                <TableBody>
   475→                  {referrals.map((r) => (
   476→                    <TableRow key={r.id}>
   477→                      <TableCell className="font-medium">{r.patientName}</TableCell>
   478→                      <TableCell>
   479→                        <div className="flex items-center gap-1 flex-wrap">
   480→                          <ModalityBadge modality={r.fromModality} />
   481→                          <span className="text-xs text-muted-foreground">→</span>
   482→                          <ModalityBadge modality={r.toModality} />
   483→                        </div>
   484→                      </TableCell>
   485→                      <TableCell>{r.specialty}</TableCell>
   486→                      <TableCell>
   487→                        <Badge variant="outline" className={`gap-1 ${urgencyStyles[r.urgency]}`}>
   488→                          {urgencyIcons[r.urgency]}
   489→                          {r.urgency}
   490→                        </Badge>
   491→                      </TableCell>
   492→                      <TableCell>
   493→                        <Badge variant="outline" className={statusStyles[r.status]}>
   494→                          {r.status}
   495→                        </Badge>
   496→                      </TableCell>
   497→                      <TableCell className="text-sm text-muted-foreground">
   498→                        {new Date(r.createdAt).toLocaleDateString()}
   499→                      </TableCell>
   500→                      <TableCell className="text-right">
   501→                        <div className="flex justify-end gap-1">
   502→                          <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={() => setDetailReferral(r)}>
   503→                            <Eye className="h-3.5 w-3.5" />
   504→                          </Button>
   505→                          {r.status === 'PENDING' && (
   506→                            <>
   507→                              <Button
   508→                                size="sm"
   509→                                variant="outline"
   510→                                className="h-7 gap-1 text-xs text-green-700 dark:text-green-400"
   511→                                onClick={() => handleStatusUpdate(r.id, 'ACCEPTED')}
   512→                              >
   513→                                <CheckCircle2 className="h-3 w-3" /> Accept
   514→                              </Button>
   515→                              <Button
   516→                                size="sm"
   517→                                variant="outline"
   518→                                className="h-7 gap-1 text-xs text-red-700 dark:text-red-400"
   519→                                onClick={() => handleStatusUpdate(r.id, 'REJECTED')}
   520→                              >
   521→                                <XCircle className="h-3 w-3" /> Reject
   522→                              </Button>
   523→                            </>
   524→                          )}
   525→                        </div>
   526→                      </TableCell>
   527→                    </TableRow>
   528→                  ))}
   529→                  {referrals.length === 0 && (
   530→                    <TableRow>

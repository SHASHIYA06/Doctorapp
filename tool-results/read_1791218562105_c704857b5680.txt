   149→// ──────────────────────────────────────────────
   150→// Component
   151→// ──────────────────────────────────────────────
   152→
   153→export function SymptomCheckerSection() {
   154→  const { activeModality, setActiveSection } = useAppStore()
   155→
   156→  // Form state
   157→  const [symptoms, setSymptoms] = useState<string[]>([])
   158→  const [symptomInput, setSymptomInput] = useState('')
   159→  const [age, setAge] = useState('')
   160→  const [gender, setGender] = useState('')
   161→  const [modality, setModality] = useState<Modality>(activeModality)
   162→
   163→  // Suggestion state
   164→  const [showSuggestions, setShowSuggestions] = useState(false)
   165→  const [filteredSuggestions, setFilteredSuggestions] = useState<string[]>([])
   166→  const inputRef = useRef<HTMLInputElement>(null)
   167→  const suggestionsRef = useRef<HTMLDivElement>(null)
   168→
   169→  // Results state
   170→  const [results, setResults] = useState<SymptomCheckResult | null>(null)
   171→  const [loading, setLoading] = useState(false)
   172→  const [expandedIssue, setExpandedIssue] = useState<string | null>(null)
   173→  const [aiLoading, setAiLoading] = useState(false)
   174→  const [aiResult, setAiResult] = useState<string | null>(null)
   175→
   176→  // Red flag detection
   177→  const RED_FLAG_SYMPTOMS = new Set(['chest pain', 'breathing difficulty', 'severe headache', 'sudden vision loss', 'slurred speech', 'weakness on one side', 'severe bleeding', 'high fever', 'seizure', 'loss of consciousness'])
   178→  const detectedRedFlags = symptoms.filter((s) => RED_FLAG_SYMPTOMS.has(s.toLowerCase()))
   179→
   180→  // ── Symptom management ─────────────────────
   181→
   182→  const addSymptom = (symptom: string) => {
   183→    const trimmed = symptom.trim().toLowerCase()
   184→    if (trimmed && !symptoms.includes(trimmed)) {
   185→      setSymptoms((prev) => [...prev, trimmed])
   186→    }
   187→    setSymptomInput('')
   188→    setShowSuggestions(false)
   189→    inputRef.current?.focus()
   190→  }
   191→
   192→  const removeSymptom = (symptom: string) => {
   193→    setSymptoms((prev) => prev.filter((s) => s !== symptom))
   194→  }
   195→
   196→  const handleSymptomKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   197→    if (e.key === 'Enter' && symptomInput.trim()) {
   198→      e.preventDefault()
   199→      addSymptom(symptomInput)
   200→    }
   201→    if (e.key === 'Backspace' && !symptomInput && symptoms.length > 0) {
   202→      setSymptoms((prev) => prev.slice(0, -1))
   203→    }
   204→  }
   205→
   206→  const handleSymInputChange = (value: string) => {
   207→    setSymptomInput(value)
   208→    if (value.trim()) {
   209→      const lower = value.toLowerCase()
   210→      const filtered = COMMON_SYMPTOMS.filter(
   211→        (s) => s.includes(lower) && !symptoms.includes(s)
   212→      )
   213→      setFilteredSuggestions(filtered.slice(0, 8))
   214→      setShowSuggestions(filtered.length > 0)
   215→    } else {
   216→      setShowSuggestions(false)
   217→    }
   218→  }
   219→
   220→  // Close suggestions on outside click
   221→  useEffect(() => {
   222→    const handler = (e: MouseEvent) => {
   223→      if (
   224→        suggestionsRef.current &&
   225→        !suggestionsRef.current.contains(e.target as Node) &&
   226→        inputRef.current &&
   227→        !inputRef.current.contains(e.target as Node)
   228→      ) {
   229→        setShowSuggestions(false)
   230→      }
   231→    }
   232→    document.addEventListener('mousedown', handler)
   233→    return () => document.removeEventListener('mousedown', handler)
   234→  }, [])
   235→
   236→  // ── Submit handler ─────────────────────────
   237→
   238→  const handleCheckSymptoms = async () => {
   239→    if (symptoms.length === 0) {
   240→      toast({ title: 'Add symptoms', description: 'Please add at least one symptom', variant: 'destructive' })
   241→      return
   242→    }
   243→    if (!age) {
   244→      toast({ title: 'Select age group', description: 'Please select your age range', variant: 'destructive' })
   245→      return
   246→    }
   247→    if (!gender) {
   248→      toast({ title: 'Select gender', description: 'Please select your gender', variant: 'destructive' })
   249→      return
   250→    }
   251→
   252→    setLoading(true)
   253→    setResults(null)
   254→    setExpandedIssue(null)
   255→
   256→    try {
   257→      const res = await fetch('/api/symptom-checker', {
   258→        method: 'POST',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({ symptoms, age, gender, modality }),
   261→      })
   262→      const data = await res.json()
   263→      const result: SymptomCheckResult = data.data ?? data
   264→      setResults(result)
   265→      toast({
   266→        title: 'Symptom Check Complete',
   267→        description: `${result.matchedIssues?.length ?? 0} possible conditions found`,
   268→      })
   269→    } catch {
   270→      toast({ title: 'Error', description: 'Failed to check symptoms', variant: 'destructive' })
   271→    } finally {
   272→      setLoading(false)
   273→    }
   274→  }
   275→
   276→  // ── Helpers ────────────────────────────────
   277→
   278→  const canSubmit = symptoms.length > 0 && age && gender
   279→
   280→  const getSeverityBadge = (severity: string) => {
   281→    const config = SEVERITY_CONFIG[severity] ?? SEVERITY_CONFIG.MODERATE
   282→    return (
   283→      <Badge variant="outline" className={config.className}>
   284→        {config.label}
   285→      </Badge>
   286→    )
   287→  }
   288→
   289→  const getRelevanceColor = (score: number) => {
   290→    if (score >= 80) return 'text-red-600 dark:text-red-400'
   291→    if (score >= 60) return 'text-orange-600 dark:text-orange-400'
   292→    if (score >= 40) return 'text-amber-600 dark:text-amber-400'
   293→    return 'text-green-600 dark:text-green-400'
   294→  }
   295→
   296→  const getWarningIcon = (severity: string) => {
   297→    if (severity === 'CRITICAL') return <ShieldAlert className="h-4 w-4 text-red-500" />
   298→    if (severity === 'WARNING') return <AlertTriangle className="h-4 w-4 text-amber-500" />
   299→    return <Info className="h-4 w-4 text-teal-500" />
   300→  }
   301→
   302→  // ── Render ─────────────────────────────────
   303→
   304→  return (
   305→    <TooltipProvider>
   306→      <motion.div {...fadeSlide} className="space-y-6">
   307→        {/* ── Red Flag Detection Banner ── */}
   308→        {detectedRedFlags.length > 0 && (
   309→          <motion.div
   310→            initial={{ opacity: 0, scale: 0.98 }}
   311→            animate={{ opacity: 1, scale: 1 }}
   312→            className="flex items-start gap-3 p-4 rounded-lg border-2 border-red-300 bg-red-50 dark:border-red-700 dark:bg-red-950"
   313→          >
   314→            <Siren className="h-6 w-6 text-red-600 dark:text-red-400 shrink-0 mt-0.5" />
   315→            <div>
   316→              <h3 className="font-bold text-red-800 dark:text-red-200 text-sm">
   317→                Red Flag Symptoms Detected!
   318→              </h3>
   319→              <p className="text-xs text-red-700 dark:text-red-300 mt-1">
   320→                The following symptoms require immediate medical attention:
   321→              </p>
   322→              <div className="flex flex-wrap gap-1.5 mt-2">
   323→                {detectedRedFlags.map((flag) => (
   324→                  <Badge key={flag} variant="outline" className="text-xs bg-red-100 text-red-800 border-red-300 dark:bg-red-900 dark:text-red-200 dark:border-red-700">
   325→                    {flag}
   326→                  </Badge>
   327→                ))}
   328→              </div>
   329→              <p className="text-xs text-red-700 dark:text-red-300 mt-2 font-medium">
   330→                Please seek emergency care or call 108 (India Emergency) immediately.
   331→              </p>
   332→            </div>
   333→          </motion.div>
   334→        )}
   335→
   336→        {/* ── Input Card ─────────────────────── */}
   337→        <Card>
   338→          <CardHeader>
   339→            <CardTitle className="text-lg flex items-center gap-2">
   340→              <Stethoscope className="h-5 w-5" />
   341→              Symptom Checker
   342→            </CardTitle>
   343→            <CardDescription>
   344→              Enter your symptoms to get potential health insights across modalities
   345→            </CardDescription>
   346→          </CardHeader>
   347→          <CardContent className="space-y-5">
   348→            {/* Symptom Tags Input */}
   349→            <div className="space-y-2">
   350→              <Label>Symptoms</Label>
   351→              <div className="flex flex-wrap items-center gap-2 p-3 border rounded-lg bg-background min-h-[48px] focus-within:ring-2 focus-within:ring-ring focus-within:ring-offset-2">
   352→                <AnimatePresence>
   353→                  {symptoms.map((symptom) => (
   354→                    <motion.span
   355→                      key={symptom}
   356→                      initial={{ opacity: 0, scale: 0.8 }}
   357→                      animate={{ opacity: 1, scale: 1 }}
   358→                      exit={{ opacity: 0, scale: 0.8 }}
   359→                      className="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-sm font-medium bg-primary/10 text-primary border border-primary/20"
   360→                    >
   361→                      {symptom}
   362→                      <button
   363→                        type="button"
   364→                        onClick={() => removeSymptom(symptom)}
   365→                        className="ml-0.5 hover:bg-primary/20 rounded-full p-0.5 transition-colors"
   366→                      >
   367→                        <X className="h-3 w-3" />
   368→                      </button>
   369→                    </motion.span>
   370→                  ))}
   371→                </AnimatePresence>
   372→                <div className="relative flex-1 min-w-[140px]">
   373→                  <input
   374→                    ref={inputRef}
   375→                    type="text"
   376→                    value={symptomInput}
   377→                    onChange={(e) => handleSymInputChange(e.target.value)}
   378→                    onKeyDown={handleSymptomKeyDown}
   379→                    onFocus={() => {
   380→                      if (symptomInput.trim()) handleSymInputChange(symptomInput)
   381→                    }}
   382→                    placeholder={symptoms.length === 0 ? 'Type a symptom (e.g. headache, fever, pet dard)...' : 'Add more...'}
   383→                    className="w-full bg-transparent outline-none text-sm placeholder:text-muted-foreground"
   384→                  />
   385→                </div>
   386→              </div>
   387→              {/* Suggestions Dropdown */}
   388→              <AnimatePresence>
   389→                {showSuggestions && filteredSuggestions.length > 0 && (
   390→                  <motion.div
   391→                    ref={suggestionsRef}
   392→                    initial={{ opacity: 0, y: -4 }}
   393→                    animate={{ opacity: 1, y: 0 }}
   394→                    exit={{ opacity: 0, y: -4 }}
   395→                    className="border rounded-lg bg-popover shadow-md overflow-hidden z-10"
   396→                  >
   397→                    <ScrollArea className="max-h-48">
   398→                      {filteredSuggestions.map((suggestion) => (
   399→                        <button
   400→                          key={suggestion}
   401→                          type="button"
   402→                          onClick={() => addSymptom(suggestion)}
   403→                          className="w-full text-left px-3 py-2 text-sm hover:bg-accent transition-colors flex items-center gap-2"
   404→                        >
   405→                          <Plus className="h-3.5 w-3.5 text-muted-foreground" />
   406→                          {suggestion}
   407→                        </button>
   408→                      ))}
   409→                    </ScrollArea>
   410→                  </motion.div>
   411→                )}
   412→              </AnimatePresence>
   413→              {symptoms.length === 0 && (
   414→                <p className="text-xs text-muted-foreground flex items-center gap-1">
   415→                  <Info className="h-3 w-3" />
   416→                  Press Enter or select from suggestions to add symptoms
   417→                </p>
   418→              )}
   419→            </div>
   420→
   421→            {/* Age, Gender, Modality Row */}
   422→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   423→              {/* Age */}
   424→              <div className="space-y-2">
   425→                <Label>Age Group</Label>
   426→                <Select value={age} onValueChange={setAge}>
   427→                  <SelectTrigger>
   428→                    <SelectValue placeholder="Select age..." />
   429→                  </SelectTrigger>
   430→                  <SelectContent>
   431→                    {AGE_OPTIONS.map((opt) => (
   432→                      <SelectItem key={opt.value} value={opt.value}>
   433→                        {opt.label}
   434→                      </SelectItem>
   435→                    ))}
   436→                  </SelectContent>
   437→                </Select>
   438→              </div>
   439→
   440→              {/* Gender */}
   441→              <div className="space-y-2">
   442→                <Label>Gender</Label>
   443→                <Select value={gender} onValueChange={setGender}>
   444→                  <SelectTrigger>
   445→                    <SelectValue placeholder="Select gender..." />
   446→                  </SelectTrigger>
   447→                  <SelectContent>
   448→                    {GENDER_OPTIONS.map((opt) => (
   449→                      <SelectItem key={opt.value} value={opt.value}>
   450→                        {opt.label}
   451→                      </SelectItem>
   452→                    ))}
   453→                  </SelectContent>
   454→                </Select>
   455→              </div>
   456→
   457→              {/* Modality */}
   458→              <div className="space-y-2">
   459→                <Label>Treatment Modality</Label>
   460→                <Select value={modality} onValueChange={(v) => setModality(v as Modality)}>
   461→                  <SelectTrigger>
   462→                    <SelectValue />
   463→                  </SelectTrigger>
   464→                  <SelectContent>
   465→                    {Object.entries(MODALITY_CONFIG).map(([key, cfg]) => {
   466→                      const Icon = cfg.icon
   467→                      return (
   468→                        <SelectItem key={key} value={key}>
   469→                          <span className="flex items-center gap-2">
   470→                            <Icon className={`h-4 w-4 ${cfg.color}`} />
   471→                            {cfg.label}
   472→                          </span>
   473→                        </SelectItem>
   474→                      )
   475→                    })}
   476→                  </SelectContent>
   477→                </Select>
   478→              </div>
   479→            </div>
   480→
   481→            {/* Check Button */}
   482→            <div className="flex flex-wrap items-center gap-3">
   483→              <Button
   484→                onClick={handleCheckSymptoms}
   485→                disabled={!canSubmit || loading}
   486→                className="gap-2 min-w-[180px]"
   487→              >
   488→                {loading ? (
   489→                  <>
   490→                    <Loader2 className="h-4 w-4 animate-spin" />
   491→                    Checking...
   492→                  </>
   493→                ) : (
   494→                  <>
   495→                    <Search className="h-4 w-4" />
   496→                    Check Symptoms
   497→                  </>
   498→                )}
   499→              </Button>
   500→              {/* Voice Input Button */}
   501→              <Button
   502→                variant="outline"
   503→                className="gap-2"
   504→                onClick={() => setActiveSection('voice')}
   505→              >
   506→                <Mic className="h-4 w-4" />
   507→                Voice Input
   508→              </Button>
   509→              {/* Get AI Analysis */}
   510→              <Button
   511→                variant="outline"
   512→                className="gap-2"
   513→                disabled={symptoms.length === 0 || aiLoading}
   514→                onClick={async () => {
   515→                  setAiLoading(true)
   516→                  setAiResult(null)
   517→                  try {
   518→                    const res = await fetch('/api/medicine-rag', {
   519→                      method: 'POST',
   520→                      headers: { 'Content-Type': 'application/json' },
   521→                      body: JSON.stringify({ query: symptoms.join(', '), modality: modality, language: 'en' }),
   522→                    })
   523→                    const data = await res.json()
   524→                    setAiResult(data.data?.answer ?? data.data?.text ?? JSON.stringify(data.data ?? data))
   525→                    toast({ title: 'AI Analysis Complete', description: 'Results loaded below' })
   526→                  } catch {
   527→                    setAiResult('AI analysis is currently unavailable. Please try again later.')
   528→                    toast({ title: 'AI Unavailable', variant: 'destructive' })
   529→                  } finally {
   530→                    setAiLoading(false)
   531→                  }
   532→                }}
   533→              >
   534→                {aiLoading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
   535→                Get AI Analysis
   536→              </Button>
   537→              {symptoms.length > 0 && (
   538→                <Button
   539→                  variant="ghost"
   540→                  size="sm"
   541→                  onClick={() => {
   542→                    setSymptoms([])
   543→                    setSymptomInput('')
   544→                    setResults(null)
   545→                    setAiResult(null)
   546→                  }}
   547→                  className="text-muted-foreground"
   548→                >
   549→                  Clear all
   550→                </Button>
   551→              )}
   552→            </div>
   553→          </CardContent>
   554→        </Card>
   555→
   556→        {/* ── Loading Skeleton ────────────────── */}
   557→        <AnimatePresence>
   558→          {loading && (
   559→            <motion.div
   560→              initial={{ opacity: 0 }}
   561→              animate={{ opacity: 1 }}
   562→              exit={{ opacity: 0 }}
   563→              className="space-y-4"
   564→            >

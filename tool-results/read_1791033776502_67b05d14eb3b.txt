   136→// ─── Helper functions ────────────────────────────────────────────────────────
   137→
   138→function isOverdue(reminder: Reminder): boolean {
   139→  if (reminder.status === 'COMPLETED' || reminder.status === 'EXPIRED') return false
   140→  const scheduled = new Date(`${reminder.scheduledDate}T${reminder.scheduledTime}`)
   141→  return scheduled < new Date()
   142→}
   143→
   144→function isUpcoming(reminder: Reminder): boolean {
   145→  if (reminder.status !== 'ACTIVE') return false
   146→  const scheduled = new Date(`${reminder.scheduledDate}T${reminder.scheduledTime}`)
   147→  const now = new Date()
   148→  const sevenDays = 7 * 24 * 60 * 60 * 1000
   149→  return scheduled >= now && (scheduled.getTime() - now.getTime()) <= sevenDays
   150→}
   151→
   152→function isDueToday(reminder: Reminder): boolean {
   153→  if (reminder.status !== 'ACTIVE') return false
   154→  const today = new Date().toISOString().slice(0, 10)
   155→  return reminder.scheduledDate === today
   156→}

   294→export function FollowUpRemindersSection() {
   295→  const { selectedPatientId } = useAppStore()
   296→
   297→  const [patients, setPatients] = useState<Patient[]>([])
   298→  const [selectedPatient, setSelectedPatient] = useState<string>('ALL')
   299→  const [reminders, setReminders] = useState<Reminder[]>([])
   300→  const [loading, setLoading] = useState(true)
   301→
   302→  // Calendar date
   303→  const [calendarDate, setCalendarDate] = useState<Date | undefined>(new Date())
   304→
   305→  // Create dialog
   306→  const [createOpen, setCreateOpen] = useState(false)
   307→  const [newTitle, setNewTitle] = useState('')
   308→  const [newDesc, setNewDesc] = useState('')
   309→  const [newDate, setNewDate] = useState(fmt(today))
   310→  const [newTime, setNewTime] = useState('09:00')
   311→  const [newType, setNewType] = useState<ReminderType>('FOLLOW_UP')
   312→  const [newRecurrence, setNewRecurrence] = useState<RecurrenceType>('ONCE')
   313→  const [newChannel, setNewChannel] = useState<ChannelType>('IN_APP')
   314→  const [createSubmitting, setCreateSubmitting] = useState(false)
   315→
   316→  // ── Fetch data ──
   317→  useEffect(() => {
   318→    fetch('/api/patients')
   319→      .then((r) => r.json())
   320→      .then((d) => {
   321→        const list = d.data ?? d.patients ?? []
   322→        if (Array.isArray(list) && list.length > 0) setPatients(list)
   323→        else setPatients(DEMO_PATIENTS)
   324→      })
   325→      .catch(() => setPatients(DEMO_PATIENTS))
   326→  }, [])
   327→
   328→  useEffect(() => {
   329→    setLoading(true)
   330→    const params = selectedPatient !== 'ALL' ? `?patientId=${selectedPatient}` : ''
   331→    fetch(`/api/follow-up-reminders${params}`)
   332→      .then((r) => r.json())
   333→      .then((d) => {
   334→        const data = d.data ?? d.reminders ?? []
   335→        if (Array.isArray(data) && data.length > 0) {
   336→          setReminders(data)
   337→        } else {
   338→          setReminders(DEMO_REMINDERS)
   339→        }
   340→      })
   341→      .catch(() => setReminders(DEMO_REMINDERS))
   342→      .finally(() => setLoading(false))
   343→  }, [selectedPatient, selectedPatientId])
   344→
   345→  // ── Stats ──
   346→  const stats = useMemo(() => {
   347→    const now = new Date()
   348→    const weekAgo = new Date(now.getTime() - 7 * 86400000)
   349→    return {
   350→      totalActive: reminders.filter((r) => r.status === 'ACTIVE').length,
   351→      dueToday: reminders.filter(isDueToday).length,
   352→      overdue: reminders.filter(isOverdue).length,
   353→      completedThisWeek: reminders.filter((r) => r.status === 'COMPLETED' && r.completedAt && new Date(r.completedAt) >= weekAgo).length,
   354→    }
   355→  }, [reminders])
   356→
   357→  // ── Categorized reminders ──
   358→  const overdueReminders = useMemo(() => reminders.filter(isOverdue), [reminders])
   359→  const upcomingReminders = useMemo(
   360→    () => reminders.filter(isUpcoming).sort((a, b) => new Date(`${a.scheduledDate}T${a.scheduledTime}`).getTime() - new Date(`${b.scheduledDate}T${b.scheduledTime}`).getTime()),
   361→    [reminders]
   362→  )
   363→
   364→  // Calendar dates with reminders
   365→  const reminderDates = useMemo(() => {
   366→    const dateMap = new Map<string, Reminder[]>()
   367→    reminders.forEach((r) => {
   368→      if (r.status === 'ACTIVE') {
   369→        const existing = dateMap.get(r.scheduledDate) ?? []
   370→        existing.push(r)
   371→        dateMap.set(r.scheduledDate, existing)
   372→      }
   373→    })
   374→    return dateMap
   375→  }, [reminders])
   376→
   377→  const selectedDateReminders = useMemo(() => {
   378→    if (!calendarDate) return []
   379→    const key = fmt(calendarDate)
   380→    return reminderDates.get(key) ?? []
   381→  }, [calendarDate, reminderDates])
   382→
   383→  // ── Mark as complete ──
   384→  const handleComplete = useCallback(async (id: string) => {
   385→    try {
   386→      await fetch('/api/follow-up-reminders', {
   387→        method: 'PUT',
   388→        headers: { 'Content-Type': 'application/json' },
   389→        body: JSON.stringify({ id, status: 'COMPLETED' }),
   390→      })
   391→    } catch {
   392→      // local update anyway
   393→    }
   394→    setReminders((prev) =>
   395→      prev.map((r) => (r.id === id ? { ...r, status: 'COMPLETED' as ReminderStatus, completedAt: new Date().toISOString() } : r))
   396→    )
   397→    toast({ title: 'Reminder Completed', description: 'The reminder has been marked as complete' })
   398→  }, [])
   399→
   400→  // ── Create reminder ──
   401→  const handleCreate = async () => {
   402→    if (!newTitle.trim()) {
   403→      toast({ title: 'Validation Error', description: 'Title is required', variant: 'destructive' })
   404→      return
   405→    }
   406→    setCreateSubmitting(true)
   407→    const patient = patients.find((p) => p.id === selectedPatient) ?? patients[0]
   408→    const newReminder: Reminder = {
   409→      id: `rem-${Date.now()}`,
   410→      patientId: selectedPatient !== 'ALL' ? selectedPatient : (patient?.id ?? 'p-001'),
   411→      patientName: patient ? `${patient.firstName} ${patient.lastName}` : 'Unknown',
   412→      title: newTitle,
   413→      description: newDesc,
   414→      type: newType,
   415→      scheduledDate: newDate,
   416→      scheduledTime: newTime,
   417→      recurrence: newRecurrence,
   418→      channel: newChannel,
   419→      deliveryStatus: 'PENDING',
   420→      status: 'ACTIVE',
   421→      completedAt: null,
   422→      createdAt: new Date().toISOString(),
   423→    }
   424→    try {
   425→      await fetch('/api/follow-up-reminders', {
   426→        method: 'POST',
   427→        headers: { 'Content-Type': 'application/json' },
   428→        body: JSON.stringify(newReminder),
   429→      })
   430→    } catch {
   431→      // local update anyway
   432→    }
   433→    setReminders((prev) => [newReminder, ...prev])
   434→    toast({ title: 'Reminder Created', description: `"${newTitle}" scheduled for ${newDate}` })
   435→    setCreateOpen(false)
   436→    resetCreateForm()
   437→    setCreateSubmitting(false)
   438→  }
   439→
   440→  const resetCreateForm = () => {
   441→    setNewTitle('')
   442→    setNewDesc('')
   443→    setNewDate(fmt(today))
   444→    setNewTime('09:00')
   445→    setNewType('FOLLOW_UP')
   446→    setNewRecurrence('ONCE')
   447→    setNewChannel('IN_APP')
   448→  }
   449→
   450→  // ── Quick actions ──
   451→  const handleQuickAction = (type: ReminderType, title: string, desc: string) => {
   452→    setNewType(type)
   453→    setNewTitle(title)
   454→    setNewDesc(desc)
   455→    setCreateOpen(true)
   456→  }
   457→
   458→  // ── Loading skeleton ──
   459→  if (loading) {
   460→    return (
   461→      <div className="space-y-3">
   462→        {[1, 2, 3, 4].map((i) => (
   463→          <Skeleton key={i} className="h-24 w-full rounded-lg" />
   464→        ))}
   465→      </div>
   466→    )
   467→  }
   468→
   469→  return (
   470→    <motion.div {...fadeSlide} className="space-y-4">
   471→      {/* ── Stats Row ── */}
   472→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   473→        <Card>
   474→          <CardContent className="p-4 text-center">
   475→            <div className="flex items-center justify-center gap-2 mb-1">
   476→              <Bell className="h-4 w-4 text-blue-600" />
   477→              <span className="text-2xl font-bold">{stats.totalActive}</span>
   478→            </div>
   479→            <p className="text-xs text-muted-foreground">Total Active</p>
   480→          </CardContent>
   481→        </Card>
   482→        <Card>
   483→          <CardContent className="p-4 text-center">
   484→            <div className="flex items-center justify-center gap-2 mb-1">
   485→              <Timer className="h-4 w-4 text-amber-600" />
   486→              <span className="text-2xl font-bold text-amber-600">{stats.dueToday}</span>
   487→            </div>
   488→            <p className="text-xs text-muted-foreground">Due Today</p>
   489→          </CardContent>
   490→        </Card>
   491→        <Card>
   492→          <CardContent className="p-4 text-center">
   493→            <div className="flex items-center justify-center gap-2 mb-1">
   494→              <CalendarX className="h-4 w-4 text-red-600" />
   495→              <span className="text-2xl font-bold text-red-600">{stats.overdue}</span>
   496→            </div>
   497→            <p className="text-xs text-muted-foreground">Overdue</p>
   498→          </CardContent>
   499→        </Card>
   500→        <Card>
   501→          <CardContent className="p-4 text-center">
   502→            <div className="flex items-center justify-center gap-2 mb-1">

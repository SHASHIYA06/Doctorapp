   520→export function HealthIssuesSection() {
   521→  const { selectedIssueId, setSelectedIssueId } = useAppStore()
   522→
   523→  // State
   524→  const [issues, setIssues] = useState<HealthIssueListItem[]>([])
   525→  const [pagination, setPagination] = useState<Pagination>({
   526→    page: 1,
   527→    limit: 20,
   528→    total: 0,
   529→    totalPages: 0,
   530→  })
   531→  const [bodySystemFilters, setBodySystemFilters] = useState<string[]>([])
   532→  const [search, setSearch] = useState('')
   533→  const [bodySystem, setBodySystem] = useState('')
   534→  const [page, setPage] = useState(1)
   535→  const [loading, setLoading] = useState(true)
   536→  const [seeding, setSeeding] = useState(false)
   537→  const [seedDone, setSeedDone] = useState(false)
   538→
   539→  // Detail panel
   540→  const [detail, setDetail] = useState<HealthIssueDetail | null>(null)
   541→  const [detailLoading, setDetailLoading] = useState(false)
   542→  const [detailOpen, setDetailOpen] = useState(false)
   543→  const [activeWing, setActiveWing] = useState('ALLOPATHY')
   544→
   545→  // ──────────────────────────────────────────
   546→  // Auto-seed on first load
   547→  // ──────────────────────────────────────────
   548→  useEffect(() => {
   549→    if (seedDone) return
   550→    const doSeed = async () => {
   551→      setSeeding(true)
   552→      try {
   553→        await fetch('/api/seed', { method: 'POST' })
   554→      } catch {
   555→        // silent — seed may already exist
   556→      } finally {
   557→        setSeeding(false)
   558→        setSeedDone(true)
   559→      }
   560→    }
   561→    doSeed()
   562→  }, [seedDone])
   563→
   564→  // ──────────────────────────────────────────
   565→  // Fetch issues list
   566→  // ──────────────────────────────────────────
   567→  const fetchIssues = useCallback(async () => {
   568→    if (!seedDone) return
   569→    setLoading(true)
   570→    try {
   571→      const params = new URLSearchParams()
   572→      if (search) params.set('search', search)
   573→      if (bodySystem) params.set('bodySystem', bodySystem)
   574→      params.set('page', String(page))
   575→      params.set('limit', '20')
   576→
   577→      const res = await fetch(`/api/health-issues?${params.toString()}`)
   578→      const data = await res.json()
   579→      setIssues(data.data ?? [])
   580→      if (data.pagination) setPagination(data.pagination)
   581→      if (data.filters?.bodySystems) {
   582→        setBodySystemFilters(data.filters.bodySystems)
   583→      }
   584→    } catch {
   585→      // handle silently
   586→    } finally {
   587→      setLoading(false)
   588→    }
   589→  }, [search, bodySystem, page, seedDone])
   590→
   591→  useEffect(() => {
   592→    fetchIssues()
   593→  }, [fetchIssues])
   594→
   595→  // ──────────────────────────────────────────
   596→  // Fetch detail
   597→  // ──────────────────────────────────────────
   598→  const openDetail = async (issueId: string) => {
   599→    setDetailLoading(true)
   600→    setDetailOpen(true)

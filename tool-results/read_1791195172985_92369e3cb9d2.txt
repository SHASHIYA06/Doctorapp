     9→// Helper: date offset from now
    10→function daysFromNow(days: number): Date {
    11→  const d = new Date()
    12→  d.setDate(d.getDate() + days)
    13→  return d
    14→}
    15→
    16→function hoursFromNow(hours: number): Date {
    17→  const d = new Date()
    18→  d.setHours(d.getHours() + hours)
    19→  return d
    20→}

   136→// ─── Helper functions ────────────────────────────────────────────────────────
   137→
   138→function isOverdue(reminder: Reminder): boolean {
   139→  if (reminder.status === 'COMPLETED' || reminder.status === 'EXPIRED') return false
   140→  const scheduled = new Date(`${reminder.scheduledDate}T${reminder.scheduledTime}`)
   141→  return scheduled < new Date()
   142→}
   143→
   144→function isUpcoming(reminder: Reminder): boolean {
   145→  if (reminder.status !== 'ACTIVE') return false
   146→  const scheduled = new Date(`${reminder.scheduledDate}T${reminder.scheduledTime}`)
   147→  const now = new Date()
   148→  const sevenDays = 7 * 24 * 60 * 60 * 1000
   149→  return scheduled >= now && (scheduled.getTime() - now.getTime()) <= sevenDays
   150→}
   151→
   152→function isDueToday(reminder: Reminder): boolean {
   153→  if (reminder.status !== 'ACTIVE') return false
   154→  const today = new Date().toISOString().slice(0, 10)
   155→  return reminder.scheduledDate === today
   156→}

   294→export function FollowUpRemindersSection() {
   295→  const { selectedPatientId } = useAppStore()
   296→
   297→  const [patients, setPatients] = useState<Patient[]>([])
   298→  const [selectedPatient, setSelectedPatient] = useState<string>('ALL')
   299→  const [reminders, setReminders] = useState<Reminder[]>([])
   300→  const [loading, setLoading] = useState(true)
   301→
   302→  // Calendar date
   303→  const [calendarDate, setCalendarDate] = useState<Date | undefined>(new Date())
   304→
   305→  // Create dialog
   306→  const [createOpen, setCreateOpen] = useState(false)
   307→  const [newTitle, setNewTitle] = useState('')
   308→  const [newDesc, setNewDesc] = useState('')
   309→  const [newDate, setNewDate] = useState(fmt(today))
   310→  const [newTime, setNewTime] = useState('09:00')
   311→  const [newType, setNewType] = useState<ReminderType>('FOLLOW_UP')
   312→  const [newRecurrence, setNewRecurrence] = useState<RecurrenceType>('ONCE')
   313→  const [newChannel, setNewChannel] = useState<ChannelType>('IN_APP')
   314→  const [createSubmitting, setCreateSubmitting] = useState(false)
   315→
   316→  // ── Fetch data ──
   317→  useEffect(() => {
   318→    fetch('/api/patients')
   319→      .then((r) => r.json())
   320→      .then((d) => {
   321→        const list = d.data ?? d.patients ?? []
   322→        if (Array.isArray(list) && list.length > 0) setPatients(list)
   323→        else setPatients(DEMO_PATIENTS)
   324→      })
   325→      .catch(() => {
   326→        setPatients(DEMO_PATIENTS)
   327→        toast({ title: 'Error', description: 'Failed to load patients', variant: 'destructive' })
   328→      })
   329→  }, [])
   330→
   331→  useEffect(() => {
   332→    setLoading(true)
   333→    const params = selectedPatient !== 'ALL' ? `?patientId=${selectedPatient}` : ''
   334→    fetch(`/api/follow-up-reminders${params}`)
   335→      .then((r) => r.json())
   336→      .then((d) => {
   337→        const data = d.data ?? d.reminders ?? []
   338→        if (Array.isArray(data) && data.length > 0) {
   339→          setReminders(data)
   340→        } else {
   341→          setReminders(DEMO_REMINDERS)
   342→        }
   343→      })
   344→      .catch(() => setReminders(DEMO_REMINDERS))
   345→      .finally(() => setLoading(false))
   346→  }, [selectedPatient, selectedPatientId])
   347→
   348→  // ── Stats ──
   349→  const stats = useMemo(() => {
   350→    const now = new Date()
   351→    const weekAgo = new Date(now.getTime() - 7 * 86400000)
   352→    return {
   353→      totalActive: reminders.filter((r) => r.status === 'ACTIVE').length,
   354→      dueToday: reminders.filter(isDueToday).length,
   355→      overdue: reminders.filter(isOverdue).length,
   356→      completedThisWeek: reminders.filter((r) => r.status === 'COMPLETED' && r.completedAt && new Date(r.completedAt) >= weekAgo).length,
   357→    }
   358→  }, [reminders])
   359→
   360→  // ── Categorized reminders ──
   361→  const overdueReminders = useMemo(() => reminders.filter(isOverdue), [reminders])
   362→  const upcomingReminders = useMemo(
   363→    () => reminders.filter(isUpcoming).sort((a, b) => new Date(`${a.scheduledDate}T${a.scheduledTime}`).getTime() - new Date(`${b.scheduledDate}T${b.scheduledTime}`).getTime()),
   364→    [reminders]
   365→  )
   366→
   367→  // Calendar dates with reminders
   368→  const reminderDates = useMemo(() => {
   369→    const dateMap = new Map<string, Reminder[]>()
   370→    reminders.forEach((r) => {
   371→      if (r.status === 'ACTIVE') {
   372→        const existing = dateMap.get(r.scheduledDate) ?? []
   373→        existing.push(r)
   374→        dateMap.set(r.scheduledDate, existing)
   375→      }
   376→    })
   377→    return dateMap
   378→  }, [reminders])
   379→
   380→  const selectedDateReminders = useMemo(() => {
   381→    if (!calendarDate) return []
   382→    const key = fmt(calendarDate)
   383→    return reminderDates.get(key) ?? []
   384→  }, [calendarDate, reminderDates])
   385→
   386→  // ── Mark as complete ──
   387→  const handleComplete = useCallback(async (id: string) => {
   388→    try {
   389→      await fetch('/api/follow-up-reminders', {
   390→        method: 'PUT',
   391→        headers: { 'Content-Type': 'application/json' },
   392→        body: JSON.stringify({ id, status: 'COMPLETED' }),
   393→      })
   394→    } catch {
   395→      // local update anyway
   396→    }
   397→    setReminders((prev) =>
   398→      prev.map((r) => (r.id === id ? { ...r, status: 'COMPLETED' as ReminderStatus, completedAt: new Date().toISOString() } : r))
   399→    )
   400→    toast({ title: 'Reminder Completed', description: 'The reminder has been marked as complete' })
   401→  }, [])
   402→
   403→  // ── Create reminder ──
   404→  const handleCreate = async () => {
   405→    if (!newTitle.trim()) {
   406→      toast({ title: 'Validation Error', description: 'Title is required', variant: 'destructive' })
   407→      return
   408→    }
   409→    setCreateSubmitting(true)
   410→    const patient = patients.find((p) => p.id === selectedPatient) ?? patients[0]
   411→    const newReminder: Reminder = {
   412→      id: `rem-${Date.now()}`,
   413→      patientId: selectedPatient !== 'ALL' ? selectedPatient : (patient?.id ?? 'p-001'),
   414→      patientName: patient ? `${patient.firstName} ${patient.lastName}` : 'Unknown',
   415→      title: newTitle,
   416→      description: newDesc,
   417→      type: newType,
   418→      scheduledDate: newDate,
   419→      scheduledTime: newTime,
   420→      recurrence: newRecurrence,
   421→      channel: newChannel,
   422→      deliveryStatus: 'PENDING',
   423→      status: 'ACTIVE',
   424→      completedAt: null,
   425→      createdAt: new Date().toISOString(),
   426→    }
   427→    try {
   428→      await fetch('/api/follow-up-reminders', {
   429→        method: 'POST',
   430→        headers: { 'Content-Type': 'application/json' },
   431→        body: JSON.stringify(newReminder),
   432→      })
   433→    } catch {
   434→      // local update anyway
   435→    }
   436→    setReminders((prev) => [newReminder, ...prev])
   437→    toast({ title: 'Reminder Created', description: `"${newTitle}" scheduled for ${newDate}` })
   438→    setCreateOpen(false)
   439→    resetCreateForm()
   440→    setCreateSubmitting(false)
   441→  }
   442→
   443→  const resetCreateForm = () => {
   444→    setNewTitle('')
   445→    setNewDesc('')
   446→    setNewDate(fmt(today))
   447→    setNewTime('09:00')
   448→    setNewType('FOLLOW_UP')
   449→    setNewRecurrence('ONCE')
   450→    setNewChannel('IN_APP')
   451→  }
   452→
   453→  // ── Quick actions ──
   454→  const handleQuickAction = (type: ReminderType, title: string, desc: string) => {
   455→    setNewType(type)
   456→    setNewTitle(title)
   457→    setNewDesc(desc)
   458→    setCreateOpen(true)
   459→  }
   460→
   461→  // ── Loading skeleton ──
   462→  if (loading) {
   463→    return (
   464→      <div className="space-y-3">
   465→        {[1, 2, 3, 4].map((i) => (
   466→          <Skeleton key={i} className="h-24 w-full rounded-lg" />
   467→        ))}
   468→      </div>
   469→    )
   470→  }
   471→
   472→  return (
   473→    <motion.div {...fadeSlide} className="space-y-4">
   474→      {/* ── Stats Row ── */}
   475→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   476→        <Card>
   477→          <CardContent className="p-4 text-center">
   478→            <div className="flex items-center justify-center gap-2 mb-1">
   479→              <Bell className="h-4 w-4 text-blue-600" />
   480→              <span className="text-2xl font-bold">{stats.totalActive}</span>
   481→            </div>
   482→            <p className="text-xs text-muted-foreground">Total Active</p>
   483→          </CardContent>
   484→        </Card>
   485→        <Card>
   486→          <CardContent className="p-4 text-center">
   487→            <div className="flex items-center justify-center gap-2 mb-1">
   488→              <Timer className="h-4 w-4 text-amber-600" />
   489→              <span className="text-2xl font-bold text-amber-600">{stats.dueToday}</span>
   490→            </div>
   491→            <p className="text-xs text-muted-foreground">Due Today</p>
   492→          </CardContent>
   493→        </Card>
   494→        <Card>
   495→          <CardContent className="p-4 text-center">
   496→            <div className="flex items-center justify-center gap-2 mb-1">
   497→              <CalendarX className="h-4 w-4 text-red-600" />
   498→              <span className="text-2xl font-bold text-red-600">{stats.overdue}</span>

   500→            <p className="text-xs text-muted-foreground">Overdue</p>
   501→          </CardContent>
   502→        </Card>
   503→        <Card>
   504→          <CardContent className="p-4 text-center">
   505→            <div className="flex items-center justify-center gap-2 mb-1">
   506→              <CheckCircle2 className="h-4 w-4 text-green-600" />

   509→            <p className="text-xs text-muted-foreground">Completed This Week</p>

   529→        </div>

   532→            <Plus className="h-3.5 w-3.5" />

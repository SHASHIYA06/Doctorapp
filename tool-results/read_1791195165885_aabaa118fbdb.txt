   272→// ─── Count-Up Animation Hook ───────────────────────────────────────
   273→
   274→function useCountUp(end: number, duration: number = 1200, startOnMount: boolean = true) {
   275→  const [count, setCount] = useState(0)
   276→  const frameRef = useRef<number>(0)
   277→
   278→  useEffect(() => {
   279→    if (!startOnMount) return
   280→    let startTime: number | null = null
   281→    const step = (timestamp: number) => {
   282→      if (!startTime) startTime = timestamp
   283→      const progress = Math.min((timestamp - startTime) / duration, 1)
   284→      const eased = 1 - Math.pow(1 - progress, 3)
   285→      setCount(Math.floor(eased * end))
   286→      if (progress < 1) {
   287→        frameRef.current = requestAnimationFrame(step)
   288→      }
   289→    }
   290→    frameRef.current = requestAnimationFrame(step)
   291→    return () => cancelAnimationFrame(frameRef.current)
   292→  }, [end, duration, startOnMount])
   293→
   294→  return count
   295→}

   419→function formatTimeAgo(iso: string): string {
   420→  const diff = Date.now() - new Date(iso).getTime()
   421→  const mins = Math.floor(diff / 60000)
   422→  if (mins < 60) return `${mins}m ago`
   423→  const hrs = Math.floor(mins / 60)
   424→  if (hrs < 24) return `${hrs}h ago`
   425→  return `${Math.floor(hrs / 24)}d ago`
   426→}

 1 | /**
 2 |  * Escape ILIKE wildcard characters in a string derived from untrusted input.
 3 |  * In PostgreSQL ILIKE patterns, % matches any sequence of characters and _
 4 |  * matches any single character. The backslash (the LIKE escape character) is
 5 |  * escaped first so a user-supplied "\" can't alter the pattern.
 6 |  */
 7 | export function escapeIlike(word: string): string {
 8 |     return word.replace(/\\/g, "\\\\").replace(/%/g, "\\%").replace(/_/g, "\\_");
 9 | }

13 |  * Prevents comma injection by escaping special characters, in addition to
14 |  * the standard ILIKE wildcard escaping.

16 | export function escapePostgrest(val: string): string {
17 |     return escapeIlike(val).replace(/"/g, '""');

 1 | /**
 2 |  * Escape ILIKE wildcard characters in a string derived from untrusted input.
 3 |  * In PostgreSQL ILIKE patterns, % matches any sequence of characters and _
 4 |  * matches any single character. Leaving them unescaped causes overly broad
 5 |  * matches that may return far more rows than intended.
 6 |  */
 7 | export function escapeIlike(word: string): string {
 8 |     return word.replace(/\\/g, "\\\\").replace(/%/g, "\\%").replace(/_/g, "\\_");
 9 | }
10 | /**
11 |  * Escapes a value for safe use in PostgREST .or() filters.
12 |  * Prevents filter injection by escaping all characters with special
13 |  * meaning in PostgREST filter syntax.
14 |  */
15 | export function escapePostgrest(val: string): string {
16 |     return val.replace(/\\/g, "\\\\").replace(/"/g, '""').replace(/%/g, "\\%").replace(/_/g, "\\_");
17 | }
18 |

20 |     return words

 3 | export function estimateTokens(text: string): number {
 4 |     const words = text.trim().split(/\s+/).filter(Boolean).length;
 5 |     return Math.ceil(words * 1.33);
 6 | }
 7 | 
 8 | export function trimHistoryByTokens(
 9 |     messages: ChatMessage[],
10 |     maxTokens: number
11 | ): { trimmedMessages: ChatMessage[]; droppedMessages: ChatMessage[] } {
12 |     try {
13 |         const trimmed: ChatMessage[] = [];
14 |         const dropped: ChatMessage[] = [];
15 |         let currentTokens = 0;
16 | 
17 |         for (let i = messages.length - 1; i >= 0; i--) {
18 |             const msg = messages[i];
19 |             const text = msg.text || msg.content || "";
20 |             const tokens = estimateTokens(text);
21 |             const msgTokens = tokens + 4; // overhead buffer
22 | 
23 |             if (currentTokens + msgTokens > maxTokens && trimmed.length > 0) {
24 |                 dropped.unshift(msg);
25 |                 continue;
26 |             }
27 | 
28 |             currentTokens += msgTokens;
29 |             trimmed.unshift(msg);
30 |         }
31 | 
32 |         return { trimmedMessages: trimmed, droppedMessages: dropped };
33 |     } catch {
34 |         return { trimmedMessages: messages.slice(-50), droppedMessages: [] };
35 |     }
